+/// A lazy kernel's blocks: its route counts, its pool's peak over its letters, and the bytes
+/// of words it kept, a letter (summed over the admitted blocks).
+fn lazy<const V: u8>(data: &[u8]) {
+    let (mut shares, mut admitted) = (Vec::new(), 0);
+    let (mut letters, mut bytes, mut rules) = (0usize, 0usize, 0usize);
+    let s = run::<CellsVariant<V>>(data, |s| {
+        let c = s.cells_counts();
+        if c.admitted > admitted {
+            admitted = c.admitted;
+            let (slots, n) = s.lazy_pool();
+            shares.push(slots as f64 / n as f64);
+            let (b, r) = s.lazy_words();
+            (letters, bytes, rules) = (letters + n, bytes + b, rules + r);
+        }
+    });
+    shares.sort_by(|a, b| a.partial_cmp(b).unwrap());
+    println!("{:?}", s.cells_counts());
+    println!(
+        "pool's peak over letters, min median max: {:.3} {:.3} {:.3}",
+        shares[0],

+        shares[shares.len() - 1]
+    );
+    println!(
+        "words kept, bytes a letter: as bytes {:.2}, as rules {:.2}, both {:.2} ({} letters)",
+        bytes as f64 / letters as f64,
+        rules as f64 / letters as f64,
+        (bytes + rules) as f64 / letters as f64,
+        letters
+    );

+/// The last eight words the lazy induction made (`LAZY_RING`), each by its key: its two
+/// factors' offsets, 32 bits each. Equal keys are equal words. The search is over the keys

+/// 2^31.)
+#[repr(C, align(64))]

+    folded: [u32; 8],
+    words: [u32; 8],
+    keys: [u64; 8],

+impl Ring {
+    fn new() -> Ring {

+            folded: [0; 8],
+            words: [0; 8],

+    #[inline(always)]

+        #[cfg(all(target_arch = "x86_64", not(miri)))]

+        // bytes, which its alignment makes aligned loads

+            let a = _mm_cmpeq_epi32(f, _mm_load_si128(p));

+/// value are found at the first's place, and are not equal.
+#[test]
+fn cells_lazy_ring_folded_keys() {
+    let fold = |key: u64| key as u32 ^ ((key >> 32) as u32).rotate_left(16);
+    let mut ring = Ring::new();
+    // nothing kept: a folded 0 is found (the ring starts as zeros), and its key is no word's
+    assert_eq!(ring.find(0), 0);
+    assert_eq!(ring.keys[0], u64::MAX);
+    assert_eq!(ring.find(7), 8);
+    let (a, b) = (0x0000_0005_0000_0003u64, 0x0005_0000_0000_0003 >> 16 << 16 | 3);
+    // (two different keys of one folded value)
+    let b = if fold(a) == fold(b) { b } else { (0x0001_0000u64) << 32 | (fold(a) ^ 1) as u64 };
+    assert!(a != b && fold(a) == fold(b), "{a:x} {b:x}");
+    ring.folded[5] = fold(a);
+    ring.keys[5] = a;
+    assert_eq!(ring.find(fold(b)), 5);
+    assert_ne!(ring.keys[5], b);
+    // the first of several
+    ring.folded[2] = fold(a);
+    assert_eq!(ring.find(fold(a)), 2);
+    ring.folded[7] = 99;
+    assert_eq!(ring.find(99), 7);
+}
