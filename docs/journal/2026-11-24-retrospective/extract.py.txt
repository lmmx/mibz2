def git(*a): return subprocess.run(['git', '-C', REPO] + list(a), capture_output=True, text=True).stdout

# ---- dated documents: every version of every journal / charter / record markdown file
DOC_REFS = [('origin/master', 'docs/journal'), ('origin/master', 'docs/team/research-charter'),
            ('origin/pr497', 'docs/journal/monodromy')]
versions = collections.defaultdict(list)   # (ref, path) -> [(time, commit)]
for ref, d in DOC_REFS:
    out = git('log', '--format=@%H %aI', '--name-only', ref, '--', d)
    cur = None
    for l in out.splitlines():
        if l.startswith('@'):
            h, t = l[1:].split(); cur = (h, P(t))
        elif l.endswith('.md') and cur:
            versions[(ref, l)].append((cur[1], cur[0]))
for k in versions: versions[k].sort()
_cat = subprocess.Popen(['git', '-C', REPO, 'cat-file', '--batch'], stdin=subprocess.PIPE, stdout=subprocess.PIPE)
_blob = {}
def show(commit, path):
    key = (commit, path)
    if key in _blob: return _blob[key]
    _cat.stdin.write(f'{commit}:{path}\n'.encode()); _cat.stdin.flush()
    hdr = _cat.stdout.readline().decode()
    if hdr.endswith('missing\n'): _blob[key] = ''; return ''
    n = int(hdr.split()[2]); data = _cat.stdout.read(n); _cat.stdout.read(1)
    _blob[key] = data.decode('utf-8', 'replace'); return _blob[key]
def doc_before(ref, path, cutoff):
    vs = [v for v in versions[(ref, path)] if v[0] < cutoff]
    return (vs[-1], show(vs[-1][1], path)) if vs else (None, None)
def doc_first(path):
    for ref, _ in DOC_REFS:
        if versions.get((ref, path)): return versions[(ref, path)][0]
    return None

MEAS = re.compile(r'\d[\d,.]*\s?(%|×|x\b|ns|µs|ms|MB|MiB|KiB|GB/s|MB/s|Ir|B/B|c/B|cycles|instructions)|\b[01]\.\d{2,3}\b|\[\s?[01]\.\d+')
OUTCOME = re.compile(r'NOT ADOPTED|ADOPTS|ADOPTED|\bKILL|\bmerged\b|\bclosed\b|[Cc]losing|not shown|[Vv]erdict|[Rr]esult|regression|slower|faster|FASTER|SLOWER|\bpass(ed|es)?\b|\bfail|\bwon\b|\blost\b|\bwin\b|negative|confirms?|refuted|falsified|INCONCLUSIVE|HELD|null\b|stays|superseded|[Ss]hipped', re.I)
PREMISE = re.compile(r'\b(predict\w*|expect\w*|hypothes\w*|premise|assum\w*|estimat\w*|priced|model(led|s)?|should|would|because|so that|the gain|saves?|cuts?|removes?|fewer|smaller|binds?|bound by|bottleneck|dominat\w*|costs? (about|only)|is (about|nearly) \d)', re.I)
LEAK = re.compile(r'NOT ADOPTED|\bKILL\b|closed unmerged|\bstands\b|\bVOID\b|failed|\blost\b|refuted|falsified|merged as|superseded|was slower|not shown|NO.GO|SHELVE', re.I)
def scrub(t):
    return '; '.join(x for x in re.split(r';\s*', t) if not LEAK.search(x))

def sentences(text):
    text = re.sub(r'<!--.*?-->', ' ', text or '', flags=re.S)
    text = re.sub(r'\s+', ' ', text)
    return [s.strip() for s in re.split(r'(?<=[.;:!?])\s+(?=[A-Z*(`-])', text) if 20 < len(s.strip()) < 600]

def clip(s, n=330): s = s.replace('\t', ' ').replace('|', '\\|'); return s if len(s) <= n else s[:n - 1] + '…'
def tsv(s): return (s or '').replace('\t', ' ').replace('\n', ' / ')

def pr_link(n): return f'https://github.com/lmmx/mibz2/pull/{n}'
def blob_link(ref, path, commit=None):
    if commit: return f'https://github.com/lmmx/mibz2/blob/{commit[:12]}/{path}'
    br = 'master' if ref == 'origin/master' else 'pull/497'
    return f'https://github.com/lmmx/mibz2/blob/{br}/{path}' if br == 'master' else f'{path} (PR #497 branch)'

def match_bundles(recipes):
    out = []
    for r in recipes or []:
        for b in bundles:
            d = b['path'].split('/')[0]
            if ('/' in r and b['path'].startswith(r)) or ('/' not in r and d == r):
                out.append(b)
    return out

ROLE = re.compile(r'^FROM:\s*(\S+(?: \S+)*)', re.M)
def from_role(body):
    m = ROLE.search(body or ''); return m.group(1).strip().lower() if m else ''

def verdict_text(ref):
    kind, _, rest = ref.partition(':')
    if kind == 'merged':
        p = prs[int(rest)]; return (P(p['merged_at']), f'merged {p["merged_at"]}', pr_link(int(rest)))
    if kind == 'c145':
        c = cbyid[int(rest)]; return (c['t'], c['body'], c.get('html_url'))
    if kind == 'pr':
        n, cid = rest.split(':'); n = int(n); cid = int(cid)
        if cid == 0:
            p = prs[n]; return (None, p['body'] or '', pr_link(n))
        c = cbyid[cid]; return (c['t'], c['body'], c['html_url'])
    if kind == 'journal':
        v = doc_first(rest); txt = open(os.path.join(REPO_WT, rest)).read() if os.path.exists(os.path.join(REPO_WT, rest)) else ''
        lv = None
        for k in versions:
            if k[1] == rest and versions[k]: lv = versions[k][-1]
        return (v[0] if v else None, txt, f'https://github.com/lmmx/mibz2/blob/master/{rest}')
    return (None, '', '')
REPO_WT = OUT.split('/docs/journal/')[0]

def verdict_quote(body):
    s = sentences(body)
    keys = re.compile(r'NOT ADOPTED|ADOPTS|KILL|[Vv]erdict|[Cc]losing|[Cc]losed|[Rr]esult|not shown|merged|FAIL|PASS|INCONCLUSIVE|stays|no trim|SHELVE|none adopts|did not|lost|NO.GO', re.I)
    hit = [x for x in s if keys.search(x)]
    return ' '.join((hit or s)[:3])

shared_first = collections.Counter(v.get('first_commit') for v in prmeta.values())
def latest_distinct(hits, k):
    seen, out = set(), []
    for h in reversed(hits):
        if h[3] in seen: continue
        seen.add(h[3]); out.append(h)
        if len(out) == k: break
    return list(reversed(out))

          f'- **commit_to_build:** {iso(ctb[0])}, {ctb[1]} ({ctb[2]})',
          f'- **Cost:** i9 {cost["i9_total_min"]} min [{cost["i9"]}]; CI: {cost["ci"]}; calendar days, commit to verdict: {cost["days"]}; agent runs/tokens: {cost["agent"]}',
          '', '**claimed_gain_source** (quoted):', ''] + [f'- {g}' for g in gain_q or ['not found by the extractor']] + \
         ['', '**must_hold** (quoted):', ''] + [f'- {h}' for h in hold_q or ['not found by the extractor']] + \
         ['', '**Premises** (quoted from the sources dated at or before commit_to_build; PR bodies are their current text):', ''] + \
         [f'- [{iso(t)}] "{clip(s, 400)}" ({l})' for t, l, s in prem or []] + ([] if prem else ['- none extracted']) + \
         ['', f'**same_quantity_facts:** {len(hits)} matching lines in {ndocs} documents dated before commit_to_build, searched on `{c["kw"]}` with a measurement on the line. All are in `facts.tsv`; the latest line from each of the 10 latest documents:', ''] + \
         [f'- [{iso(h[0])}] {h[3]} ({h[1]}): "{clip(h[4], 300)}" — {h[2]}' for h in latest_distinct(hits, 10)] + ([] if hits else ['- none']) + \
         ['', f'**Counterfactual real-block i9 measurements (≤ 10 min) before commit_to_build, matching the terms:** {"; ".join(cf) or "none found"}', '',
          f'**Review routing:** ' + ' / '.join(routing), '',
          f'**monodromy_involved:** {row["monodromy_involved"]}', '',
          f'**coordinator_involved:** {row["coordinator_involved"]}', '']
    if row['notes']: md += [f'**Notes:** {row["notes"]}', '']
    mdparts.append('\n'.join(md))

cols = list(rows[0].keys())
with open(f'{OUT}/cases.tsv', 'w') as f:
    f.write('\t'.join(cols) + '\n')
    for r in rows: f.write('\t'.join(tsv(str(r[k])) for k in cols) + '\n')
with open(f'{OUT}/facts.tsv', 'w') as f:
    f.write('id\tdate\tsource\tlink\tline\n')
    for r in facts: f.write('\t'.join(r) + '\n')

    for r in sealed: f.write('\t'.join(r) + '\n')

    f.write('# Retrospective cases: the extracted facts\n\n'

            'which coders do not open. Method and limits are in `README.md`.\n\n')
    f.write('\n'.join(mdparts))
