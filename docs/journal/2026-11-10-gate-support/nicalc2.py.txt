import json, math, sys
import numpy as np
from scipy import stats
plan = json.load(open(sys.argv[1])); menu = json.load(open(sys.argv[2]))["cells"]
CELLS = sorted(plan["cells"])
LRV = np.array([max(b["lrv"] for b in menu[c]["batches"]) for c in CELLS])
TAU = {n: np.array([plan["cells"][c][n]["tau"] for c in CELLS]) for n in ("strict", "all")}
PER_ROUND = 963.04 / 2880; SETUP = 135.0
def pw(name, R, k, true=1.005, mult=1.0, keep=None):
    sd = np.sqrt((mult*TAU[name])**2 + LRV/R)
    if keep is not None: sd = sd[keep]
    p = stats.nct.sf(stats.t.ppf(0.95, k-1), k-1, math.log(1.02/true)*math.sqrt(k)/sd)
    return max(0.0, 1-(1-p).sum()), float(p.prod()), float(p.min())
print("fine grid, lower bound strict / all (minutes)")
for R in (160, 200, 240, 320, 480):
    print(f" R{R}: " + " | ".join(f"k{k} {pw('strict',R,k)[0]:.3f}/{pw('all',R,k)[0]:.3f} ({k*(SETUP+PER_ROUND*R):.0f})" for k in (9,10,11,12)))
print("true ratio sensitivity at R240, strict/all lower bound")
for true in (1.000, 1.0025, 1.005, 1.0075, 1.010):
    print(f" true {true}: " + " | ".join(f"k{k} {pw('strict',240,k,true)[0]:.3f}/{pw('all',240,k,true)[0]:.3f}" for k in (6,8,10,11,12)))
hot = [CELLS.index(c) for c in ("pattern/4/cpu","pattern/4/wall","text/4/wall","random/2/wall")]
keep = np.array([i for i in range(len(CELLS)) if i not in hot])
print("without the four high-tau cells (strict), R240:", [(k, round(pw('strict',240,k,keep=keep)[0],3)) for k in (4,6,8,10)])
for name in ("strict","all"):
    sd = np.sqrt(TAU[name]**2 + LRV/240)
    sp = math.sqrt((sd**2).sum())/len(sd)
    print(name, "pooled mean of 42 cells, independent: sd", round(sp,5),
          [(k, round(float(stats.nct.sf(stats.t.ppf(0.95,k-1),k-1,math.log(1.02/1.005)*math.sqrt(k)/sp)),4)) for k in (3,4,6)])
print("one cell hidden in a pooled geometric mean at 1.02:", round(1.02**42,3), "; at pooled 1.005 truth elsewhere 1.0:", round(1.005**42,3))
print("t quantiles", {k: round(float(stats.t.ppf(0.95,k-1)),3) for k in (6,8,10,11,12)})
# slowest shard minutes at R (records 2.22 s/round), pilot pace and x1.75
for R in (240,480): print("R",R,"records minutes", round(2.22*R/60,1), round(2.22*R/60*1.75,1), "+15 setup")
# validity exposure
for q in (0.001,0.002,0.005,0.01):
    print("q",q, {k: round((1-q)**(9*k),3) for k in (8,10,11,12,16)})
