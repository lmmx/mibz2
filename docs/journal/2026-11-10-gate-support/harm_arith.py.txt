"""#324's fallback harm: the dose and price arithmetic, from archived figures only.

Sources: agg3's corrected counts (research/run-front-fallback @ 72eb8af4, pad 0, one 890,000-byte
block each of `random` and `records`, columns master / head); #324's three runs' intervals
(docs/journal/2026-11-10-run-front-production.md §6); cachegrind's simulated conditional
mispredicts in `mtf_avx2` (master, local build, the same two blocks); the native probe's share
of launches whose frame meets `counts` (22 of 64 by geometry; 20/60 and 25/60 measured).
"""
import math
N = 890_000
agg = {  # class: (random master, random head, records master, records head)
    "D<-H 4K (counts -> STEPS)": (13_829, 56_154, 14_255, 26_933),
    "S<-H 4K (counts -> frame)": (46_146, 787_938, 54_358, 868_196),
    "H<-S 4K (frame -> counts)": (48_608, 193_967, 48_608, 194_399),
    "read_item S<-S 1K": (17_836, 95, 17_835, 74),
}
print("excess candidates a byte, head - master: random, records, records/random")
ex = {}
for k, (a, b, c, d) in agg.items():
    ex[k] = ((b - a) / N, (d - c) / N)
    print(f"  {k:28s} {ex[k][0]:+.4f} {ex[k][1]:+.4f} {ex[k][1] / ex[k][0]:.2f}")
mis = (525_180 / N, 243_265 / N)
ir = (84_474_095 / 899_981, 95_404_355 / 899_980)
print(f"mtf_avx2 simulated conditional mispredicts a byte: {mis[0]:.3f} {mis[1]:.3f} ratio {mis[1] / mis[0]:.2f}")
print(f"instructions a byte (decode_block): {ir[0]:.1f} {ir[1]:.1f} ratio {ir[1] / ir[0]:.2f}")

runs = {  # run: (random@1 %, lo, hi, records@1 %, lo, hi)
    "run 1 (Xeon 8370C)": (1.3, -1.1, 3.8, -0.6, -2.7, 1.5),
    "run 2 (EPYC 7763)": (3.4, 2.2, 4.6, 1.5, 0.2, 2.9),
    "run 3 (EPYC 9V45)": (5.3, 4.5, 6.1, 2.1, 1.0, 3.3),
}
T9 = 2.262
p = 22 / 64
dE = 1.0  # candidates a byte between a launch whose frame meets `counts` and one whose doesn't (0.99-1.27 archived)
print(f"\nshare of launches meeting: {p:.3f}; SD of the arms' exposure difference: {math.sqrt(2 * p * (1 - p)) * dE:.3f} candidates a byte")
for name, (h, lo, hi, r, rlo, rhi) in runs.items():
    sd = (hi - lo) / 2 / T9 * math.sqrt(10)
    line = f"{name}: round SD {sd:.2f}%"
    if lo > 0:
        ratio = r / h
        c = sd / math.sqrt(2 * p * (1 - p))      # central: all the round variance given to the class
        c2 = 2.4 * sd                              # two discordant rounds of ten, same sign (97.7%)
        need = h / ex["D<-H 4K (counts -> STEPS)"][0]
        need_lo = lo / ex["D<-H 4K (counts -> STEPS)"][0]
        line += (f"; records/random harm {ratio:.2f} ({rlo / hi:.2f}-{rhi / lo:.2f})"
                 f"; a frame pair costs at most {c / dE:.2f}% of the decode a candidate/byte (central), {c2 / dE:.2f}% (conservative)"
                 f"; counts->STEPS needs {need:.0f}% ({need_lo:.0f}% at the interval's low end)"
                 f": {need / (c / dE):.0f}x central, {need_lo / (c2 / dE):.0f}x conservative"
                 f"; at a frame pair's price it explains {ex['D<-H 4K (counts -> STEPS)'][0] * c / dE:.2f}% (central), "
                 f"{ex['D<-H 4K (counts -> STEPS)'][0] * c2 / dE:.2f}% (conservative) of {h}%"
                 f"; per simulated mispredict: {h / mis[0]:.1f}% of the decode a mispredict/byte")
    print(line)
print("P(at most one discordant round of ten):", round((1 - 2 * p * (1 - p)) ** 10 + 10 * 2 * p * (1 - p) * (1 - 2 * p * (1 - p)) ** 9, 4))
