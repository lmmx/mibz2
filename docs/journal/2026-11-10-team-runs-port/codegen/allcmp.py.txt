"""Every mibz2 function present in both builds of the harness: is its machine code the same
(addresses masked)? Usage: python3 -I allcmp.py BASE_AB HEAD_AB"""
import subprocess, sys, re
def syms(path):
    out = subprocess.run(["nm", "-C", "-S", "--defined-only", path], capture_output=True, text=True).stdout
    d = {}
    for l in out.splitlines():
        p = l.split(None, 3)
        if len(p) == 4 and p[2] in "tT" and "mibz2::" in p[3]:
            d.setdefault(p[3], (int(p[0], 16), int(p[1], 16)))
    return d
def dis(path):
    out = subprocess.run(["objdump", "-d", "--no-show-raw-insn", path], capture_output=True, text=True).stdout
    code = {}
    for l in out.splitlines():
        m = re.match(r"\s*([0-9a-f]+):\s+(.*)", l)
        if m:
            t = m.group(2)
            t = re.sub(r"\b[0-9a-f]{4,}\b( <[^>]*>)?", "ADDR", t)
            t = re.sub(r"0x[0-9a-f]+\(%rip\)", "RIP", t)
            t = re.sub(r"#.*", "", t).strip()
            code[int(m.group(1), 16)] = t
    return code
def body(code, addrs, a, n):
    return [code[x] for x in addrs if a <= x < a + n]
import bisect
res = []
A, B = syms(sys.argv[1]), syms(sys.argv[2])
ca, cb = dis(sys.argv[1]), dis(sys.argv[2])
ka, kb = sorted(ca), sorted(cb)
def get(code, keys, a, n):
    i = bisect.bisect_left(keys, a); j = bisect.bisect_left(keys, a + n)
    return [code[k] for k in keys[i:j]]
same = diff = 0
differing = []
for name in sorted(set(A) & set(B)):
    x, y = get(ca, ka, *A[name]), get(cb, kb, *B[name])
    if x == y: same += 1
    else:
        diff += 1; differing.append((name, len(x), len(y)))
print(f"in both: {len(set(A)&set(B))}; identical {same}; different {diff}; only in base {len(set(A)-set(B))}; only in head {len(set(B)-set(A))}")
for n, x, y in differing: print(f"  DIFFERENT {x:>5} -> {y:<5} {n[:120]}")
for n in sorted(set(A) - set(B)): print(f"  only in base: {n[:120]}")
