env = dict(os.environ, CARGO_TARGET_DIR=os.path.join(wt, "target"))
def run(cmd):
    p = subprocess.run(cmd, cwd=wt, env=env, capture_output=True, text=True, timeout=3000)
    tail = (p.stdout + p.stderr)
    return p.returncode, tail

def failed_tests(text):
    return sorted({l.split()[1] for l in text.splitlines() if l.startswith("test ") and l.rstrip().endswith("FAILED")})

only = sys.argv[3:]  # optional: names to run
with open(out, "a") as f:
    for name, path, old, new, occ, of, suites in M:
        if only and name not in only:
            continue
        full = os.path.join(wt, path)
        src = open(full).read()
        if src.count(old) != of:
            f.write(f"{name}\tSKIPPED\tfound {src.count(old)} of {of}\n"); f.flush(); continue
        i = -1
        for _ in range(occ + 1):
            i = src.index(old, i + 1)
        open(full, "w").write(src[:i] + new + src[i + len(old):])
        t = time.time()
        try:
            verdict, why = "SURVIVED", ""
            for kind, cmd in (("L", LIB), ("I", INTEG), ("S", SLOW)):
                if kind == "L" and "L" not in suites: continue
                if kind == "I" and "I" not in suites: continue
                rc, text = run(cmd)
                if rc != 0:
                    ft = failed_tests(text)
                    if not ft and "error" in text and "could not compile" in text:
                        verdict, why = "UNBUILDABLE", text[-300:].replace("\n", " ")
                    else:
                        verdict, why = "KILLED", (kind + ": " + ", ".join(ft)[:400]) if ft else kind + ": nonzero exit"
                    break
        finally:
            open(full, "w").write(src)
        f.write(f"{name}\t{verdict}\t{why}\t{time.time()-t:.0f}s\n"); f.flush()
    f.write("END\n")
