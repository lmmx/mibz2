#!/bin/bash
# final verification of the port at its local commit; one log
cd SCRATCH/port
L=SCRATCH/verify.log
: > $L
run() { echo "== $*" >> $L; "$@" >> $L 2>&1; echo "== exit $? : $*" >> $L; }
export CARGO_TARGET_DIR=SCRATCH/port/target
run cargo test
run cargo test --release
run cargo test --features walk-stats --lib coop::
run cargo test --example ab
run cargo build --profile fuzz --example fuzz
for m in classic pipeline service; do run SCRATCH/port/target/fuzz/examples/fuzz 30 1 SCRATCH/fuzz2-$m $m; done
export CARGO_TARGET_DIR=SCRATCH/port/target-miri
run cargo +nightly miri test --lib -- miri_runs miri_build_runs
export RUSTFLAGS="-C target-feature=+avx2,+bmi1,+bmi2,+lzcnt,+popcnt"
export CARGO_TARGET_DIR=SCRATCH/port/target-miri-avx2
run cargo +nightly miri test --lib -- miri_runs miri_build_runs
echo DONE >> $L
