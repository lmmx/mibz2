# Record 121 check: the MTF + RLE2 + run-record stage has an exact chunk summary.
# Symbols as in bzip2: 0 = RUNA, 1 = RUNB (bijective base 2 run of MTF index 0), s >= 2 = MTF index s-1.
# A chunk is decoded against a list of 256 placeholders. Its summary is
#   (lead digits (S, q), symbolic run records, permutation, trailing digits (S, q)).
# The stitch is a serial fold over chunks: a carry (N, k), the live list, a prefix sum.
import random
def serial(sym, lst):
    lst=list(lst); out=[]; N=0; k=0
    for s in sym:
        if s<2: N+=(s+1)<<k; k+=1
        else:
            if N: out.append([lst[0],N]) if not out or out[-1][0]!=lst[0] else out[-1].__setitem__(1,out[-1][1]+N)
            N=0; k=0
            b=lst.pop(s-1); lst.insert(0,b); out.append([b,1])
    if N:
        if out and out[-1][0]==lst[0]: out[-1][1]+=N
        else: out.append([lst[0],N])
    return out
def chunk(sym):
    lst=list(range(256)); i=0; S=0; q=0
    while i<len(sym) and sym[i]<2: S+=(sym[i]+1)<<q; q+=1; i+=1
    lead=(S,q); recs=[]; N=0; k=0
    for s in sym[i:]:
        if s<2: N+=(s+1)<<k; k+=1
        else:
            if N: recs[-1][1]+=N               # a run of index 0 extends the record just emitted
            N=0; k=0
            b=lst.pop(s-1); lst.insert(0,b); recs.append([b,1])
    return lead,recs,lst,(N,k),(i==len(sym))
def stitch(chunks, lst0):
    live=list(lst0); out=[]; N=0; k=0
    def flush():
        nonlocal N,k
        if N:
            if out and out[-1][0]==live[0]: out[-1][1]+=N
            else: out.append([live[0],N])
        N=0; k=0
    for (S,q),recs,perm,(tN,tk),alldigits in chunks:
        N+=S<<k; k+=q                          # lead digits continue the carried run
        if alldigits: continue
        flush()
        out.extend([live[b],ln] for b,ln in recs)   # one table lookup per run record
        live=[live[p] for p in perm]
        N,k=tN,tk
    flush(); return out
random.seed(3); cases=0
for t in range(3000):
    n=random.randint(1,400)
    sym=[random.choice([0,0,0,1,1]+[random.randint(2,6)]*3+[random.randint(2,256)]) for _ in range(n)]
    lst0=list(range(256)); random.shuffle(lst0)
    cuts=sorted(random.sample(range(n+1),min(n+1,random.randint(0,8))))
    parts=[sym[a:b] for a,b in zip([0]+cuts,cuts+[n])]
    assert stitch([chunk(p) for p in parts],lst0)==serial(sym,lst0); cases+=1
print(cases,"streams: chunked symbolic decode + stitch equals serial decode")
