# Frozen predictions for the 0.5.5 recalibration sweep (MONODROMY record 56). Model: records 46/49/52/53.
import itertools
def wall(W, N, pinned, beta, nu_k, kernel="lazy", R7=2.59):
    if pinned:
        cap = N - nu_k*W - 0.125*N
        wc = min(0.93*W, cap); run = wc/0.93
        c = 1 + beta*W*run; t = c/wc
    else:
        u = 0.941 - 0.0103*(W-7); run = W*u/0.93
        c = 1 + beta*W*run; t = c/(W*u)
    # DRAM: wide traffic 4.42 + 0.074*W*W_run B/B; throughput anchored at wide 7 unpinned = 2.59 GB/s
    if kernel=="lazy":
        u7=0.941; t7=(1+beta*7*7*u7/0.93)/(7*u7)
        R=R7*t7/t; dram=(4.42+0.074*W*run)*R; Bmax=33.4-0.25*max(0,W-4)
        util=dram/Bmax; pen=1+max(0.0,(util-0.80)/0.10)*0.038
        return t*pen, util
    return t, None
BETA={"lazy":(0.0088,0.0116,0.0150),"eager":(0.020,0.0244,0.030)}
NU=(0.14,0.18,0.22)
cells=[]  # (threads, pinned, shipped W, kernel, arms)
cells+= [(10,True,7,"lazy",[6,8,10]),(8,True,6,"lazy",[5,7,8]),(6,True,5,"eager",[4,6]),(4,True,3,"eager",[2,4])]
cells+= [(4,False,4,"eager",[3,5]),(6,False,6,"eager",[5,7]),(8,False,8,"lazy",[7,9]),(10,False,7,"lazy",[6,8,10])]
for T in (12,16,20): cells.append((T,False,7,"lazy",[6,8,10]))
for T,pin,Ws,ker,arms in cells:
    N=T if pin else 20
    rows=[]
    allW=sorted(set(arms+[Ws]))
    nom={W:wall(W,N,pin,BETA[ker][1],NU[1],ker) for W in allW}
    best=min(allW,key=lambda W:nom[W][0])
    out=[]
    for W in arms:
        rs=[wall(W,N,pin,b,v,ker)[0]/wall(Ws,N,pin,b,v,ker)[0] for b in BETA[ker] for v in NU]
        r=nom[W][0]/nom[Ws][0]
        lo,hi=min(rs)-0.02,max(rs)+0.02
        out.append("W%d: %.3f [%.2f, %.2f]"%(W,r,lo,hi))
    util=nom[Ws][1]
    print("T=%2d %-8s shipped %d (%s) | best %d | %s | DRAM util at shipped %s"%(T,"pinned" if pin else "unpinned",Ws,ker,best," · ".join(out),"%.0f%%"%(100*util) if util else "-"))
