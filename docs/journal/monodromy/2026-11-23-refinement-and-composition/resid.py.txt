"""Residual geometry of Open's first-return map T_S (S = rows of runs of width <= TAU, plus p),
and of its composites T_S^k. Measured on the marked LF cycle.
- geometric pieces of T^k: maximal intervals of adjacent retained rows with equal k-return words
  (equal words <=> the rows travel together; see record 92 Lemma 3);
- rank pieces of g^k: maximal runs of consecutive retained ranks whose k-images are consecutive ranks
  (transport rigid in rank coordinates; words may differ);
- distinct k-return words (what memoisation across non-adjacent pieces could share)."""
import json, sys, os
B = sys.argv[1]; TAU = int(sys.argv[2]) if len(sys.argv) > 2 else 16; KS = range(1, 9)
man = json.load(open(os.path.join(B, 'manifest.json')))
for blk in man['blocks']:
    L = open(os.path.join(B, blk['file']), 'rb').read(); n = len(L); p = blk['primary_index']
    cnt = [0]*256
    for b in L: cnt[b] += 1
    C = [0]*256; s = 0
    for c in range(256): C[c] = s; s += cnt[c]
    occ = C[:]; LF = [0]*n
    for x in range(n):
        c = L[x]; LF[x] = occ[c]; occ[c] += 1
    inS = bytearray(n); x = 0
    heads = [0] + [y for y in range(1, n) if L[y] != L[y-1]]; ends = heads[1:] + [n]
    for d, e in zip(heads, ends):
        if e - d <= TAU:
            for y in range(d, e): inS[y] = 1
    inS[p] = 1
    # cycle order from p
    cyc = [0]*n; pos = [0]*n; y = p
    for i in range(n):
        cyc[i] = y; pos[y] = i; y = LF[y]
    assert y == p
    seq = bytes(L[y] for y in cyc)
    Sidx = [i for i in range(n) if inS[cyc[i]]]           # cycle positions of retained rows
    m = len(Sidx)
    nxt = {}                                              # cycle position -> next retained cycle position
    for a, b in zip(Sidx, Sidx[1:] + [Sidx[0] + n]): nxt[a] = b
    Srows = sorted(cyc[i] for i in Sidx); rank = {row: j for j, row in enumerate(Srows)}
    def kimg(i, k):                                       # cycle pos after k returns (unwrapped)
        for _ in range(k): i = nxt[i % n] + (i // n) * n
        return i
    seq2 = seq + seq
    print(f"{blk['file']}: n {n} m {m} ({m/n:.3f} n)", flush=True)
    for k in KS:
        geo_pieces = 0; widths = []; w = 0; rank_pieces = 0
        prev = None; words = set()
        imgs = {}
        for row in Srows:
            i = pos[row]; j = kimg(i, k); imgs[row] = j
            words.add(seq2[i:j] if j - i <= 4096 else hash(seq2[i:j]))
        for jr, row in enumerate(Srows):
            i = pos[row]; wd = seq2[i:imgs[row]]
            if prev is not None and row == prev[0] + 1 and wd == prev[1]:
                w += 1
            else:
                if prev is not None: widths.append(w)
                geo_pieces += 1; w = 1
            prev = (row, wd)
        widths.append(w)
        # rank pieces
        rimg = [rank[cyc[imgs[row] % n]] for row in Srows]
        rank_pieces = 1 + sum(1 for a in range(1, m) if rimg[a] != rimg[a-1] + 1)
        rw = sum(x*x for x in widths) / m
        print(f"  k {k}: geometric pieces {geo_pieces:7d} ({geo_pieces/m:.3f} m), mean width {m/geo_pieces:.2f}, row-weighted {rw:.2f}, max {max(widths)};"
              f" rank pieces {rank_pieces:7d} ({rank_pieces/m:.3f} m); distinct words {len(words):7d} ({len(words)/m:.3f} m)", flush=True)
