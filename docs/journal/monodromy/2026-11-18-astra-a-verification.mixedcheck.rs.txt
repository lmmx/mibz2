//! Independent check of #575 (mixed Cells steps): an abstract two-order induction (doubly linked
//! lists), run scalar and with the joint substitution applied directly from its table.
//! Research only; not built by the crate.
use mibz2::stream::{Event, Scanner, SliceSource};
use mibz2::{mtf, Kernels};
const NIL: u32 = u32::MAX;
#[derive(Default, Clone, Copy, Debug, PartialEq)]
struct Counts { steps: u64, t: u64, f: u64, m: u64, h: u64, g: u64, h_te: u64, h_tb: u64, diff: u64, eq_long: u64, bytes: u64, sep: u64, glue: u64, adj_bot: u64, m_any: u64, h_any: u64, d_is_a: u64, h_touched: u64, trace: u64, h_tt: u64, same_dir: u64, bsteps: u64, old_lit_t: u64, prev: u8, rule_win: u64 }
struct Ind {
    // per letter (original or cell)
    w: Vec<u64>, tpos: Vec<u64>, bpos: Vec<u64>, hits: Vec<(u64, u64)>, word: Vec<u32>, orig: Vec<bool>, touched: Vec<bool>, byte: Vec<i16>,
    tp: Vec<u32>, tn: Vec<u32>, bp: Vec<u32>, bn: Vec<u32>, ttail: u32, btail: u32,
    // grammar: leaves (byte, marked) and products (l, r); len and marked per word
    gl: Vec<u32>, gr: Vec<u32>, glen: Vec<u64>, gmark: Vec<bool>, gbyte: Vec<i16>,
}
impl Ind {
    fn new(l: &[u8], p: usize) -> Ind {
        // letters: maximal runs of L, the run holding row p split around it; top = L order
        let n = l.len(); let mut lets: Vec<(usize, usize, bool)> = Vec::new(); // (start, len, marked)
        let mut x = 0; while x < n { let mut y = x + 1; while y < n && l[y] == l[x] { y += 1; }
            if p >= x && p < y { if p > x { lets.push((x, p - x, false)); } lets.push((p, 1, true)); if y > p + 1 { lets.push((p + 1, y - p - 1, false)); } }
            else { lets.push((x, y - x, false)); }
            x = y; }
        let m = lets.len();
        let mut s = Ind { w: vec![], tpos: vec![], bpos: vec![], hits: vec![], word: vec![], orig: vec![], touched: vec![], byte: vec![], tp: vec![], tn: vec![], bp: vec![], bn: vec![], ttail: NIL, btail: NIL,
            gl: vec![], gr: vec![], glen: vec![], gmark: vec![], gbyte: vec![] };
        for (i, &(st, ln, mk)) in lets.iter().enumerate() {
            let g = s.gl.len() as u32; s.gl.push(NIL); s.gr.push(NIL); s.glen.push(1); s.gmark.push(mk); s.gbyte.push(l[st] as i16);
            s.w.push(ln as u64); s.tpos.push(st as u64); s.bpos.push(0); s.word.push(g); s.orig.push(true); s.touched.push(false); s.byte.push(l[st] as i16);
            s.tp.push(if i == 0 { NIL } else { i as u32 - 1 }); s.tn.push(if i + 1 == m { NIL } else { i as u32 + 1 });
            s.bp.push(NIL); s.bn.push(NIL);
        }
        s.ttail = m as u32 - 1;
        let mut ord: Vec<usize> = (0..m).collect(); ord.sort_by_key(|&i| l[lets[i].0]); // stable
        for (j, &i) in ord.iter().enumerate() { s.bp[i] = if j == 0 { NIL } else { ord[j - 1] as u32 }; s.bn[i] = if j + 1 == m { NIL } else { ord[j + 1] as u32 }; }
        let mut acc = 0u64; for &i in &ord { s.bpos[i] = acc; acc += lets[i].1 as u64; }
        s.btail = ord[m - 1] as u32; s
    }
    fn mk_word(&mut self, a: u32, b: u32) -> u32 { let g = self.gl.len() as u32; self.gl.push(a); self.gr.push(b);
        self.glen.push(self.glen[a as usize] + self.glen[b as usize]); self.gmark.push(self.gmark[a as usize] || self.gmark[b as usize]); self.gbyte.push(-1); g }
    fn new_letter(&mut self, w: u64, word: u32) -> u32 { let i = self.w.len() as u32; self.w.push(w); self.tpos.push(u64::MAX); self.bpos.push(u64::MAX); self.word.push(word); self.orig.push(false); self.touched.push(true); self.byte.push(-1);
        self.tp.push(NIL); self.tn.push(NIL); self.bp.push(NIL); self.bn.push(NIL); i }
    // list primitives, top (o = 0) and bottom (o = 1)
    fn links(&mut self, o: u8) -> (&mut Vec<u32>, &mut Vec<u32>, &mut u32) { if o == 0 { (&mut self.tp, &mut self.tn, &mut self.ttail) } else { (&mut self.bp, &mut self.bn, &mut self.btail) } }
    fn remove(&mut self, o: u8, x: u32) { let (p, nx, tail) = self.links(o); let (a, b) = (p[x as usize], nx[x as usize]);
        if a != NIL { nx[a as usize] = b; } if b != NIL { p[b as usize] = a; } else { *tail = a; } }
    fn insert_after(&mut self, o: u8, x: u32, y: u32) { let (p, nx, tail) = self.links(o); let b = nx[x as usize];
        p[y as usize] = x; nx[y as usize] = b; nx[x as usize] = y; if b != NIL { p[b as usize] = y; } else { *tail = y; } }
    fn replace(&mut self, o: u8, x: u32, y: u32) { self.insert_after(o, x, y); self.remove(o, x); }
    fn marked(&self, x: u32) -> bool { self.gmark[self.word[x as usize] as usize] }
    /// Run to the marked closure. joint: apply the two-cut table directly on a candidate.
    fn run(&mut self, joint: bool, c: &mut Counts) -> u32 {
        loop {
            let (a, b) = (self.ttail, self.btail);
            self.touched[a as usize] = true; self.touched[b as usize] = true;
            if self.tpos[a as usize] + self.w[a as usize] != self.bpos[b as usize] + self.w[b as usize] { c.glue += 1_000_000; }
            if a == b && self.tpos[a as usize] != self.bpos[a as usize] { c.glue += 1_000_000; }
            if a == b { c.prev = 0; self.remove(0, a); self.remove(1, a); if self.marked(a) { return self.word[a as usize]; } continue; }
            let (wa, wb) = (self.w[a as usize], self.w[b as usize]);
            c.steps += 1;
            c.trace = (c.trace ^ (wa.wrapping_mul(0x9E3779B97F4A7C15) ^ wb.rotate_left(29))).wrapping_mul(0x100000001B3);
            let dir = if wa > wb { 1u8 } else if wa < wb { 2 } else { 3 };
            if dir != 3 { let wn = if dir == 1 { a } else { b }; let g = self.word[wn as usize] as usize; if self.gl[g] != NIL && self.glen[g] > 16 && !self.gmark[g] { c.rule_win += 1; } }
            if dir != 3 && dir == c.prev { c.same_dir += 1; } c.prev = dir; if dir == 2 { c.bsteps += 1; }
            if wa > wb {
                c.t += 1;
                if self.orig[b as usize] && !self.marked(b) { c.old_lit_t += 1; }
                let d = self.bp[b as usize];
                let fresh = d != NIL && self.orig[d as usize] && !self.touched[d as usize];
                let s = wa - wb;
                if d == a { c.d_is_a += 1; }
                if d != NIL && d != a && s <= self.w[d as usize] {
                    c.m_any += 1;
                    let ua = self.word[a as usize];
                    if self.orig[b as usize] && self.orig[d as usize] && self.byte[b as usize] == self.byte[d as usize] && !self.marked(b) && !self.marked(d)
                        && !self.gmark[ua as usize] && self.glen[ua as usize] + 1 <= 16 { c.h_any += 1; if !fresh { c.h_touched += 1; } }
                }
                if fresh {
                    c.f += 1; let wd = self.w[d as usize];
                    if s > wd { let ua = self.word[a as usize];
                        if self.orig[b as usize] && self.byte[b as usize] == self.byte[d as usize] && !self.marked(b) && !self.marked(d) && !self.gmark[ua as usize] && self.glen[ua as usize] + 1 <= 16 { c.h_tt += 1; } }
                    if s <= wd {
                        c.m += 1; if s < wd { c.g += 1; }
                        let (ua, vb, wdd) = (self.word[a as usize], self.word[b as usize], self.word[d as usize]);
                        let lit_eq = self.orig[b as usize] && self.byte[b as usize] == self.byte[d as usize] && !self.marked(b) && !self.marked(d);
                        let plen = self.glen[ua as usize] + 1;
                        let short = !self.gmark[ua as usize] && lit_eq && plen <= 16;
                        if lit_eq && !short { c.eq_long += 1; } if !lit_eq { c.diff += 1; }
                        if short { c.h += 1; c.bytes += plen; if s == wd { c.h_te += 1 } else { c.h_tb += 1 } }
                        if joint {
                            let xw = self.mk_word(ua, vb);
                            c.trace = (c.trace ^ (s.wrapping_mul(0x9E3779B97F4A7C15) ^ wd.rotate_left(29))).wrapping_mul(0x100000001B3);
                            let yw = if short { c.steps += 1; xw } else { c.steps += 1; self.mk_word(ua, wdd) };
                            let x = self.new_letter(wb, xw); let y = self.new_letter(s, yw);
                            self.tpos[x as usize] = self.tpos[b as usize]; self.bpos[x as usize] = self.bpos[a as usize] + s;
                            self.bpos[y as usize] = self.bpos[a as usize];
                            self.tpos[y as usize] = self.tpos[d as usize] + (wd - s);
                            if short { self.hits.push((self.bpos[x as usize], plen)); }
                            // the table, at the old positions
                            self.remove(0, a); self.replace(0, b, x);
                            self.replace(1, a, y); self.insert_after(1, y, x); self.remove(1, b);
                            if s == wd { self.replace(0, d, y); self.remove(1, d); }
                            else { self.insert_after(0, d, y); self.w[d as usize] = wd - s; }
                            if short { if self.tn[y as usize] == x { c.glue += 1 } else { c.sep += 1 }
                                       if self.bn[y as usize] == x { c.adj_bot += 1 } }
                            continue;
                        }
                    }
                }
                let x = { let wd = self.mk_word(self.word[a as usize], self.word[b as usize]); self.new_letter(wb, wd) };
                self.tpos[x as usize] = self.tpos[b as usize]; self.bpos[x as usize] = self.bpos[a as usize] + s;
                self.replace(0, b, x); self.w[a as usize] = s; self.remove(1, b); self.insert_after(1, a, x);
            } else if wa < wb {
                let x = { let wd = self.mk_word(self.word[a as usize], self.word[b as usize]); self.new_letter(wa, wd) };
                self.bpos[x as usize] = self.bpos[a as usize]; self.tpos[x as usize] = self.tpos[b as usize] + (wb - wa);
                self.replace(1, a, x); self.w[b as usize] = wb - wa; self.remove(0, a); self.insert_after(0, b, x);
            } else {
                let x = { let wd = self.mk_word(self.word[a as usize], self.word[b as usize]); self.new_letter(wa, wd) };
                self.tpos[x as usize] = self.tpos[b as usize]; self.bpos[x as usize] = self.bpos[a as usize];
                self.replace(0, b, x); self.remove(0, a); self.replace(1, a, x); self.remove(1, b);
            }
        }
    }
    fn expand(&self, g: u32) -> (Vec<u8>, usize) {
        let mut out = Vec::with_capacity(self.glen[g as usize] as usize); let mut mark = usize::MAX; let mut st = vec![g];
        while let Some(x) = st.pop() { if self.gl[x as usize] == NIL { if self.gmark[x as usize] { mark = out.len(); } out.push(self.gbyte[x as usize] as u8); }
            else { st.push(self.gr[x as usize]); st.push(self.gl[x as usize]); } }
        (out, mark)
    }
}
/// The marked cycle by LF inversion, as bzip2 emits it: s[n-1-k] = L[LF^k(p)] over the cycle of p.
fn invert(l: &[u8], p: usize) -> Vec<u8> {
    let n = l.len(); let mut cnt = [0usize; 257]; for &c in l { cnt[c as usize + 1] += 1; } for c in 1..257 { cnt[c] += cnt[c - 1]; }
    let mut lf = vec![0usize; n]; for (x, &c) in l.iter().enumerate() { lf[x] = cnt[c as usize]; cnt[c as usize] += 1; }
    let mut v = vec![]; let mut x = p; loop { v.push(l[x]); x = lf[x]; if x == p { break; } } v.reverse(); v
}
fn kasai_rows(l: &[u8]) -> Option<(Vec<u32>, Vec<u32>, Vec<u32>)> {
    let n = l.len(); let mut cnt = [0usize; 257]; for &c in l { cnt[c as usize + 1] += 1; } for c in 1..257 { cnt[c] += cnt[c - 1]; }
    let mut lf = vec![0u32; n]; for (x, &c) in l.iter().enumerate() { lf[x] = cnt[c as usize] as u32; cnt[c as usize] += 1; }
    let (mut y, mut m) = (lf[0] as usize, 1usize); while y != 0 { y = lf[y] as usize; m += 1; } if m != n { return None; }
    let (mut t, mut sa) = (vec![0u8; n], vec![0u32; n]); let mut x = 0usize;
    for kk in 0..n { let pos = (n - kk) % n; sa[x] = pos as u32; t[(pos + n - 1) % n] = l[x]; x = lf[x] as usize; }
    let mut rank = vec![0u32; n]; for (row, &q) in sa.iter().enumerate() { rank[q as usize] = row as u32; }
    let mut lcp = vec![0u32; n]; let mut h = 0usize;
    for i in 0..n { let r = rank[i] as usize; if r == 0 { h = 0; continue; } let j = sa[r - 1] as usize;
        while h < n && t[(i + h) % n] == t[(j + h) % n] { h += 1; } lcp[r] = h as u32; if h > 0 { h -= 1; } }
    let mut fl = vec![0u32; n]; for x in 0..n { fl[lf[x] as usize] = x as u32; }
    // d(x): tau-steps to the chain's terminal (run boundary); initial: no tau-predecessor
    let ind = |x: usize| x >= 1 && l[x] != l[x - 1];
    let mut d = vec![u32::MAX; n]; let mut init = vec![0u32; n];
    for tt in 1..n { if !ind(tt) { continue; } d[tt] = 0; let mut y = tt; let mut e = 0u32;
        loop { let q = fl[y] as usize; if q == 0 || ind(q) { break; } e += 1; d[q] = e; y = q; } init[y] = 1; }
    Some((lcp, d, init))
}
fn analyse(l: &[u8], hits: &[(u64, u64)]) -> String {
    let Some((lcp, d, init)) = kasai_rows(l) else { return "null".into() };
    let (mut lam_ge, mut lam_eq, mut initc, mut dge, mut deq, mut dz) = (0, 0, 0, 0, 0, 0);
    let mut lam_minus = std::collections::BTreeMap::<i64, u64>::new();
    let mut dmin = std::collections::BTreeMap::<i64, u64>::new();
    for &(z, w) in hits { let z = z as usize; let lam = lcp[z] as u64;
        if lam >= w { lam_ge += 1; } if lam == w { lam_eq += 1; } if init[z] == 1 { initc += 1; }
        if d[z] != u32::MAX && d[z] as u64 + 1 >= w { dge += 1; } if d[z] as u64 + 1 == w { deq += 1; } if d[z] == 0 { dz += 1; }
        *lam_minus.entry((lam as i64 - w as i64).clamp(-3, 6)).or_default() += 1;
        *dmin.entry((d[z] as i64 - w as i64).clamp(-3, 6)).or_default() += 1; }
    format!("{{\"hits\":{},\"lam_ge_w\":{lam_ge},\"lam_eq_w\":{lam_eq},\"init\":{initc},\"d_plus1_ge_w\":{dge},\"d_plus1_eq_w\":{deq},\"d0\":{dz},\"lam_minus_w\":\"{:?}\",\"d_minus_w\":\"{:?}\"}}", hits.len(), lam_minus, dmin)
}
fn check(l: &[u8], p: usize) -> (Counts, Counts, bool) {
    let want = invert(l, p);
    let mut res = [Counts::default(); 2]; let mut ok = true;
    for (k, joint) in [false, true].into_iter().enumerate() {
        let mut s = Ind::new(l, p); let g = s.run(joint, &mut res[k]);
        if joint && std::env::var("GEOM").is_ok() { println!("{}", analyse(l, &s.hits)); }
        if !joint && std::env::var("DUPS").is_ok() {
            let mut seen = std::collections::HashSet::new(); let (mut short, mut dup) = (0u64, 0u64);
            for gidx in 0..s.gl.len() { if s.gl[gidx] == NIL || s.gmark[gidx] || s.glen[gidx] > 16 { continue; }
                short += 1; let (w, _) = s.expand(gidx as u32); if !seen.insert(w) { dup += 1; } }
            // canonical tree ids: leaves by byte, products by (canon l, canon r); spelling -> set of trees
            let mut canon = vec![0u32; s.gl.len()]; let mut intern = std::collections::HashMap::new();
            for x in 0..s.gl.len() { let key = if s.gl[x] == NIL { (u32::MAX, s.gbyte[x] as u32) } else { (canon[s.gl[x] as usize], canon[s.gr[x] as usize]) };
                let nid = intern.len() as u32; canon[x] = *intern.entry(key).or_insert(nid); }
            let mut by_spell: std::collections::HashMap<Vec<u8>, u32> = std::collections::HashMap::new(); let mut tree_conflict = 0u64;
            for gidx in 0..s.gl.len() { if s.gl[gidx] == NIL || s.gmark[gidx] || s.glen[gidx] > 16 { continue; }
                let (w, _) = s.expand(gidx as u32); let e = by_spell.entry(w).or_insert(canon[gidx]); if *e != canon[gidx] { tree_conflict += 1; } }
            println!("{{\"short_words\":{short},\"dup\":{dup},\"distinct\":{},\"tree_conflict\":{tree_conflict}}}", seen.len());
        }
        let (w, mark) = s.expand(g);
        // the word, rotated to start one past the mark, is the emitted block
        let m = w.len(); let rot: Vec<u8> = (0..m).map(|i| w[(mark + 1 + i) % m]).collect();
        if rot != want { ok = false; }
    }
    (res[0], res[1], ok)
}
fn main() {

    if a[1] == "--rle-fuzz" { let mut st = 0x1234_5678u64; let mut rnd = || { st ^= st << 13; st ^= st >> 7; st ^= st << 17; st };
        let (mut cases, mut bad, mut mixed) = (0, 0, 0);
        for _ in 0..200000 { let n = 1 + (rnd() % 300) as usize; let mut enc = Vec::new();
            while enc.len() < n { let b = (rnd() % 3) as u8; let r = 1 + (rnd() % 6) as usize; for _ in 0..r { enc.push(b); } }
            // make every 4-run followed by a count byte (valid RLE1): rewrite through an encoder of a random decoded text
            let mut dec = Vec::new(); for &b in &enc { let rep = if rnd() % 7 == 0 { (rnd() % 40) as usize } else { 1 }; for _ in 0..rep { dec.push(b); } }
            let mut e2 = Vec::new(); let mut i = 0; while i < dec.len() { let b = dec[i]; let mut j = i; while j < dec.len() && dec[j] == b && j - i < 255 + 4 { j += 1; }
                let r = j - i; if r >= 4 { for _ in 0..4 { e2.push(b); } e2.push((r - 4) as u8); } else { for _ in 0..r { e2.push(b); } } i = j; }
            let (fin, marks) = rle1(&e2); if fin != dec { bad += 1; continue; }
            let (mut lo, mut hi) = (0i64, 0i64); for &(ms, md, _, k) in &marks { let d = (md + k) as i64 - (ms + 1) as i64; lo = lo.min(d); hi = hi.max(d); } if lo < 0 && hi > 0 { mixed += 1; }
            cases += 1; if inplace_repair(&e2, &marks, fin.len(), 0).map_or(true, |r| r != fin) { bad += 1; } }
        println!("rle-fuzz: cases {cases} mixed-sign {mixed} bad {bad}"); return; }
    if a[1] == "--astrab-small" {
        let mut r = [[0u64; 3]; 2]; // [rev][ok, bad, ...]
        let mut tbad = [0u64; 2]; let mut cases = 0u64; let mut short = 0u64; let mut cut_more = 0u64; let mut cut_less = 0u64;
        for (k, maxn) in [(2u8, 9usize), (3, 6)] { for n in 1..=maxn { let tot = (k as usize).pow(n as u32);
            for code in 0..tot { let mut l = vec![0u8; n]; let mut c = code; for i in 0..n { l[i] = (c % k as usize) as u8; c /= k as usize; }
                for pr in 0..n { cases += 1; if invert(&l, pr).len() < n { short += 1; }
                    let sec = sec_equal_byte(&l);
                    for (ri, rev) in [false, true].into_iter().enumerate() { let (ok, _, _, _) = residual_run(&l, pr, &sec, rev); if ok { r[ri][0] += 1 } else { r[ri][1] += 1 } }
                    for (ri, rev) in [false, true].into_iter().enumerate() { let st = sec_tiles(&l, pr, 2); let (ok, cuts, _, _) = residual_run(&l, pr, &st, rev); if !ok { tbad[ri] += 1; }
                        if ri == 1 && ok { let base = { let mut s2 = Ind::new(&l, pr); let mut c2 = Counts::default(); s2.run(false, &mut c2); c2.steps }; if cuts > base { cut_more += 1 } else if cuts < base { cut_less += 1 } } } } } } }
        println!("astrab-small: cases {cases} (short {short}) equal-byte fwd ok/bad {:?} rev ok/bad {:?} | tiles(2) bad fwd {} rev {} | tile residual cuts > Cells in {cut_more}, < in {cut_less}", &r[0][..2], &r[1][..2], tbad[0], tbad[1]); return; }
    if a[1] == "--astra-small" {
        let mut st = [0u64; 8]; let (mut sec_n, mut sec_bad, mut clip, mut short_n) = (0u64, 0u64, 0u64, 0u64);
        for (k, maxn) in [(2u8, 9usize), (3, 6)] { for n in 1..=maxn { let tot = (k as usize).pow(n as u32);

/// #579: the owner-schedule writer and the reachable-offer fallback, simulated on the model's grammar.
#[derive(Default, Debug, Clone, Copy)]
struct W { full: bool, r: u64, m: u64, j: u64, k: u64, copy_bytes: u64, lit_edges: u64, lit_bytes: u64, wraps: u64, from_later: u64,
    cert_bad: u64, unassigned: u64, missing_src: u64, src_incomplete: u64, overwrite: u64, multi_cmd: u64, max_cmd_over_read: i64, ok: bool, fb_ok: bool, min_pos_differs: u64 }
fn writer_sim(l: &[u8], prim: usize, thr: u64, owner_unguarded: bool) -> W {
    let mut s = Ind::new(l, prim); let mut c = Counts::default(); let root = s.run(false, &mut c);
    let want = invert(l, prim); let p = want.len(); let n = l.len();
    let mut w = W { full: p == n, ..Default::default() };
    let is_rule = |x: u32| s.gl[x as usize] != NIL && (s.glen[x as usize] > thr || s.gmark[x as usize]);
    let rules: Vec<u32> = (0..s.gl.len() as u32).filter(|&x| is_rule(x)).collect();
    let mut ridx = vec![u32::MAX; s.gl.len()]; for (k, &x) in rules.iter().enumerate() { ridx[x as usize] = k as u32; }
    w.r = rules.len() as u64; w.m = rules.iter().filter(|&&x| s.gmark[x as usize]).count() as u64;
    // mark offset in root
    let (mut x, mut off) = (root, 0u64); while s.gl[x as usize] != NIL { let a = s.gl[x as usize]; if s.gmark[a as usize] { x = a; } else { off += s.glen[a as usize]; x = s.gr[x as usize]; } }
    let pp = p as u64; let q0 = (pp - (off + 1) % pp) % pp;
    let child = |r: u32, sd: usize| if sd == 0 { s.gl[r as usize] } else { s.gr[r as usize] };
    // true last parent of each rule child
    let mut last = vec![u32::MAX; s.gl.len()];
    for (k, &r) in rules.iter().enumerate() { for sd in 0..2 { let ch = child(r, sd); if is_rule(ch) { last[ch as usize] = k as u32; } } }
    let mut lit_buf: Vec<u8> = Vec::new();
    let mut put = |out: &mut Vec<i16>, d: u64, bytes: &[u8], ov: &mut u64| { for (i, &b) in bytes.iter().enumerate() { let t = ((d + i as u64) % pp) as usize; if out[t] >= 0 { *ov += 1; } out[t] = b as i16; } };
    // --- owner schedule ---
    if (w.full || owner_unguarded) && is_rule(root) {
        let mut out = vec![-1i16; p]; let mut pos = vec![u64::MAX; rules.len()]; pos[ridx[root as usize] as usize] = q0;
        let mut cmds: Vec<(u32, u64)> = Vec::new(); let mut read = 0i64;
        for k in (0..rules.len()).rev() {
            let r = rules[k]; read += 1; let q = pos[k];
            let mut ncmd = 0;
            if q == u64::MAX { w.unassigned += 1; continue; }
            let a = s.gl[r as usize]; let la = s.glen[a as usize];
            if q + la >= pp { w.wraps += 1; }
            for (sd, d) in [(0usize, q), (1usize, (q + la) % pp)] {
                let ch = child(r, sd);
                if is_rule(ch) {
                    if !s.gmark[ch as usize] { w.j += 1; }
                    let cert = !(k + 1 < rules.len() && child(rules[k + 1], sd) == ch);
                    if cert != (last[ch as usize] == k as u32) { w.cert_bad += 1; }
                    if cert { pos[ridx[ch as usize] as usize] = d; } else { cmds.push((ch, d)); ncmd += 1; }
                } else {
                    lit_buf.clear(); lit_buf.extend(s.expand(ch).0); w.lit_edges += 1; w.lit_bytes += lit_buf.len() as u64;
                    let mut ov = 0; put(&mut out, d, &lit_buf, &mut ov); w.overwrite += ov;
                }
            }
            if ncmd > 1 { w.multi_cmd += 1; }
            w.max_cmd_over_read = w.max_cmd_over_read.max(cmds.len() as i64 - read);
        }
        w.k = cmds.len() as u64;
        for &(ch, d) in cmds.iter().rev() {
            let src = pos[ridx[ch as usize] as usize]; let len = s.glen[ch as usize];
            if src == u64::MAX { w.missing_src += 1; continue; }
            if src > d { w.from_later += 1; }
            let mut buf = Vec::with_capacity(len as usize);
            for i in 0..len { let v = out[((src + i) % pp) as usize]; if v < 0 { w.src_incomplete += 1; break; } buf.push(v as u8); }
            if buf.len() as u64 != len { continue; }
            w.copy_bytes += len; let mut ov = 0; put(&mut out, d, &buf, &mut ov); w.overwrite += ov;
        }
        w.ok = out.iter().zip(want.iter()).all(|(&o, &b)| o == b as i16);
    } else if !is_rule(root) { w.ok = s.expand(root).0.len() == p; }
    // --- fallback: reachable pass, any reachable parent's offer, forward expansion with equality test ---
    {
        let mut out = vec![-1i16; p]; let mut pos = vec![u64::MAX; rules.len()]; let mut minp = vec![u64::MAX; rules.len()];
        if is_rule(root) { pos[ridx[root as usize] as usize] = q0; minp[ridx[root as usize] as usize] = q0; }
        for k in (0..rules.len()).rev() { let r = rules[k]; let q = pos[k]; if q == u64::MAX { continue; }
            let la = s.glen[s.gl[r as usize] as usize];
            for (sd, d) in [(0usize, q), (1usize, (q + la) % pp)] { let ch = child(r, sd); if is_rule(ch) { let i = ridx[ch as usize] as usize; pos[i] = d; minp[i] = minp[i].min(d); } } }
        for k in 0..rules.len() { if pos[k] != minp[k] { w.min_pos_differs += 1; } }
        let mut bad = false;
        for k in 0..rules.len() { let r = rules[k]; let q = pos[k]; if q == u64::MAX { continue; }
            let la = s.glen[s.gl[r as usize] as usize];
            for (sd, d) in [(0usize, q), (1usize, (q + la) % pp)] { let ch = child(r, sd);
                if is_rule(ch) { let src = pos[ridx[ch as usize] as usize]; if src != d { let len = s.glen[ch as usize];
                        for i in 0..len { let v = out[((src + i) % pp) as usize]; if v < 0 { bad = true; } out[((d + i) % pp) as usize] = v; } } }
                else { let e = s.expand(ch).0; for (i, &b) in e.iter().enumerate() { out[((d + i as u64) % pp) as usize] = b as i16; } } } }
        if !is_rule(root) { w.fb_ok = true; } else { w.fb_ok = !bad && out.iter().zip(want.iter()).all(|(&o, &b)| o == b as i16); }
    }
    w
}

/// #582: ownership trees on the reachable grammar of the marked root.
#[derive(Default, Debug, Clone)]
struct O { full: bool, ok_walk: bool, r: u64, e: u64, k: u64, k_bytes: u64, p_left: u64, groups: u64, short_writes: u64,
    walk_copies: u64, walk_bytes: u64, walk_max_stack: u64, walk_not_ready: u64,
    lu_rounds: u64, min_rounds: u64, lu_span: u64, min_span: u64, min_check_rounds: u64, min_check_span: u64, fo_rounds: u64,
    lu_later_logical: u64, lines: [u64; 3], brute: Option<(u64, u64)>, cap_over: i64, lines_alt: u64 }
fn lru_touch(lru: &mut std::collections::HashMap<u64, u64>, clock: &mut u64, cap: usize, line: u64, fills: &mut u64) {
    *clock += 1;
    if lru.insert(line, *clock).is_none() { *fills += 1;
        if lru.len() > cap { let (&old, _) = lru.iter().min_by_key(|(_, &t)| t).unwrap(); lru.remove(&old); } }
}
fn owner_study(l: &[u8], prim: usize, thr: u64, cache: bool) -> O {
    let mut s = Ind::new(l, prim); let mut c = Counts::default(); let root = s.run(false, &mut c);
    let want = invert(l, prim); let p = want.len() as u64; let mut o = O { full: p as usize == l.len(), ..Default::default() };
    let is_rule = |x: u32| s.gl[x as usize] != NIL && (s.glen[x as usize] > thr || s.gmark[x as usize]);
    if !is_rule(root) { o.ok_walk = true; return o; }
    let ch = |x: u32, sd: usize| if sd == 0 { s.gl[x as usize] } else { s.gr[x as usize] };
    // reachable rules, in birth (index) order
    let mut reach = vec![false; s.gl.len()]; reach[root as usize] = true;
    for x in (0..s.gl.len()).rev() { if reach[x] && is_rule(x as u32) { for sd in 0..2 { let y = ch(x as u32, sd); if is_rule(y) { reach[y as usize] = true; } } } }
    let rules: Vec<u32> = (0..s.gl.len() as u32).filter(|&x| reach[x as usize] && is_rule(x)).collect();
    o.r = rules.len() as u64;
    let mut indeg = vec![0u64; s.gl.len()]; let mut parents: Vec<Vec<u32>> = vec![Vec::new(); s.gl.len()];
    for &x in &rules { for sd in 0..2 { let y = ch(x, sd); if is_rule(y) { o.e += 1; indeg[y as usize] += 1; parents[y as usize].push(x); } else { o.short_writes += 1; } }
        if is_rule(ch(x, 0)) { o.p_left += 1; } }
    o.k = o.e + 1 - o.r;
    for &x in &rules { if x != root && indeg[x as usize] >= 2 { o.groups += 1; o.k_bytes += (indeg[x as usize] - 1) * s.glen[x as usize]; } }
    // dependency rounds/span for a tree given by selected parent: ready(v) bottom-up
    let ready = |sel: &dyn Fn(u32, u32) -> bool, wt: &dyn Fn(u32) -> u64| -> u64 {
        let mut rd = vec![0u64; s.gl.len()];
        for &x in &rules { let mut m = 0; for sd in 0..2 { let y = ch(x, sd); if is_rule(y) { let v = rd[y as usize] + if sel(x, y) { 0 } else { wt(y) }; m = m.max(v); } } rd[x as usize] = m; }
        rd[root as usize]
    };
    let last_par = |y: u32| *parents[y as usize].iter().max().unwrap();
    o.lu_rounds = ready(&|x, y| last_par(y) == x, &|_| 1); o.lu_span = ready(&|x, y| last_par(y) == x, &|y| s.glen[y as usize]);
    // the recurrence (top-down, descending birth): dist(v) = max(a, b + w), select a parent attaining a
    for (wsel, which) in [(0u8, 0), (1u8, 1)] {
        let w = |y: u32| if wsel == 0 { 1 } else { s.glen[y as usize] };
        let mut dist = vec![0u64; s.gl.len()]; let mut selp = vec![NIL; s.gl.len()];
        for &v in rules.iter().rev() { if v == root { continue; }
            let mut ds: Vec<(u64, u32)> = parents[v as usize].iter().map(|&u| (dist[u as usize], u)).collect(); ds.sort();
            let (a, ua) = *ds.last().unwrap(); let b = if ds.len() >= 2 { Some(ds[ds.len() - 2].0) } else { None };
            dist[v as usize] = match b { Some(b) => a.max(b + w(v)), None => a }; selp[v as usize] = ua; }
        // leaves: holes at copies; the critical path ends with a charge into some v: max over edges of dist(u)+w(v) if unselected
        let mut best = 0u64; for &v in &rules { if v == root { continue; } for &u in &parents[v as usize] { let d = dist[u as usize] + if selp[v as usize] == u { 0 } else { w(v) }; best = best.max(d); } }
        let chk = ready(&|x, y| selp[y as usize] == x, &w);
        if which == 0 { o.min_rounds = best; o.min_check_rounds = chk; } else { o.min_span = best; o.min_check_span = chk; }
    }
    // brute force over all ownership trees when few
    { let nonroot: Vec<u32> = rules.iter().cloned().filter(|&v| v != root).collect();
      let mut prod = 1u64; for &v in &nonroot { prod = prod.saturating_mul(parents[v as usize].len() as u64); }
      if prod <= 4096 { let mut br = (u64::MAX, u64::MAX); let mut choice = vec![0usize; nonroot.len()];
        loop { let mut selp = vec![NIL; s.gl.len()]; for (i, &v) in nonroot.iter().enumerate() { selp[v as usize] = parents[v as usize][choice[i]]; }
            let r1 = ready(&|x, y| selp[y as usize] == x, &|_| 1); let r2 = ready(&|x, y| selp[y as usize] == x, &|y| s.glen[y as usize]);
            br = (br.0.min(r1), br.1.min(r2));
            let mut i = 0; loop { if i == nonroot.len() { break; } choice[i] += 1; if choice[i] < parents[nonroot[i] as usize].len() { break; } choice[i] = 0; i += 1; }
            if i == nonroot.len() { break; } }
        o.brute = Some(br); } }
    // mark offset and root position
    let (mut x, mut off) = (root, 0u64); while s.gl[x as usize] != NIL { let a = s.gl[x as usize]; if s.gmark[a as usize] { x = a; } else { off += s.glen[a as usize]; x = s.gr[x as usize]; } }
    let q0 = (p - (off + 1) % p) % p;
    // --- first-occurrence walk (logical left-to-right from the root's left end; cursor wraps) ---
    let mut out = vec![-1i16; p as usize]; let mut tag = vec![u64::MAX; s.gl.len()]; let mut fo_par = vec![NIL; s.gl.len()];
    let mut cur = q0; let mut stack: Vec<(u32, u32)> = Vec::new(); // (pending right descriptor, its parent)
    let mut events: Vec<(u8, u64, u64, u64)> = Vec::new(); // (0 write | 1 copy, dest, len, src)
    let mut node = root; let mut from = NIL;
    loop {
        // visit `node` (a child of `from`)
        if is_rule(node) {
            if tag[node as usize] != u64::MAX { let src = tag[node as usize]; let len = s.glen[node as usize];
                for i in 0..len { if out[((src + i) % p) as usize] < 0 { o.walk_not_ready += 1; break; } }
                for i in 0..len { out[((cur + i) % p) as usize] = out[((src + i) % p) as usize]; }
                o.walk_copies += 1; o.walk_bytes += len; events.push((1, cur, len, src)); cur = (cur + len) % p;
            } else { tag[node as usize] = cur; fo_par[node as usize] = from;
                let (a, b) = (s.gl[node as usize], s.gr[node as usize]);
                if is_rule(a) { stack.push((b, node)); o.walk_max_stack = o.walk_max_stack.max(stack.len() as u64); from = node; node = a; continue; }
                let e = s.expand(a).0; for (i, &bb) in e.iter().enumerate() { out[((cur + i as u64) % p) as usize] = bb as i16; } events.push((0, cur, e.len() as u64, 0)); cur = (cur + e.len() as u64) % p;
                from = node; node = b; continue; }
        } else { let e = s.expand(node).0; for (i, &bb) in e.iter().enumerate() { out[((cur + i as u64) % p) as usize] = bb as i16; } events.push((0, cur, e.len() as u64, 0)); cur = (cur + e.len() as u64) % p; }
        match stack.pop() { Some((b, par)) => { from = par; node = b; } None => break }
    }
    o.ok_walk = o.walk_not_ready == 0 && out.iter().zip(want.iter()).all(|(&a, &b)| a == b as i16);
    let fo_sel = |x: u32, y: u32| fo_par[y as usize] == x;
    o.fo_rounds = ready(&fo_sel, &|_| 1);
    // --- two-phase schedules (study 5) for a given tree: (reverse scan writes, then forward commands) ---
    let cap_over = std::cell::Cell::new(i64::MIN);
    let two_phase_o = |sel: &dyn Fn(u32, u32) -> bool, rl: bool| -> (Vec<(u8, u64, u64, u64)>, u64) {
        let mut pos = vec![u64::MAX; s.gl.len()]; pos[root as usize] = q0; let mut ev = Vec::new(); let mut cmds = Vec::new(); let mut read = 0i64;
        for &x in rules.iter().rev() { let q = pos[x as usize]; let la = s.glen[s.gl[x as usize] as usize]; read += 1;
            let order = if rl { [(1usize, (q + la) % p), (0usize, q)] } else { [(0usize, q), (1usize, (q + la) % p)] };
            for (sd, d) in order { let y = ch(x, sd);
                if is_rule(y) { if sel(x, y) { pos[y as usize] = d; } else { cmds.push((y, d)); } }
                else { ev.push((0u8, d, s.glen[y as usize], 0)); } }
            cap_over.set(cap_over.get().max(cmds.len() as i64 - read)); }
        let mut later = 0u64;
        for &(y, d) in cmds.iter().rev() { let src = pos[y as usize]; if (src + p - q0) % p > (d + p - q0) % p { later += 1; } ev.push((1, d, s.glen[y as usize], src)); }
        (ev, later)
    };
    let two_phase = |sel: &dyn Fn(u32, u32) -> bool| two_phase_o(sel, false);
    let (ev_lu, later) = two_phase(&|x, y| last_par(y) == x); o.lu_later_logical = later;
    if cache {
        let (ev_fo2, _) = two_phase(&fo_sel);
        for (k, ev) in [ev_lu, ev_fo2, events].iter().enumerate() {
            let mut lru = std::collections::HashMap::new(); let mut clk = 0u64; let mut fills = 0u64;
            for &(kind, d, len, src) in ev.iter() { if len == 0 { continue; }
                if kind == 1 { let (a, b) = (src / 64, (src + len - 1) / 64); let wrap = src + len > p;
                    if !wrap { for ln in a..=b { lru_touch(&mut lru, &mut clk, 512, ln, &mut fills); } } else { for i in 0..len { let ln = ((src + i) % p) / 64; if i == 0 || ((src + i) % p) % 64 == 0 { lru_touch(&mut lru, &mut clk, 512, ln, &mut fills); } } } }
                let wrap = d + len > p;
                if !wrap { for ln in d / 64..=(d + len - 1) / 64 { lru_touch(&mut lru, &mut clk, 512, ln, &mut fills); } }
                else { for i in 0..len { let a = (d + i) % p; if i == 0 || a % 64 == 0 { lru_touch(&mut lru, &mut clk, 512, a / 64, &mut fills); } } } }
            o.lines[k] = fills;
        }
        let (ev_alt, _) = two_phase_o(&fo_sel, true);
        let mut lru = std::collections::HashMap::new(); let mut clk = 0u64; let mut fills = 0u64;
        for &(kind, d, len, src) in ev_alt.iter() { if len == 0 { continue; }
            if kind == 1 { for i in 0..len { let a = (src + i) % p; if i == 0 || a % 64 == 0 { lru_touch(&mut lru, &mut clk, 512, a / 64, &mut fills); } } }
            for i in 0..len { let a = (d + i) % p; if i == 0 || a % 64 == 0 { lru_touch(&mut lru, &mut clk, 512, a / 64, &mut fills); } } }
        o.lines_alt = fills;
    }
    { let _ = two_phase_o(&fo_sel, false); }
    o.cap_over = cap_over.get();
    o
}

/// #591: the endpoint sweep. Check the first-arrival identities inside the scalar model, then rebuild the
/// induction from arrivals alone (initial letters + mark, a max-heap, no widths or links) and compare.
#[derive(Default, Debug)]
struct Sw { cuts: u64, closures: u64, ident_bad: u64, sign_bad: u64, recon_ok: bool, first_diff: i64, ev_cuts: u64,
    born_top: u64, born_bot: u64, born_diag: u64, out_group_a: u64, out_group_b: u64, multi_side: u64, unit_both: u64, unit_min: u64, eq_untouched: u64 }
fn sweep_check(l: &[u8], prim: usize) -> Sw {
    let mut sw = Sw::default();
    // ---- scalar run with recorded events; identities checked at every cut ----
    let mut s = Ind::new(l, prim); let m = s.w.len();
    let mut ev_s: Vec<(u8, u32, u32, u64)> = Vec::new(); // (0 cut | 1 closure, A, B (or X,X), width)
    loop {
        let (a, b) = (s.ttail, s.btail);
        let (wa, wb) = (s.w[a as usize], s.w[b as usize]);
        if a == b { ev_s.push((1, a, a, wa)); s.remove(0, a); s.remove(1, a); sw.closures += 1; if s.marked(a) { break; } s.touched[a as usize] = true; continue; }
        let f = s.tpos[a as usize] + wa; // frontier
        if s.bpos[b as usize] + wb != f { sw.ident_bad += 1; }
        let ad = f as i64 - (s.bpos[a as usize] + wa) as i64; let bd = f as i64 - (s.tpos[b as usize] + wb) as i64; // a = -d_A, b = d_B
        if ad <= 0 || bd <= 0 { sw.sign_bad += 1; }
        if wa == 1 && wb == 1 { sw.unit_both += 1; } if wa.min(wb) == 1 { sw.unit_min += 1; }
        if wa == wb && s.orig[a as usize] && s.orig[b as usize] && !s.touched[a as usize] && !s.touched[b as usize] { sw.eq_untouched += 1; }
        s.touched[a as usize] = true; s.touched[b as usize] = true;
        let w = wa.min(wb); ev_s.push((0, a, b, w)); sw.cuts += 1;
        let xw = s.mk_word(s.word[a as usize], s.word[b as usize]); let x = s.new_letter(w, xw);
        if wa > wb { s.tpos[x as usize] = s.tpos[b as usize]; s.bpos[x as usize] = s.bpos[a as usize] + (wa - wb);
            s.replace(0, b, x); s.w[a as usize] = wa - wb; s.remove(1, b); s.insert_after(1, a, x); }
        else if wa < wb { s.bpos[x as usize] = s.bpos[a as usize]; s.tpos[x as usize] = s.tpos[b as usize] + (wb - wa);
            s.replace(1, a, x); s.w[b as usize] = wb - wa; s.remove(0, a); s.insert_after(0, b, x); }
        else { s.tpos[x as usize] = s.tpos[b as usize]; s.bpos[x as usize] = s.bpos[a as usize]; s.replace(0, b, x); s.remove(0, a); s.replace(1, a, x); s.remove(1, b); }
        // child's right ends: t = F - b, b = F - a
        let (tx, bx) = (s.tpos[x as usize] + w, s.bpos[x as usize] + w);
        if tx as i64 != f as i64 - bd || bx as i64 != f as i64 - ad { sw.ident_bad += 1; }
    }
    // ---- reconstruction from arrivals only ----
    let n = l.len();
    // letters exactly as Ind::new makes them (runs, marked row split): (start, len, byte, marked)
    let mut lets: Vec<(usize, usize, u8, bool)> = Vec::new(); let mut x = 0;
    while x < n { let mut y = x + 1; while y < n && l[y] == l[x] { y += 1; }
        if prim >= x && prim < y { if prim > x { lets.push((x, prim - x, l[x], false)); } lets.push((prim, 1, l[x], true)); if y > prim + 1 { lets.push((prim + 1, y - prim - 1, l[x], false)); } }
        else { lets.push((x, y - x, l[x], false)); } x = y; }
    let mut below = [0u64; 257]; for &c in l { below[c as usize + 1] += 1; } for c in 1..257 { below[c] += below[c - 1]; }
    let mut cur = [0u64; 256];
    let mut tend = Vec::new(); let mut bend = Vec::new(); let mut marked = Vec::new();
    for &(st, ln, c, mk) in &lets { tend.push((st + ln) as u64); cur[c as usize] += ln as u64; bend.push(below[c as usize] + cur[c as usize]); marked.push(mk); }
    let mut heap = std::collections::BinaryHeap::new(); // (h, side: 0 top / 1 bottom / 2 diagonal, band)
    let side = |t: u64, b: u64| if b > t { 1u8 } else if b < t { 0 } else { 2 };
    for i in 0..m { heap.push((tend[i].max(bend[i]), side(tend[i], bend[i]), i as u32)); }
    let mut ev_r: Vec<(u8, u32, u32, u64)> = Vec::new();
    let (mut ta, mut tb): (Option<(u32, u64)>, Option<(u32, u64)>) = (None, None); // (band, a) and (band, b)
    let mut pending: Option<usize> = None; // index in ev_r of the last cut or closure awaiting its width
    let mut next_id = m as u32; let mut f_prev = n as u64 + 0; let _ = f_prev;
    let mut done = false;
    while let Some(&(f, _, _)) = heap.peek() {
        let mut at: Vec<(u8, u32)> = Vec::new();
        while let Some(&(g, sd, band)) = heap.peek() { if g != f { break; } heap.pop(); at.push((sd, band)); }
        if at.len() > 2 || (at.len() == 2 && (at[0].0 == at[1].0 || at.iter().any(|e| e.0 == 2))) { sw.multi_side += 1; }
        if let Some(i) = pending.take() { let fp = ev_r[i].3; ev_r[i].3 = fp - f; } // width = previous frontier - this one
        for &(sd, band) in &at { match sd {
            0 => { let a = tend[band as usize] - bend[band as usize]; ta = Some((band, a)); }
            1 => { let b = bend[band as usize] - tend[band as usize]; tb = Some((band, b)); }
            _ => { ev_r.push((1, band, band, f)); pending = Some(ev_r.len() - 1);
                   if marked[band as usize] { ev_r.last_mut().unwrap().3 = 1; done = true; } ta = None; tb = None; } } }
        if done { break; }
        if at.iter().any(|e| e.0 == 2) { continue; }
        let ((ab, a), (bb, b)) = (ta.unwrap(), tb.unwrap());
        let id = next_id; next_id += 1; sw.ev_cuts += 1;
        let (tc, bc) = (f - b, f - a); tend.push(tc); bend.push(bc); marked.push(marked[ab as usize] || marked[bb as usize]);
        let h = tc.max(bc); let sdc = side(tc, bc); heap.push((h, sdc, id));
        match sdc { 0 => sw.born_top += 1, 1 => sw.born_bot += 1, _ => sw.born_diag += 1 }
        if h / 64 != (f - 1) / 64 { sw.out_group_a += 1; } if (h - 1) / 64 != (f - 1) / 64 { sw.out_group_b += 1; }
        ev_r.push((0, ab, bb, f)); pending = Some(ev_r.len() - 1);
        // the survivor stays; the replaced side(s) come from the next frontier's events
        ta = Some((ab, a)); tb = Some((bb, b));
        // a side keeps its operand until an arrival replaces it (the theorem: the exhausted side gets one)
    }
    sw.recon_ok = ev_r == ev_s;
    sw.first_diff = ev_r.iter().zip(ev_s.iter()).position(|(x, y)| x != y).map_or(if ev_r.len() == ev_s.len() { -1 } else { ev_r.len().min(ev_s.len()) as i64 }, |p| p as i64);
    sw
}

/// Astra C (#601): pieces of S = LF^{-1} from the coalesced runs alone, the word/exit theorem,
/// the first-return clip, and the section cursor.
struct Pieces { lo: Vec<u32>, hi: Vec<u32>, d: Vec<i64>, c: Vec<u8>, of_row: Vec<u32>, a_lo: [u32; 257] }
fn pieces(l: &[u8]) -> Pieces {
    let n = l.len(); let mut cnt = [0u32; 257]; for &c in l { cnt[c as usize + 1] += 1; } for c in 1..257 { cnt[c] += cnt[c - 1]; }
    let mut seen = [0u32; 256]; let (mut lo, mut hi, mut d, mut cc) = (vec![], vec![], vec![], vec![]);
    let mut x = 0; while x < n { let c = l[x]; let mut y = x + 1; while y < n && l[y] == c { y += 1; }
        let f0 = cnt[c as usize] + seen[c as usize]; lo.push(f0); hi.push(f0 + (y - x) as u32); d.push(x as i64 - f0 as i64); cc.push(c); seen[c as usize] += (y - x) as u32; x = y; }
    let mut of_row = vec![0u32; n]; for j in 0..lo.len() { for r in lo[j]..hi[j] { of_row[r as usize] = j as u32; } }
    Pieces { lo, hi, d, c: cc, of_row, a_lo: cnt }
}
fn dense_walk(pc: &Pieces, n: usize, p: usize) -> Vec<u8> { let mut out = Vec::with_capacity(n); let mut x = p as i64;
    for _ in 0..n { let j = pc.of_row[x as usize] as usize; out.push(pc.c[j]); x += pc.d[j]; } out }
#[derive(Default, Debug, Clone)]
struct Sec { ok: bool, f: u64, prefix: u64, probes: u64, hits: u64, partial_hits: u64, direct: u64, copied: u64, certs: u64, posted: u64, singles: u64, first_copy: i64, last_publish: i64, filled: u64, full: bool, clip_strict: u64 }
fn section_cursor(l: &[u8], p: usize, anchor: u8) -> Sec {
    let n = l.len(); let pc = pieces(l); let want = dense_walk(&pc, n, p);
    let (alo, ahi) = (pc.a_lo[anchor as usize] as i64, pc.a_lo[anchor as usize + 1] as i64);
    let mut sc = Sec { f: (ahi - alo) as u64, first_copy: -1, last_publish: -1, ..Default::default() };
    let mut entry: Vec<Option<(u64, u64, i64)>> = vec![None; (ahi - alo) as usize];
    let mut out: Vec<u8> = Vec::with_capacity(n); let mut x = p as i64; let in_a = |x: i64| x >= alo && x < ahi;
    // prefix before the first anchor visit
    while out.len() < n && !in_a(x) { let j = pc.of_row[x as usize] as usize; out.push(pc.c[j]); x += pc.d[j]; sc.prefix += 1; sc.direct += 1; }
    while out.len() < n {
        // x in A
        sc.probes += 1;
        if let Some((src, len, disp)) = entry[(x - alo) as usize] {
            let take = (len as usize).min(n - out.len()); if take < len as usize { sc.partial_hits += 1; }
            if sc.first_copy < 0 { sc.first_copy = out.len() as i64; }
            for i in 0..take { let b = out[src as usize + i]; out.push(b); }
            sc.hits += 1; sc.copied += take as u64; x += disp; continue;
        }
        // miss: walk to the first return, building the certificate
        let s = x; let start = out.len() as u64; let (mut clo, mut chi) = (i64::MIN, i64::MAX); let mut steps = 0u64;
        loop { let j = pc.of_row[x as usize] as usize; clo = clo.max(pc.lo[j] as i64 + s - x); chi = chi.min(pc.hi[j] as i64 + s - x);
            out.push(pc.c[j]); x += pc.d[j]; steps += 1; sc.direct += 1;
            if out.len() >= n || in_a(x) { break; } }
        if out.len() >= n && !in_a(x) { break; } // trailing partial miss: nothing to publish
        if out.len() >= n && in_a(x) { /* finished exactly at a return: still a complete certificate */ }
        let dsp = x - s; let (ulo, uhi) = (clo, chi); clo = clo.max(alo - dsp); chi = chi.min(ahi - dsp);
        if (ulo, uhi) != (clo, chi) { sc.clip_strict += 1; }
        sc.certs += 1; if chi - clo == 1 { sc.singles += 1; }
        for r in clo..chi { let e = &mut entry[(r - alo) as usize]; if e.is_none() { *e = Some((start, steps, dsp)); sc.posted += 1; } }
        sc.last_publish = start as i64; let _ = steps;
    }
    sc.filled = entry.iter().filter(|e| e.is_some()).count() as u64;
    // full period: the walk from p returns to p only after n steps
    let mut y = p as i64; let mut k = 0; loop { let j = pc.of_row[y as usize] as usize; y += pc.d[j]; k += 1; if y == p as i64 || k > n { break; } } sc.full = k == n;
    sc.ok = out == want; sc
}
/// Exhaustive small checks: walk = LF inversion; word/exit rigidity; first-return certificates equal maximal classes.
fn small_theorems(l: &[u8], p: usize, st: &mut [u64; 8]) {
    let n = l.len(); let pc = pieces(l); let s = |x: usize| (x as i64 + pc.d[pc.of_row[x] as usize]) as usize;
    // walk vs LF inversion (invert gives the marked cycle in bzip2 order; the walk is n bytes of it, repeated)
    let cyc = invert(l, p); let w = dense_walk(&pc, n, p); let ok = (0..n).all(|i| w[i] == cyc[i % cyc.len()]); st[0] += 1; if !ok { st[1] += 1; }
    // rigidity for k = 1..=n+2
    for k in 1..=n + 2 { let mut rows: Vec<(Vec<u8>, usize, usize, Vec<u32>)> = (0..n).map(|x| { let (mut y, mut word, mut ps) = (x, vec![], vec![]);
            for _ in 0..k { ps.push(pc.of_row[y]); word.push(pc.c[pc.of_row[y] as usize]); y = s(y); } (word, x, y, ps) }).collect();
        rows.sort();
        let mut i = 0; while i < n { let mut j = i; while j < n && rows[j].0 == rows[i].0 { j += 1; }
            // interval: rows i..j consecutive x
            for t in i + 1..j { if rows[t].1 != rows[t - 1].1 + 1 { st[2] += 1; } if rows[t].2 <= rows[t - 1].2 || (rows[t].2 as i64 - rows[t].1 as i64) < (rows[t - 1].2 as i64 - rows[t - 1].1 as i64) { st[3] += 1; } }
            for a in i..j { for b in a + 1..j { st[4] += 1; let same_exit = rows[b].2 as i64 - rows[a].2 as i64 == rows[b].1 as i64 - rows[a].1 as i64; if same_exit != (rows[a].3 == rows[b].3) { st[5] += 1; } } }
            i = j; } }
    // first-return certificates for every anchor byte present: clipped certificate == maximal (word, exit) class interval
    let mut present = [false; 256]; for &c in l { present[c as usize] = true; }
    for anc in 0..256usize { if !present[anc] { continue; } let (alo, ahi) = (pc.a_lo[anc] as i64, pc.a_lo[anc + 1] as i64);
        let ret = |a: i64| { let (mut x, mut word, mut ps) = (a, vec![], vec![]); let mut k = 0; loop { let j = pc.of_row[x as usize] as usize; ps.push(j); word.push(pc.c[j]); x += pc.d[j]; k += 1; if (x >= alo && x < ahi) || k > n + 1 { break; } } (word, x - a, ps) };
        let cls: Vec<_> = (alo..ahi).map(ret).collect();
        for a in alo..ahi { let (ref w0, d0, ref ps0) = cls[(a - alo) as usize];
            // certificate by intersection along the path, then the final clip
            let (mut clo, mut chi) = (i64::MIN, i64::MAX); let mut x = a; for &j in ps0 { clo = clo.max(pc.lo[j] as i64 + a - x); chi = chi.min(pc.hi[j] as i64 + a - x); x += pc.d[j]; }
            let unclipped = (clo, chi); clo = clo.max(alo - d0); chi = chi.min(ahi - d0);
            // maximal interval around a of rows with the same (word, exit)
            let same = |r: i64| r >= alo && r < ahi && { let c2 = &cls[(r - alo) as usize]; c2.0 == *w0 && c2.1 == d0 };
            let (mut l0, mut h0) = (a, a + 1); while same(l0 - 1) { l0 -= 1; } while same(h0) { h0 += 1; }
            st[6] += 1; if (clo, chi) != (l0, h0) { st[7] += 1; } if unclipped != (clo, chi) { st[6] += 0; }
        } }
}

/// Astra B (#602): residual induction on the first return of T = LF to a section, with LF-departure words.
impl Ind {
    /// Bands in top (domain-rank) order: (image start, width, word bytes, mark offset within the word).
    fn from_bands(bands: &[(u64, u64, Vec<u8>, Option<usize>)], rev: bool) -> Ind {
        let m = bands.len();
        let mut s = Ind { w: vec![], tpos: vec![], bpos: vec![], hits: vec![], word: vec![], orig: vec![], touched: vec![], byte: vec![], tp: vec![], tn: vec![], bp: vec![], bn: vec![], ttail: NIL, btail: NIL,
            gl: vec![], gr: vec![], glen: vec![], gmark: vec![], gbyte: vec![] };
        let mut acc = 0u64;
        for (i, (img, w, word, mk)) in bands.iter().enumerate() {
            let bytes: Vec<u8> = if rev { word.iter().rev().cloned().collect() } else { word.clone() };
            let mko = mk.map(|o| if rev { word.len() - 1 - o } else { o });
            let mut g = NIL;
            for (k, &b) in bytes.iter().enumerate() { let leaf = s.gl.len() as u32; s.gl.push(NIL); s.gr.push(NIL); s.glen.push(1); s.gmark.push(mko == Some(k)); s.gbyte.push(b as i16);
                g = if g == NIL { leaf } else { s.mk_word(g, leaf) }; }
            s.w.push(*w); s.tpos.push(acc); s.bpos.push(*img); acc += w; s.word.push(g); s.orig.push(word.len() == 1); s.touched.push(false); s.byte.push(if word.len() == 1 { word[0] as i16 } else { -1 });
            s.tp.push(if i == 0 { NIL } else { i as u32 - 1 }); s.tn.push(if i + 1 == m { NIL } else { i as u32 + 1 }); s.bp.push(NIL); s.bn.push(NIL);
        }
        s.ttail = m as u32 - 1;
        let mut ord: Vec<usize> = (0..m).collect(); ord.sort_by_key(|&i| bands[i].0);
        for (j, &i) in ord.iter().enumerate() { s.bp[i] = if j == 0 { NIL } else { ord[j - 1] as u32 }; s.bn[i] = if j + 1 == m { NIL } else { ord[j + 1] as u32 }; }
        s.btail = ord[m - 1] as u32; s
    }
}
/// The residual of T restricted to `sec` (a row mask): bands, or None when the mark lies on a section-free cycle.
fn residual_bands(l: &[u8], p: usize, sec: &[bool]) -> Result<Vec<(u64, u64, Vec<u8>, Option<usize>)>, Vec<u8>> {
    let n = l.len(); let mut cnt = [0usize; 257]; for &c in l { cnt[c as usize + 1] += 1; } for c in 1..257 { cnt[c] += cnt[c - 1]; }
    let mut lf = vec![0usize; n]; let mut c2 = cnt; for (x, &c) in l.iter().enumerate() { lf[x] = c2[c as usize]; c2[c as usize] += 1; }
    let mut rank = vec![u64::MAX; n]; let mut k = 0u64; for x in 0..n { if sec[x] { rank[x] = k; k += 1; } }
    // first return from each section row; where the mark falls
    let mut arcs: Vec<(u64, Vec<u8>, Option<usize>)> = Vec::new(); let mut marked_found = false;
    for x in 0..n { if !sec[x] { continue; }
        let mut word = vec![l[x]]; let mut mk = if x == p { marked_found = true; Some(0) } else { None }; let mut y = lf[x];
        while !sec[y] { if y == p { mk = Some(word.len()); marked_found = true; } word.push(l[y]); y = lf[y]; }
        arcs.push((rank[y], word, mk)); }
    if !marked_found { // the mark is on a cycle avoiding the section: spell it directly
        let mut v = vec![]; let mut x = p; loop { v.push(l[x]); x = lf[x]; if x == p { break; } } v.reverse(); return Err(v); }
    // coalesce consecutive ranks with consecutive targets and equal words; the marked arc stays alone
    let mut bands: Vec<(u64, u64, Vec<u8>, Option<usize>)> = Vec::new();
    for (i, (t, w, mk)) in arcs.into_iter().enumerate() { let _ = i;
        if let Some(last) = bands.last_mut() { if mk.is_none() && last.3.is_none() && last.0 + last.1 == t && last.2 == w { last.1 += 1; continue; } }
        bands.push((t, 1, w, mk)); }
    Ok(bands)
}
fn residual_run(l: &[u8], p: usize, sec: &[bool], rev: bool) -> (bool, u64, u64, u64) {
    let want = invert(l, p);
    match residual_bands(l, p, sec) {
        Err(v) => (v == want, 0, 0, 0),
        Ok(bands) => { let nb = bands.len() as u64; let mut s = Ind::from_bands(&bands, rev); let mut c = Counts::default(); let g = s.run(false, &mut c);
            let (w, mark) = s.expand(g); let m = w.len(); let rot: Vec<u8> = (0..m).map(|i| w[(mark + 1 + i) % m]).collect();
            let rules = (0..s.gl.len()).filter(|&x| s.gl[x] != NIL && (s.glen[x] > 16 || s.gmark[x])).count() as u64;
            (rot == want, c.steps, nb, rules) }
    }
}
fn sec_equal_byte(l: &[u8]) -> Vec<bool> { let mut f = l.to_vec(); f.sort(); (0..l.len()).map(|x| l[x] != f[x]).collect() }
fn sec_tiles(l: &[u8], p: usize, b: usize) -> Vec<bool> {
    let n = l.len(); let mut cnt = [0usize; 257]; for &c in l { cnt[c as usize + 1] += 1; } for c in 1..257 { cnt[c] += cnt[c - 1]; }
    let mut lf = vec![0usize; n]; let mut c2 = cnt; for (x, &c) in l.iter().enumerate() { lf[x] = c2[c as usize]; c2[c as usize] += 1; }
    let mut sec = vec![true; n];
    for j in 0..n / b { let s0 = j * b; if p / b == j { continue; }
        let same = (s0..s0 + b).all(|x| l[x] == l[s0]) && lf[s0] % b == 0 && (s0..s0 + b).all(|x| lf[x] == lf[s0] + (x - s0));
        // a single constant-byte *run*: same byte suffices for consecutive rows (a run in L)
        if same { for x in s0..s0 + b { sec[x] = false; } } }
    sec
}

/// Astra D (#600): forward affine tiles, RLE1 edits, in-place mixed-direction repair, CRC.
fn bz_crc(data: &[u8]) -> u32 { let mut c: u32 = 0xFFFF_FFFF; for &b in data { c ^= (b as u32) << 24; for _ in 0..8 { c = if c & 0x8000_0000 != 0 { (c << 1) ^ 0x04C1_1DB7 } else { c << 1 }; } } !c }
fn rle1(enc: &[u8]) -> (Vec<u8>, Vec<(usize, usize, u8, usize)>) { // (count-byte source, dest, byte, count)
    let mut out = Vec::new(); let mut marks = Vec::new(); let (mut last, mut run) = (-1i32, 0usize); let mut i = 0;
    while i < enc.len() { let b = enc[i];
        if run == 4 { let k = b as usize; marks.push((i, out.len(), last as u8, k)); for _ in 0..k { out.push(last as u8); } run = 0; last = -1; i += 1; continue; }
        if b as i32 == last { run += 1 } else { last = b as i32; run = 1 } out.push(b); i += 1; }
    (out, marks)
}
/// The study's three-step in-place repair, simulated in one slot.
fn inplace_repair(enc: &[u8], marks: &[(usize, usize, u8, usize)], fin_len: usize, slack: usize) -> Option<Vec<u8>> {
    let cap = enc.len().max(fin_len) + slack; let mut buf = vec![0u8; cap]; buf[..enc.len()].copy_from_slice(enc);
    // literal intervals: (src_lo, src_hi, dst_lo)
    let mut ivs = Vec::new(); let (mut s, mut d) = (0usize, 0usize);
    for &(ms, md, _, k) in marks { ivs.push((s, ms, d)); s = ms + 1; d = md + k; } ivs.push((s, enc.len(), d));
    for &(lo, hi, dl) in &ivs { if dl < lo && hi > lo { buf.copy_within(lo..hi, dl); } }           // left shifts, forward order
    for &(lo, hi, dl) in ivs.iter().rev() { if dl > lo && hi > lo { if dl + (hi - lo) > cap { return None; } buf.copy_within(lo..hi, dl); } } // right shifts, reverse
    for &(_, md, b, k) in marks { for j in 0..k { buf[md + j] = b; } }
    buf.truncate(fin_len); Some(buf)
}
fn tile_probes(pc: &Pieces, n: usize, p: usize, t: usize) -> (u64, u64, u64) {
    // split each piece's F-interval at t-row boundaries; per tile, pieces sorted by start
    let mut tiles: Vec<Vec<(u32, u32)>> = vec![Vec::new(); (n + t - 1) / t]; // (local end inclusive, piece) per tile, in row order
    let mut order: Vec<usize> = (0..pc.lo.len()).collect(); order.sort_by_key(|&j| pc.lo[j]);
    let mut npieces = 0u64;
    for &j in &order { let (mut a, b) = (pc.lo[j] as usize, pc.hi[j] as usize);
        while a < b { let tt = a / t; let e = b.min((tt + 1) * t); tiles[tt].push(((e - 1 - tt * t) as u32, j as u32)); npieces += 1; a = e; } }
    let mut probes = 0u64; let mut x = p;
    for _ in 0..n { let tt = x / t; let loc = (x - tt * t) as u32; let v = &tiles[tt];
        let mut k = 0; while v[k].0 < loc { k += 1; } probes += k as u64 + 1;
        let j = v[k].1 as usize; x = (x as i64 + pc.d[j]) as usize; }
    (npieces, tiles.len() as u64, probes)
}

/// Astra A (#599): permanent original domains learning words from two adjacent completed spans.
#[derive(Default, Debug, Clone)]
struct Wd { ok: bool, visits: u64, inst: u64, both_multi: u64, retire: u64, copies: u64, copy_bytes: u64, copies16: u64, wide_visits: u64, singles: u64, wide: u64, audit_rows: u64, audit_bytes: u64, audit_bad: u64, straddle_bad: u64 }
fn whole_domain(l: &[u8], p: usize, audit: bool) -> (Wd, Vec<u8>) {
    let n = l.len(); let pc = pieces(l); let m = pc.lo.len();
    let mut wd = Wd::default();
    // per domain: word length, displacement, source (None = literal), finished
    let mut len = vec![1u64; m]; let mut disp: Vec<i64> = pc.d.clone(); let mut src: Vec<Option<u64>> = vec![None; m]; let mut fin = vec![false; m];
    for j in 0..m { if pc.hi[j] - pc.lo[j] == 1 { wd.singles += 1 } else { wd.wide += 1 } }
    let mut out: Vec<u8> = Vec::with_capacity(n); let mut x = p as i64;
    let mut prev: Option<(usize, u64, i64, u64)> = None; // (domain, emitted length, displacement, out start)
    while out.len() < n {
        let j = pc.of_row[x as usize] as usize; let start = out.len() as u64;
        let (k, f) = (len[j], disp[j]); let take = (k as usize).min(n - out.len());
        match src[j] { None => out.push(pc.c[j]), Some(s0) => { for i in 0..take { let b = out[s0 as usize + i]; out.push(b); } wd.copies += 1; wd.copy_bytes += take as u64; if take <= 16 { wd.copies16 += 1; } } }
        wd.visits += 1; if pc.hi[j] - pc.lo[j] > 1 { wd.wide_visits += 1; }
        if out.len() >= n { break; } // a final phrase closes the output without installing
        let mut cur = Some((j, k, f, start));
        if let Some((u, mu, eu, su)) = prev {
            let (a, b) = (pc.lo[u] as i64, pc.hi[u] as i64);
            if b - a >= 2 && !fin[u] {
                let (c, d) = (pc.lo[j] as i64, pc.hi[j] as i64);
                if a + eu >= c && b + eu <= d {
                    // install UV on all of U
                    let (nl, nd) = (mu + k, eu + f);
                    if audit { // every row of U spells out[su..su+nl] and exits at x+nd
                        for r in a..b { let mut y = r; for t in 0..nl { let jj = pc.of_row[y as usize] as usize; if pc.c[jj] != out[(su + t) as usize] { wd.audit_bad += 1; break; } y += pc.d[jj]; } if y != r + nd { wd.audit_bad += 1; } wd.audit_rows += 1; wd.audit_bytes += nl; } }
                    if mu > 1 && k > 1 { wd.both_multi += 1; }
                    len[u] = nl; disp[u] = nd; src[u] = Some(su); wd.inst += 1;
                    if u == j { cur = None; } // the current visit's descriptor changed: forget it as an operand
                } else {
                    // retire: the image straddles at least two original domains
                    if audit { let lo = pc.of_row[(a + eu) as usize]; let hi = pc.of_row[(b + eu - 1) as usize]; if lo == hi { wd.straddle_bad += 1; } }
                    fin[u] = true; wd.retire += 1; }
            }
        }
        prev = cur; x += f;
    }
    let want = dense_walk(&pc, n, p); wd.ok = out == want; (wd, out)
}
