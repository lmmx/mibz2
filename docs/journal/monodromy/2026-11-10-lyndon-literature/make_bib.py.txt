def load():
    recs, unknown = {}, []
    for c in CLUSTERS:
        try:
            r = json.load(open(f"{S}/reports/lit2_{c}.json"))
        except FileNotFoundError:
            continue
        for b in r["bibliography"]:
            k = (c, b["key"])
            if k in SKIP:
                continue
            key = KEYS.get(k) or EXTRA.get(b["key"])
            if key is None:
                unknown.append(k)
                continue
            recs[key] = b
    return recs, unknown


def clean(s):
    return " ".join((s or "").split())


def item(key, b, lines):
    out = [f"- **[{key}]** {clean(b['authors']).replace(';', ',')}. *{clean(b['title'])}.*"]
    loc = clean(b.get("locator", ""))
    ven = clean(b["venue"])
    yr = clean(b["year"])
    out.append(f"  {ven}, {yr}." if ven else f"  {yr}.")
    if loc:
        out.append(f"  Versions and what was read: {loc}")
    ids = []
    if clean(b.get("doi", "")):
        ids.append(f"DOI `{clean(b['doi'])}`")
    if clean(b.get("arxiv", "")):
        ids.append(f"arXiv `{clean(b['arxiv'])}`")
    if ids:
        out.append("  " + "; ".join(ids) + ".")
    if lines:
        out.append("  Scholar lines: " + ", ".join(lines) + ".")
    out.append(f"  *What it is:* {clean(b['one_line'])}")
    out.append(f"  *Record from:* {clean(b['source_of_record'])}")
    if clean(b.get("unverified", "")):
        out.append(f"  *Not verified:* {clean(b['unverified'])}")
    return "\n".join(out)
