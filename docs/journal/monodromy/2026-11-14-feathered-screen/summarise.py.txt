#!/usr/bin/env python3
"""Summary over blocks of results/<POLICY>-blocks.jsonl: min / quartiles / max."""
import json, sys, statistics, os
D = os.path.dirname(os.path.abspath(__file__))

def q(v):
    v = sorted(v)
    qs = statistics.quantiles(v, n=4, method="inclusive")
    return (v[0], qs[0], qs[1], qs[2], v[-1])

def fmt(t, d):
    return " / ".join(f"{x:.{d}f}" for x in t)

out = []
for pol in sys.argv[1:]:
    path = f"{D}/results/{pol}-blocks.jsonl"
    if not os.path.exists(path):
        continue
    B = [json.loads(l) for l in open(path)]
    n = len(B)
    chk = dict(
        blocks=n,
        finished=sum(b["finished"] for b in B),
        hash_ok=sum(b["hash_ok"] for b in B),
        rows_equal=sum(b["rows_removed"] == b["n"] - b["points_left"] for b in B),
        one_point_left=sum(b["points_left"] == 1 and b["letters_left"] == 1 for b in B),
        nodes_equal_sum_m=sum(b["nodes"] == b["sum_m"] for b in B),
        S_equals_census_cells=sum(b["S"] == b["census_cells"] for b in B),
        r_equals_census_letters=sum(b["r"] == b["census_letters"] for b in B),
        S_equal_steps_is_r_minus_1=sum(b["S_equal"] == b["r"] - 1 for b in B),
        max_growth_ratio=max(b["max_ratio"] for b in B),
        rounds_with_r_growing_before_coalescing=sum(b["rounds_growing_before"] for b in B),
        total_rounds=sum(b["rounds"] for b in B),
        total_joins=sum(b["joins"] for b in B),
        empty_m0=sum(b["empty_m0"] for b in B),
        empty_no_row=sum(b["empty_no_row"] for b in B),
    )
    rows = [
        ("rounds", [b["rounds"] for b in B], 0),
        ("empty rounds (m = 0)", [b["empty_m0"] for b in B], 0),
        ("empty rounds (no row removed)", [b["empty_no_row"] for b in B], 0),
        ("total work / S  (cells scanned + fragments written before coalescing)", [(b["cells_scanned"] + b["frag_written_before"]) / b["S"] for b in B], 2),
        ("total work / S  (fragments counted after coalescing)", [(b["cells_scanned"] + b["frag_written_after"]) / b["S"] for b in B], 2),
        ("total work / n  (per output byte of the block)", [(b["cells_scanned"] + b["frag_written_before"]) / b["n"] for b in B], 3),
        ("selection: cells scanned / S", [b["cells_scanned"] / b["S"] for b in B], 2),
        ("rebuild: fragments written / S", [b["frag_written_before"] / b["S"] for b in B], 2),
        ("first round alone: (cells + fragments) / S", None, 2),
        ("max r reached / r", [max(b["max_r_before"], b["r"]) / b["r"] for b in B], 3),
        ("largest r_new before coalescing / r", [b["max_r_before"] / b["r"] for b in B], 3),
        ("sum m / S  (word compositions)", [b["sum_m"] / b["S"] for b in B], 3),
        ("max over rounds of (r_new - r)/m", [b["max_ratio"] for b in B], 3),
        ("rows removed in round 1 / n", [b["removed_first"][0] / b["n"] for b in B], 3),
        ("rows removed in rounds 1-4 / n", [b["removed_first"][1] / b["n"] for b in B], 3),
        ("rows removed in rounds 1-16 / n", [b["removed_first"][2] / b["n"] for b in B], 3),
        ("rounds until 50% of rows removed", [b["rounds_to_rows"][0] for b in B], 0),
        ("rounds until 90% of rows removed", [b["rounds_to_rows"][1] for b in B], 0),
        ("rounds until 99% of rows removed", [b["rounds_to_rows"][2] for b in B], 0),
        ("S / r (serial)", [b["S"] / b["r"] for b in B], 3),
    ]
    # the first round's work, from the rounds file
    first = {}
    rp = f"{D}/results/{pol}-rounds.jsonl"
    if os.path.exists(rp):
        for l in open(rp):
            at = l.index('"rounds":[[') + len('"rounds":[[')
            f = [int(x) for x in l[at:l.index("]", at)].split(",")]
            blk = int(l[len('{"block":'):l.index(",")])
            first[blk] = f[1] + f[5]
    out.append(f"\n### {pol}  ({n} blocks)\n")
    out.append("checks: " + ", ".join(f"{k} {v}" for k, v in chk.items()) + "\n")
    out.append("| quantity | min / q1 / median / q3 / max |\n|---|---|")
    for name, v, d in rows:
        if v is None:
            if len(first) != n:
                continue
            v = [first[b["block"]] / b["S"] for b in B]
        out.append(f"| {name} | {fmt(q(v), d)} |")
    med_rounds = statistics.median(b["rounds"] for b in B)
    med_work = statistics.median((b["cells_scanned"] + b["frag_written_before"]) / b["S"] for b in B)
    out.append(f"\ndecision rule: median rounds {med_rounds:.0f} (bound about 200: {'met' if med_rounds <= 200 else 'NOT met'}); median total work {med_work:.2f} S (bound 1.3 S: {'met' if med_work <= 1.3 else 'NOT met'}); blocks with rounds <= 200: {sum(b['rounds'] <= 200 for b in B)} of {n}; blocks with work <= 1.3 S: {sum((b['cells_scanned'] + b['frag_written_before']) <= 1.3 * b['S'] for b in B)} of {n}; least work on a block {min((b['cells_scanned'] + b['frag_written_before']) / b['S'] for b in B):.2f} S")
print("\n".join(out))
