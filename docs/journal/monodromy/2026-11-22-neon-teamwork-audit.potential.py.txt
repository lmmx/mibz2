def lz77(S):
    n=len(S); i=0; z=0
    while i<n:
        best=0
        for st in range(i):
            l=0
            while i+l<n and S[st+l]==S[i+l]: l+=1
            best=max(best,l)
        i+=max(best,1); z+=1
    return z
