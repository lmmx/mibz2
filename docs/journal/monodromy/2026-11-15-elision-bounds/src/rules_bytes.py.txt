"""Bytes each static rule removes alone (pooled), from out/blocks.jsonl."""
import json, sys
B = [json.loads(l) for l in open(sys.argv[1])]
tot = {'slots': 0, 'ind': 0}
R = {k: {'slots': 0, 'ind': 0, 'live': 0} for k in ('R1', 'R2', 'R3', 'R4', 'dead')}
for b in B:
    for key, v in b['t'].items():
        s, f = map(int, key.split('.'))
        stage = 'ind' if 5 <= s < 34 else 'slots'
        by, dead = v[0], v[1]
        if s in (31, 32):
            dead = v[-1]
        tot[stage] += by
        R['dead'][stage] += dead
        live = by - v[1]
        def add(k):
            R[k][stage] += by; R[k]['live'] += live
        slots_slot = s in (0, 1, 34, 35)
        if slots_slot and f == 0: add('R1')
        if (slots_slot and f in (2, 3)) or s == 2: add('R2')
        one = (slots_slot and s >= 34) or (5 <= s < 29 and (s - 5) // 2 % 2 == 1)
        if f == 1 and one and (slots_slot or 5 <= s < 29): add('R3')
        bottom = s in (1, 35) or (5 <= s < 29 and (s - 5) // 4 in (1, 3, 4))
        if bottom and f in (1, 2, 3) and (slots_slot or 5 <= s < 29): add('R4')
T = tot['slots'] + tot['ind']
for k, v in R.items():
    print(f"{k}: slots {v['slots']:,} ({v['slots']/tot['slots']:.4f} of slots) | induction {v['ind']:,} ({v['ind']/tot['ind']:.4f} of induction) | all {(v['slots']+v['ind'])/T:.4f} | of which live (read) bytes {v['live']:,}")
print(tot)
