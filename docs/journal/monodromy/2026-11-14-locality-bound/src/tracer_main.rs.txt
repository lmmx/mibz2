//! Untimed locality study of the Cells induction's step orders (C1a registration).
//!
//! locality CORPUS.bz2 CENSUS.jsonl SLOT KLIST EVERY [OPTS]
//!   SLOT   20 (master) or 24 (the memtrace census's layout, for the cross-check)
//!   KLIST  comma list of greedy windows, e.g. 64 or 64,4096 (0 = none)
//!   EVERY  take every EVERY-th admitted block (1 = all 196)
//! Output: one JSON line per admitted block on stdout.
//!
//! The induction is a port of `Kernel::slots` and `Kernel::induce_cells` (src/cells.rs at
//! origin/master) on a byte buffer laid out exactly as the kernel lays it out; every load and
//! store of the induction is recorded with its offset and width (not the stack, not the
//! writer).

use mibz2::bwt::{InverseBwt, Segmented};
use mibz2::stream::{Event, Scanner, SliceSource};
use mibz2::{decode_block, Kernels, Scratch};
use std::cell::RefCell;
use std::collections::{BTreeSet, HashMap};

thread_local! {
    static CAP: RefCell<Option<(Vec<u8>, u32)>> = const { RefCell::new(None) };
}

#[derive(Default)]
struct Capture(Segmented);
impl InverseBwt for Capture {
    fn invert(&mut self, ll: &[u8], counts: &[u32; 256], p: u32, out: &mut Vec<u8>) {
        CAP.with(|c| *c.borrow_mut() = Some((ll.to_vec(), p)));
        self.0.invert(ll, counts, p, out)
    }
}

const PAD: usize = 4096;
const FAR: usize = 12;
const MARK: u32 = 1 << 31;
const SMALL: u64 = 16;
const BYTES: usize = 16;

// access flags
const W: u8 = 1; // store
const AHEAD: u8 = 2; // the fetch-ahead's two loads (`ahead!`)

#[derive(Clone, Copy)]
struct Acc {
    at: u32,
    w: u8,
    f: u8,
}

#[derive(Default)]
struct Trace {
    // per step: accesses (kernel widths), semantic accesses (for hazards), deps
    acc: Vec<Acc>,
    acc_off: Vec<u32>,
    sem: Vec<Acc>,
    sem_off: Vec<u32>,
    deps: Vec<Vec<u32>>,
    /// accesses after the last step (the closing read), fixed at the end in every order
    epi: Vec<Acc>,
    /// the order stage's (slots) stores, in order: the optional prefix
    pre: Vec<Acc>,
    /// allocations by step, for the second address model: (region, start, step)
    allocs: Vec<(u8, u32, u32, u32)>,
    bots: usize,
    arena: usize,
    // layout
    ft0: usize,
    fb0: usize,
    dyn0: usize,
    rules: usize,
    buf_len: usize,
    letters: usize,
    drops: usize,
}

const NONE: u32 = u32::MAX;

struct Block {
    n: usize,
    recs: Vec<u16>,
    p: u32,
}

fn records(ll: &[u8]) -> Vec<u16> {
    let mut v = Vec::new();
    let mut i = 0;
    while i < ll.len() {
        let c = ll[i];
        let mut j = i;
        while j < ll.len() && ll[j] == c {
            j += 1;
        }
        let mut left = j - i;
        while left > 0 {
            let l = left.min(127);
            v.push((c as u16) << 8 | l as u16);
            left -= l;
        }
        i = j;
    }
    v
}

struct Sim {
    buf: Vec<u8>,
    slot: usize,
}
impl Sim {
    fn rd32(&self, at: usize) -> u32 {
        u32::from_le_bytes(self.buf[at..at + 4].try_into().unwrap())
    }
    fn rd64(&self, at: usize) -> u64 {
        u64::from_le_bytes(self.buf[at..at + 8].try_into().unwrap())
    }
    fn wr32(&mut self, at: usize, v: u32) {
        self.buf[at..at + 4].copy_from_slice(&v.to_le_bytes())
    }
    fn wr64(&mut self, at: usize, v: u64) {
        self.buf[at..at + 8].copy_from_slice(&v.to_le_bytes())
    }
}

/// `slots` + `induce_cells`, traced. Returns the trace and the cell count, or None if refused.
fn run(b: &Block, slot: usize, budget: usize) -> Option<(Trace, usize)> {
    let n = b.n;
    let recs = &b.recs;
    let p = b.p;
    let mut tr = Trace::default();
    // ---- slots (gates as the kernel's, not forced)
    if recs.len() * 10 >= n || recs.len() * 64 < n {
        return None;
    }
    let mut cnt = [0u32; 257];
    let (mut prev, mut row, mut start) = (256usize, 0u32, 0u32);
    let (mut at, mut at_start) = (0usize, 0u32);
    for (i, &rec) in recs.iter().enumerate() {
        let (c, len) = ((rec >> 8) as usize, (rec & 0xFF) as u32);
        if c != prev {
            cnt[c + 1] += 1;
            start = row;
        }
        prev = c;
        if p.wrapping_sub(row) < len {
            (at, at_start) = (i, start);
        }
        row += len;
    }
    assert_eq!(row as usize, n);
    let cz = (recs[at] >> 8) as usize;
    let (mut ia, mut ib) = (at, at + 1);
    while ia > 0 && (recs[ia - 1] >> 8) as usize == cz {
        ia -= 1;
    }
    while ib < recs.len() && (recs[ib] >> 8) as usize == cz {
        ib += 1;
    }
    let total: u32 = recs[ia..ib].iter().map(|&x| (x & 0xFF) as u32).sum();
    let off = p - at_start;
    cnt[cz + 1] += (off > 0) as u32 + (total - off > 1) as u32;
    let letters = cnt.iter().map(|&v| v as usize).sum::<usize>();
    let cap = (recs.len() + 2 + budget) * slot;
    let tops = PAD;
    let bots = tops + cap;
    let arena = bots + cap;
    let rules = arena + BYTES + 256 + 16 + 16 * budget + 32;
    let allocation = rules + 16 * budget + 16;
    let mut s = Sim { buf: vec![0u8; allocation], slot };
    for c in 0..256 {
        s.buf[arena + BYTES + c] = c as u8;
        tr.pre.push(Acc { at: (arena + BYTES + c) as u32, w: 1, f: W });
    }
    for k in 0..256 {
        cnt[k + 1] += cnt[k];
        cnt[k] = (bots + cnt[k] as usize * slot) as u32;
    }
    let first = arena as u64 + BYTES as u64;
    let mut t = PAD;
    {
        let mut letter = |s: &mut Sim, tr: &mut Trace, lam: u32, c: usize, t: &mut usize| -> usize {
            let k = cnt[c] as usize;
            cnt[c] = (k + slot) as u32;
            let lw = 1 | (first + c as u64) << 32;
            s.wr64(*t, t.wrapping_sub(slot) as u32 as u64 | (lam as u64) << 32);
            s.wr64(*t + 8, lw);
            s.wr32(*t + 16, k as u32);
            s.wr64(k, k.wrapping_sub(slot) as u32 as u64 | (lam as u64) << 32);
            s.wr64(k + 8, lw);
            s.wr32(k + 16, *t as u32);
            tr.pre.push(Acc { at: *t as u32, w: 20, f: W });
            tr.pre.push(Acc { at: k as u32, w: 20, f: W });
            let tt = *t;
            *t += slot;
            tt
        };
        let nrec = recs.len();
        let mut i = 0;
        while i < nrec {
            if i == ia {
                if off > 0 {
                    letter(&mut s, &mut tr, off, cz, &mut t);
                }
                let tz = letter(&mut s, &mut tr, 1, cz, &mut t);
                let kz = s.rd32(tz + 16) as usize;
                s.wr32(tz + 8, 1 | MARK);
                s.wr32(kz + 8, 1 | MARK);
                if total - off > 1 {
                    letter(&mut s, &mut tr, total - off - 1, cz, &mut t);
                }
                i = ib;
                continue;
            }
            let rec = recs[i];
            let c = (rec >> 8) as usize;
            let mut lam = (rec & 0xFF) as u32;
            i += 1;
            while i < nrec && (recs[i] >> 8) as usize == c {
                lam += (recs[i] & 0xFF) as u32;
                i += 1;
            }
            letter(&mut s, &mut tr, lam, c, &mut t);
        }
    }
    s.wr32(PAD, PAD as u32);
    s.wr32(bots, bots as u32);
    if letters - 1 > budget {
        return None;
    }
    // ---- induce_cells, traced
    const M: u64 = 0xFFFF_FFFF;
    let rr = letters;
    let mut top = arena + BYTES + 256 + 16;
    let mut rp = rules;
    let (ft0, fb0) = (tops + rr * slot, bots + rr * slot);
    let (mut ft, mut fb) = (ft0, fb0);
    tr.ft0 = ft0;
    tr.fb0 = fb0;
    tr.dyn0 = top;
    tr.rules = rules;
    tr.buf_len = allocation;
    tr.bots = bots;
    tr.arena = arena;
    tr.letters = letters;
    let mut cells = 0usize;
    // writer step of each slot (by slot index, both regions), for the data DAG
    let nslots = 2 * cap / slot + 4;
    let mut writer = vec![NONE; nslots];
    let sidx = |a: usize| -> usize {
        if a >= bots {
            cap / slot + 2 + (a - bots) / slot
        } else {
            (a - tops) / slot
        }
    };
    // pending accesses for the next step
    let mut pacc: Vec<Acc> = Vec::new();
    let mut psem: Vec<Acc> = Vec::new();
    let mut pdeps: Vec<u32> = Vec::new();
    let (mut na, mut nb);
    let (mut ra, mut rb);
    let (mut aw, mut bw);
    let (mut bs, mut ts);
    let (mut sa, mut sb) = (ft0 - slot, fb0 - slot);
    // a slot read whole (fields 0..20 as the kernel's rd32 / rd32 / rd64 / rd_other)
    macro_rules! rdslot {
        ($s:expr) => {{
            let x = $s;
            pacc.push(Acc { at: x as u32, w: 4, f: 0 });
            pacc.push(Acc { at: (x + 4) as u32, w: 4, f: 0 });
            pacc.push(Acc { at: (x + 8) as u32, w: 8, f: 0 });
            pacc.push(Acc { at: (x + 16) as u32, w: 4, f: 0 });
            psem.push(Acc { at: x as u32, w: 20, f: 0 });
            let wi = writer[sidx(x)];
            if wi != NONE {
                pdeps.push(wi);
            }
            (s.rd32(x) as usize, s.rd32(x + 4) as u64, s.rd64(x + 8), s.rd32(x + 16) as usize)
        }};
    }
    macro_rules! ahead {
        ($s:expr) => {{
            let far = $s - FAR * slot;
            pacc.push(Acc { at: (far + 16) as u32, w: 4, f: AHEAD });
            pacc.push(Acc { at: (far + 12) as u32, w: 4, f: AHEAD });
        }};
    }
    // the step being recorded: its own accesses go after the pending ones
    let mut cur: usize;
    let mut acc: Vec<Acc> = Vec::new();
    let mut sem: Vec<Acc> = Vec::new();
    macro_rules! own {
        ($at:expr, $w:expr, $f:expr) => {{
            acc.push(Acc { at: ($at) as u32, w: $w as u8, f: $f });
        }};
    }
    macro_rules! osem {
        ($at:expr, $w:expr, $f:expr) => {{
            if $w > 0 {
                sem.push(Acc { at: ($at) as u32, w: $w as u8, f: $f });
            }
        }};
    }
    macro_rules! close_step {
        ($chain:expr) => {{
            let mut all: Vec<u32> = pdeps.drain(..).collect();
            if let Some(c) = $chain {
                all.push(c);
            }
            all.sort_unstable();
            all.dedup();
            tr.deps.push(all);
            tr.acc.extend(pacc.drain(..));
            tr.acc.extend(acc.drain(..));
            tr.acc_off.push(tr.acc.len() as u32);
            tr.sem.extend(psem.drain(..));
            tr.sem.extend(sem.drain(..));
            tr.sem_off.push(tr.sem.len() as u32);
        }};
    }
    tr.acc_off.push(0);
    tr.sem_off.push(0);
    let mut chain: Option<u32>;
    'fresh: loop {
        (na, ra, aw, bs) = rdslot!(sa);
        (nb, rb, bw, ts) = rdslot!(sb);
        ahead!(sa);
        ahead!(sb);
        chain = None;
        if bs == sb {
            if aw as u32 & MARK != 0 {
                tr.epi.extend(pacc.drain(..));
                psem.clear();
                pdeps.clear();
                return Some((tr, cells));
            }
            tr.drops += 1;
            (sa, sb) = (na, nb);
            continue 'fresh;
        }
        loop {
            if cells == budget {
                return None;
            }
            cur = cells;
            cells += 1;
            let (la, lb) = (aw & M, bw & M);
            let len = la + lb;
            let small = len <= SMALL;
            if small {
                // copy16 twice: 16-byte loads and stores (semantic: the words' own bytes)
                let (wa, wb) = ((aw >> 32) as usize, (bw >> 32) as usize);
                own!(wa, 16, 0);
                own!(top, 16, W);
                own!(wb, 16, 0);
                own!(top + la.min(16) as usize, 16, W);
                osem!(wa, la, 0);
                osem!(wb, lb, 0);
                osem!(top, len, W);
                let tmp: Vec<u8> = s.buf[wa..wa + 16].to_vec();
                s.buf[top..top + 16].copy_from_slice(&tmp);
                let tmp: Vec<u8> = s.buf[wb..wb + 16].to_vec();
                let d = top + la.min(16) as usize;
                s.buf[d..d + 16].copy_from_slice(&tmp);
                tr.allocs.push((0, top as u32, cur as u32, len as u32));
            } else {
                own!(rp, 8, W);
                own!(rp + 8, 8, W);
                osem!(rp, 16, W);
                s.wr64(rp, aw);
                s.wr64(rp + 8, bw);
                tr.allocs.push((1, rp as u32, cur as u32, 16));
            }
            let word = if small { top } else { rp };
            top += if small { len as usize } else { 0 };
            rp += if small { 0 } else { 16 };
            let cw = len | (word as u64) << 32;
            if ra == rb {
                s.wr32(ts + 4, ra as u32);
                s.wr64(ts + 8, cw);
                s.wr32(ts + 16, bs as u32);
                s.wr32(bs + 4, ra as u32);
                s.wr64(bs + 8, cw);
                s.wr32(bs + 16, ts as u32);
                own!(ts + 4, 4, W);
                own!(ts + 8, 8, W);
                own!(ts + 16, 4, W);
                own!(bs + 4, 4, W);
                own!(bs + 8, 8, W);
                own!(bs + 16, 4, W);
                osem!(ts + 4, 16, W);
                osem!(bs + 4, 16, W);
                writer[sidx(ts)] = cur as u32;
                writer[sidx(bs)] = cur as u32;
                close_step!(chain);
                (sa, sb) = (na, nb);
                continue 'fresh;
            } else if ra > rb {
                s.wr32(ts + 4, rb as u32);
                s.wr64(ts + 8, cw);
                s.wr32(ts + 16, bs as u32);
                let v = s.rd32(bs);
                s.wr32(fb, v);
                s.wr32(bs, fb as u32);
                s.wr32(bs + 4, rb as u32);
                s.wr64(bs + 8, cw);
                s.wr32(bs + 16, ts as u32);
                own!(ts + 4, 4, W);
                own!(ts + 8, 8, W);
                own!(ts + 16, 4, W);
                own!(bs, 4, 0);
                own!(fb, 4, W);
                own!(bs, 4, W);
                own!(bs + 4, 4, W);
                own!(bs + 8, 8, W);
                own!(bs + 16, 4, W);
                osem!(ts + 4, 16, W);
                osem!(bs, 4, 0);
                osem!(fb, 4, W);
                osem!(bs, 20, W);
                writer[sidx(ts)] = cur as u32;
                writer[sidx(bs)] = cur as u32;
                writer[sidx(fb)] = cur as u32;
                tr.allocs.push((3, fb as u32, cur as u32, slot as u32));
                close_step!(chain);
                chain = Some(cur as u32);
                bs = fb;
                fb += slot;
                ra -= rb;
                let sn = nb;
                ahead!(sn);
                (nb, rb, bw, ts) = rdslot!(sn);
            } else {
                s.wr32(bs + 4, ra as u32);
                s.wr64(bs + 8, cw);
                s.wr32(bs + 16, ts as u32);
                let v = s.rd32(ts);
                s.wr32(ft, v);
                s.wr32(ts, ft as u32);
                s.wr32(ts + 4, ra as u32);
                s.wr64(ts + 8, cw);
                s.wr32(ts + 16, bs as u32);
                own!(bs + 4, 4, W);
                own!(bs + 8, 8, W);
                own!(bs + 16, 4, W);
                own!(ts, 4, 0);
                own!(ft, 4, W);
                own!(ts, 4, W);
                own!(ts + 4, 4, W);
                own!(ts + 8, 8, W);
                own!(ts + 16, 4, W);
                osem!(bs + 4, 16, W);
                osem!(ts, 4, 0);
                osem!(ft, 4, W);
                osem!(ts, 20, W);
                writer[sidx(ts)] = cur as u32;
                writer[sidx(bs)] = cur as u32;
                writer[sidx(ft)] = cur as u32;
                tr.allocs.push((2, ft as u32, cur as u32, slot as u32));
                close_step!(chain);
                chain = Some(cur as u32);
                ts = ft;
                ft += slot;
                rb -= ra;
                let sn = na;
                ahead!(sn);
                (na, ra, aw, bs) = rdslot!(sn);
            }
        }
    }
}

// ------------------------------------------------------------------ LRU (exact stack distance)
struct Fen {
    t: Vec<i32>,
}
impl Fen {
    fn new(n: usize) -> Self {
        Fen { t: vec![0; n + 1] }
    }
    fn add(&mut self, mut i: usize, v: i32) {
        i += 1;
        while i < self.t.len() {
            self.t[i] += v;
            i += i & i.wrapping_neg();
        }
    }
    fn sum(&self, mut i: usize) -> i64 {
        // sum of [0, i)
        let mut s = 0i64;
        while i > 0 {
            s += self.t[i] as i64;
            i -= i & i.wrapping_neg();
        }
        s
    }
    /// smallest index i with sum(0..=i) >= k (k >= 1)
    fn kth(&self, mut k: i64) -> usize {
        let mut pos = 0usize;
        let mut step = self.t.len().next_power_of_two();
        while step > 0 {
            if pos + step < self.t.len() && (self.t[pos + step] as i64) < k {
                pos += step;
                k -= self.t[pos] as i64;
            }
            step >>= 1;
        }
        pos
    }
}

const CAPS: [usize; 3] = [20480, 81920, 32768]; // 1.25 MiB, 5 MiB, 2 MiB in 64-byte lines

#[derive(Default, Clone)]
struct Eval {
    accesses: u64,
    misses: [u64; 3],
    last_d: usize,
    cold: u64,
    hist: Vec<u64>, // log2 buckets of stack distance; last bucket = cold
}

struct Lru {
    caps: Vec<usize>,
    last: HashMap<u32, usize>,
    fen: Fen,
    now: usize,
    ev: Eval,
}
impl Lru {
    fn new(tmax: usize) -> Self {
        Lru {
            caps: CAPS.to_vec(),
            last: HashMap::new(),
            fen: Fen::new(tmax + 1),
            now: 0,
            ev: Eval { hist: vec![0; 26], ..Default::default() },
        }
    }
    fn touch(&mut self, line: u32) {
        let t = self.now;
        self.now += 1;
        self.ev.accesses += 1;
        match self.last.insert(line, t) {
            None => {
                self.ev.last_d = usize::MAX;
                self.ev.cold += 1;
                for m in self.ev.misses.iter_mut() {
                    *m += 1;
                }
                self.ev.hist[25] += 1;
            }
            Some(p) => {
                let d = (self.fen.sum(t) - self.fen.sum(p + 1)) as usize; // distinct since
                self.ev.last_d = d;
                self.fen.add(p, -1);
                for (i, &c) in self.caps.iter().enumerate() {
                    if d >= c {
                        self.ev.misses[i] += 1;
                    }
                }
                let b = if d == 0 { 0 } else { (usize::BITS - d.leading_zeros()) as usize };
                self.ev.hist[b.min(24)] += 1;
            }
        }
        self.fen.add(t, 1);
    }
    /// time threshold: a line is resident at capacity c iff its last access >= thr
    fn thr(&self, c: usize) -> usize {
        let total = self.last.len() as i64;
        if total <= c as i64 {
            return 0;
        }
        self.fen.kth(total - c as i64) + 1
    }
}

fn lines_of(a: Acc, out: &mut Vec<u32>) {
    let s = a.at as usize / 64;
    let e = (a.at as usize + a.w as usize - 1) / 64;
    for l in s..=e {
        out.push(l as u32);
    }
}

/// address remap for model 2 (allocation in execution order); identity for model 1
struct Remap {
    // per region: sorted starts (R addresses) and new starts
    starts: [Vec<u32>; 4],
    news: [Vec<u32>; 4],
    lo: [u32; 4],
    hi: [u32; 4],
}
impl Remap {
    fn map(&self, a: u32) -> u32 {
        for r in 0..4 {
            if a >= self.lo[r] && a < self.hi[r] {
                let v = &self.starts[r];
                let i = match v.binary_search(&a) {
                    Ok(i) => i,
                    Err(0) => return a,
                    Err(i) => i - 1,
                };
                return self.news[r][i] + (a - v[i]);
            }
        }
        a
    }
}

fn build_remap(tr: &Trace, order: &[u32]) -> Remap {
    // regions: 0 arena (dynamic words), 1 rules, 2 fresh top slots, 3 fresh bottom slots
    let mut by_step: Vec<Vec<usize>> = vec![Vec::new(); tr.deps.len()];
    for (i, &(_, _, st, _)) in tr.allocs.iter().enumerate() {
        by_step[st as usize].push(i);
    }
    let mut bump = [tr.dyn0 as u32, tr.rules as u32, tr.ft0 as u32, tr.fb0 as u32];
    let mut newa = vec![0u32; tr.allocs.len()];
    for &st in order {
        for &i in &by_step[st as usize] {
            let (r, _, _, sz) = tr.allocs[i];
            newa[i] = bump[r as usize];
            bump[r as usize] += sz;
        }
    }
    let mut starts: [Vec<u32>; 4] = Default::default();
    let mut news: [Vec<u32>; 4] = Default::default();
    let mut idx: Vec<usize> = (0..tr.allocs.len()).collect();
    idx.sort_by_key(|&i| (tr.allocs[i].0, tr.allocs[i].1));
    for i in idx {
        let (r, a, _, _) = tr.allocs[i];
        starts[r as usize].push(a);
        news[r as usize].push(newa[i]);
    }
    let lo = [tr.dyn0 as u32, tr.rules as u32, tr.ft0 as u32, tr.fb0 as u32];
    let hi = [tr.rules as u32, tr.buf_len as u32, tr.bots as u32, tr.arena as u32];
    Remap { starts, news, lo, hi }
}

fn evaluate(tr: &Trace, order: &[u32], ahead: bool, prefix: bool, remap: Option<&Remap>) -> Eval {
    let tmax = tr.acc.len() * 3 + tr.pre.len() * 2 + tr.epi.len() * 2 + 16;
    let mut lru = Lru::new(tmax);
    let mut ls = Vec::new();
    let mut feed = |lru: &mut Lru, a: Acc| {
        if !ahead && a.f & AHEAD != 0 {
            return;
        }
        let a = match remap {
            Some(m) => Acc { at: m.map(a.at), ..a },
            None => a,
        };
        ls.clear();
        lines_of(a, &mut ls);
        for &l in ls.iter() {
            lru.touch(l);
        }
    };
    if prefix {
        for &a in &tr.pre {
            feed(&mut lru, a);
        }
    }
    for &st in order {
        let (s, e) = (tr.acc_off[st as usize] as usize, tr.acc_off[st as usize + 1] as usize);
        for &a in &tr.acc[s..e] {
            feed(&mut lru, a);
        }
    }
    if !order.is_empty() {
        for &a in &tr.epi {
            feed(&mut lru, a);
        }
    }
    lru.ev
}

// ------------------------------------------------------------------ validity
/// every dep before its step
fn is_linext(tr: &Trace, order: &[u32]) -> bool {
    let s = tr.deps.len();
    if order.len() != s {
        return false;
    }
    let mut pos = vec![NONE; s];
    for (i, &st) in order.iter().enumerate() {
        if pos[st as usize] != NONE {
            return false;
        }
        pos[st as usize] = i as u32;
    }
    for st in 0..s {
        for &d in &tr.deps[st] {
            if pos[d as usize] >= pos[st] {
                return false;
            }
        }
    }
    true
}

/// every read/write/write hazard of R's semantic accesses (byte-exact) kept by the order:
/// then every step reads exactly the values it read in R, and writes the same, so the rules,
/// the arena and the slots end as in R: the same grammar and derivation tree, the same writer
fn hazards_ok(tr: &Trace, order: &[u32]) -> (bool, u64) {
    let s = tr.deps.len();
    let mut pos = vec![0u32; s];
    for (i, &st) in order.iter().enumerate() {
        pos[st as usize] = i as u32 + 1;
    }
    let mut lastw = vec![0u32; tr.buf_len];
    let mut maxall = vec![0u32; tr.buf_len];
    let mut bad = 0u64;
    for st in 0..s {
        let p = pos[st];
        let (a0, a1) = (tr.sem_off[st] as usize, tr.sem_off[st + 1] as usize);
        for a in &tr.sem[a0..a1] {
            for x in a.at as usize..a.at as usize + a.w as usize {
                if a.f & W != 0 {
                    if p < maxall[x] {
                        bad += 1;
                    }
                    lastw[x] = p;
                } else if p < lastw[x] {
                    bad += 1;
                }
                maxall[x] = maxall[x].max(p);
            }
        }
    }
    (bad == 0, bad)
}

// ------------------------------------------------------------------ greedy
fn greedy(tr: &Trace, k: usize, capi: usize, ahead: bool) -> Vec<u32> {
    let s = tr.deps.len();
    let mut succ_off = vec![0u32; s + 1];
    let mut indeg = vec![0u8; s];
    for st in 0..s {
        for &d in &tr.deps[st] {
            succ_off[d as usize + 1] += 1;
            indeg[st] += 1;
        }
    }
    for i in 0..s {
        succ_off[i + 1] += succ_off[i];
    }
    let mut succ = vec![0u32; succ_off[s] as usize];
    let mut fill = succ_off.clone();
    for st in 0..s {
        for &d in &tr.deps[st] {
            succ[fill[d as usize] as usize] = st as u32;
            fill[d as usize] += 1;
        }
    }
    // per step: its distinct lines (model 1)
    let mut sl: Vec<Vec<u32>> = Vec::with_capacity(s);
    let mut ls = Vec::new();
    for st in 0..s {
        ls.clear();
        for a in &tr.acc[tr.acc_off[st] as usize..tr.acc_off[st + 1] as usize] {
            if !ahead && a.f & AHEAD != 0 {
                continue;
            }
            lines_of(*a, &mut ls);
        }
        ls.sort_unstable();
        ls.dedup();
        sl.push(ls.clone());
    }
    let tmax = tr.acc.len() * 3 + 16;
    let mut lru = Lru::new(tmax);
    let mut ready: BTreeSet<u32> = BTreeSet::new();
    for st in 0..s {
        if indeg[st] == 0 {
            ready.insert(st as u32);
        }
    }
    let cap = CAPS[capi];
    let mut order = Vec::with_capacity(s);
    while let Some(_) = ready.first() {
        let thr = lru.thr(cap);
        let mut best = (u64::MAX, u64::MAX, u32::MAX);
        for &c in ready.iter().take(k) {
            let mut miss = 0u64;
            let mut age = 0u64;
            for &l in &sl[c as usize] {
                match lru.last.get(&l) {
                    Some(&t) if t >= thr => age += (lru.now - t) as u64,
                    _ => {
                        miss += 1;
                        age += lru.now as u64;
                    }
                }
            }
            let key = (miss, age, c);
            if key < best {
                best = key;
            }
        }
        let c = best.2;
        ready.remove(&c);
        order.push(c);
        for a in &tr.acc[tr.acc_off[c as usize] as usize..tr.acc_off[c as usize + 1] as usize] {
            if !ahead && a.f & AHEAD != 0 {
                continue;
            }
            ls.clear();
            lines_of(*a, &mut ls);
            for &l in &ls {
                lru.touch(l);
            }
        }
        for &x in &succ[succ_off[c as usize] as usize..succ_off[c as usize + 1] as usize] {
            indeg[x as usize] -= 1;
            if indeg[x as usize] == 0 {
                ready.insert(x);
            }
        }
    }
    order
}

fn ev_json(e: &Eval) -> String {
    format!(
        "{{\"accesses\":{},\"misses\":[{},{},{}],\"cold\":{},\"hist\":{:?}}}",
        e.accesses, e.misses[0], e.misses[1], e.misses[2], e.cold, e.hist
    )
}

/// the LRU model against a hand-computed trace and against a naive list LRU
fn selftest() {
    let names = ['A', 'B', 'C', 'A', 'D', 'B', 'A'];
    let mut lru = Lru::new(64);
    lru.caps = vec![2, 3, 4];
    println!("hand trace A B C A D B A, capacities 2, 3, 4 lines");
    for (t, &c) in names.iter().enumerate() {
        lru.touch(c as u32);
        let d = lru.ev.last_d;
        println!("t={t} {c} stack distance {}", if d == usize::MAX { "cold".to_string() } else { d.to_string() });
    }
    println!("misses at 2, 3, 4: {:?} (hand: [7, 5, 4]); cold {} (hand: 4)", lru.ev.misses, lru.ev.cold);
    assert_eq!(lru.ev.misses, [7, 5, 4]);
    let thr = lru.thr(3);
    let mut res: Vec<char> = "ABCD".chars().filter(|&c| lru.last[&(c as u32)] >= thr).collect();
    res.sort();
    println!("resident at 3 lines after the trace: {res:?} (hand: [A, B, D])");
    assert_eq!(res, vec!['A', 'B', 'D']);
    // random traces against a naive LRU list, capacities 1..=8, and residency
    let mut x = 12345u64;
    let mut checked = 0;
    for _ in 0..300 {
        let len = 200;
        let caps = [1usize, 3, 8];
        let mut lru = Lru::new(len + 1);
        lru.caps = caps.to_vec();
        let mut naive: Vec<Vec<u32>> = vec![Vec::new(); 3];
        let mut nm = [0u64; 3];
        for _ in 0..len {
            x ^= x << 13; x ^= x >> 7; x ^= x << 17;
            let l = (x % 12) as u32;
            lru.touch(l);
            for (i, &c) in caps.iter().enumerate() {
                let v = &mut naive[i];
                if let Some(p) = v.iter().position(|&y| y == l) {
                    v.remove(p);
                } else {
                    nm[i] += 1;
                    if v.len() == c {
                        v.pop();
                    }
                }
                v.insert(0, l);
                let thr = lru.thr(c);
                for y in 0..12u32 {
                    let r1 = lru.last.get(&y).map_or(false, |&t| t >= thr);
                    assert_eq!(r1, v.contains(&y));
                }
            }
            checked += 1;
        }
        assert_eq!(lru.ev.misses, nm);
    }
    println!("random traces: {checked} accesses, misses and residency equal to a naive LRU list at 1, 3, 8 lines");
}

fn main() {
    let args: Vec<String> = std::env::args().collect();
    if args.get(1).map(|s| s.as_str()) == Some("--selftest") {
        selftest();
        return;
    }
    let data = std::fs::read(&args[1]).expect("corpus");
    let census = std::fs::read_to_string(&args[2]).expect("census");
    let slot: usize = args[3].parse().unwrap();
    let ks: Vec<usize> = args[4].split(',').map(|x| x.parse().unwrap()).filter(|&x| x > 0).collect();
    let every: usize = args[5].parse().unwrap();
    let only: Option<u64> = args.get(6).map(|x| x.parse().unwrap());
    // admitted blocks: index -> (n, r, letters, cells)
    let mut adm: HashMap<u64, (usize, usize, usize, usize)> = HashMap::new();
    let mut adm_list = Vec::new();
    for line in census.lines() {
        if !line.contains("\"outcome\":\"admitted\"") {
            continue;
        }
        let g = |key: &str| -> usize {
            let k = format!("\"{key}\":");
            let i = line.find(&k).unwrap() + k.len();
            line[i..].split(|c: char| !c.is_ascii_digit()).next().unwrap().parse().unwrap()
        };
        let b = g("block") as u64;
        adm.insert(b, (g("n"), g("r"), g("letters"), g("cells")));
        adm_list.push(b);
    }
    let take: std::collections::HashSet<u64> = adm_list
        .iter()
        .enumerate()
        .filter(|(i, _)| i % every == 0)
        .map(|(_, &b)| b)
        .filter(|&b| only.map_or(true, |o| o == b))
        .collect();
    let k = Kernels::detect();
    let mut sc = Scanner::new(SliceSource(&data));
    let mut scr = Scratch::<Capture>::default();
    let mut out = Vec::new();
    let mut syms = Vec::new();
    loop {
        match sc.next(syms).expect("scan") {
            Event::Block { index, symbols, .. } => {
                let want = take.contains(&index);
                if want {
                    out.clear();
                    decode_block(&symbols, k, &mut scr, &mut out).expect("decode");
                    let (ll, p) = CAP.with(|c| c.borrow_mut().take()).expect("captured");
                    let (cn, cr, cl, ccells) = adm[&index];
                    let n = ll.len();
                    let recs = records(&ll);
                    let budget = n / 25 * 4 + n % 25 * 4 / 25;
                    let blk = Block { n, recs, p };
                    let (tr, cells) = run(&blk, slot, budget).expect("admitted by the port");
                    let order_r: Vec<u32> = (0..tr.deps.len() as u32).collect();
                    let mut line = format!(
                        "{{\"block\":{index},\"n\":{n},\"census_n\":{cn},\"r\":{},\"census_r\":{cr},\"letters\":{},\"census_letters\":{cl},\"cells\":{cells},\"census_cells\":{ccells},\"drops\":{},\"slot\":{slot}",
                        blk.recs.len(), tr.letters, tr.drops
                    );
                    let acc_ahead = tr.acc.iter().filter(|a| a.f & AHEAD != 0).count();
                    let bytes_r: u64 = tr.acc.iter().filter(|a| a.f & (W | AHEAD) == 0).map(|a| a.w as u64).sum();
                    let bytes_w: u64 = tr.acc.iter().filter(|a| a.f & W != 0).map(|a| a.w as u64).sum();
                    line += &format!(
                        ",\"loads\":{},\"stores\":{},\"ahead_loads\":{acc_ahead},\"bytes_read\":{},\"bytes_written\":{bytes_w}",
                        tr.acc.iter().filter(|a| a.f & W == 0).count() + tr.epi.len(),
                        tr.acc.iter().filter(|a| a.f & W != 0).count(),
                        bytes_r + tr.epi.iter().map(|a| a.w as u64).sum::<u64>(),
                    );
                    // R: crosscheck form (ahead included), primary (no ahead), with prefix
                    let r_x = evaluate(&tr, &order_r, true, false, None);
                    let r_p = evaluate(&tr, &order_r, false, false, None);
                    let r_pre = evaluate(&tr, &order_r, false, true, None);
                    line += &format!(",\"R_kernel\":{},\"R\":{},\"R_prefix\":{}", ev_json(&r_x), ev_json(&r_p), ev_json(&r_pre));
                    let r_xpre = evaluate(&tr, &order_r, true, true, None);
                    let pre_only = evaluate(&tr, &[], true, true, None);
                    line += &format!(",\"R_kernel_prefix\":{},\"pre_only\":{}", ev_json(&r_xpre), ev_json(&pre_only));
                    let (ok_r, _) = hazards_ok(&tr, &order_r);
                    line += &format!(",\"R_linext\":{},\"R_hazards_ok\":{ok_r}", is_linext(&tr, &order_r));
                    for &kk in &ks {
                        let capis: Vec<usize> = std::env::var("CAPI").map(|v| v.split(',').map(|x| x.parse().unwrap()).collect()).unwrap_or(vec![1, 0, 2]);
                        for capi in capis {
                            let ob = greedy(&tr, kk, capi, false);
                            let lin = is_linext(&tr, &ob);
                            let (hz, nbad) = hazards_ok(&tr, &ob);
                            let b1 = evaluate(&tr, &ob, false, false, None);
                            let bpre = evaluate(&tr, &ob, false, true, None);
                            let rm = build_remap(&tr, &ob);
                            let b2 = evaluate(&tr, &ob, false, false, Some(&rm));
                            let moved = ob.iter().enumerate().filter(|(i, &x)| *i as u32 != x).count();
                            line += &format!(
                                ",\"B_k{kk}_c{capi}\":{{\"linext\":{lin},\"hazards_ok\":{hz},\"hazard_violations\":{nbad},\"moved\":{moved},\"m1\":{},\"m1_prefix\":{},\"m2\":{}}}",
                                ev_json(&b1), ev_json(&bpre), ev_json(&b2)
                            );
                        }
                    }
                    line.push('}');
                    println!("{line}");
                }
                syms = symbols.syms;
            }
            Event::Done => break,
        }
    }
}
