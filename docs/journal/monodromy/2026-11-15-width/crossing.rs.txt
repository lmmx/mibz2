//! scratch census (not committed): LF's cut-crossing numbers on each block.
//! For a cut m of the rows, up(m) = runs whose top part lies below m and whose image reaches
//! at or above m (a < b, a < m < b + λ); down(m) the mirror (b < a, b < m < a + λ).
use mibz2::stream::{Event, Scanner, SliceSource};
use mibz2::{mtf, Kernels};
fn main() {
    let a: Vec<String> = std::env::args().collect();
    let data = std::fs::read(&a[1]).unwrap();
    let k = Kernels::detect();
    let mut sc = Scanner::new(SliceSource(&data));
    let (mut ll, mut counts, mut syms) = (Vec::new(), [0u32; 256], Vec::new());
    let mut pooled = (0f64, 0f64, 0f64, 0f64, 0usize);
    let mut rows = Vec::new();
    while let Event::Block { symbols: b, .. } = sc.next(std::mem::take(&mut syms)).unwrap() {
        mtf::mtf(&b, &mut ll, &mut counts, k).unwrap();
        let n = ll.len();
        if n < 65536 { continue; }
        // runs
        let mut runs: Vec<(usize, usize, u8)> = Vec::new();
        let mut i = 0;
        while i < n { let c = ll[i]; let s = i; while i < n && ll[i] == c { i += 1; } runs.push((s, i - s, c)); }
        let r = runs.len();
        // images: stable sort by byte
        let mut cc = [0usize; 257];
        for &(_, l, c) in &runs { cc[c as usize + 1] += l; }
        for x in 0..256 { cc[x + 1] += cc[x]; }
        let mut up = vec![0i64; n + 2];
        let mut dn = vec![0i64; n + 2];
        let mut fixed = 0usize;
        for &(s, l, c) in &runs {
            let bimg = cc[c as usize]; cc[c as usize] += l;
            if s < bimg { up[s + 1] += 1; up[bimg + l] -= 1; }
            else if bimg < s { dn[bimg + 1] += 1; dn[s + l] -= 1; }
            else { fixed += 1; }
        }
        let (mut cu, mut cd, mut pu, mut pd, mut pboth, mut mean) = (0i64, 0i64, 0i64, 0i64, 0i64, 0f64);
        for m in 0..=n { cu += up[m]; cd += dn[m]; pu = pu.max(cu); pd = pd.max(cd); pboth = pboth.max(cu + cd); mean += (cu + cd) as f64; }
        mean /= (n + 1) as f64;
        rows.push((pu as f64 / r as f64, pd as f64 / r as f64, pboth as f64 / r as f64, mean / r as f64));
        pooled.0 += pu as f64; pooled.1 += pd as f64; pooled.2 += pboth as f64; pooled.3 += r as f64; pooled.4 += 1;
        let _ = fixed;
    }
    let q = |f: &dyn Fn(&(f64, f64, f64, f64)) -> f64| { let mut v: Vec<f64> = rows.iter().map(f).collect(); v.sort_by(|a, b| a.partial_cmp(b).unwrap()); (v[0], v[v.len() / 2], v[v.len() - 1]) };
    println!("blocks {}", pooled.4);
    println!("peak up/r   min/med/max {:?}", q(&|x| x.0));
    println!("peak down/r min/med/max {:?}", q(&|x| x.1));
    println!("peak (up+down)/r       {:?}", q(&|x| x.2));
    println!("mean (up+down)/r       {:?}", q(&|x| x.3));
    println!("pooled peak up/r {:.3} down/r {:.3} both/r {:.3}", pooled.0 / pooled.3, pooled.1 / pooled.3, pooled.2 / pooled.3);
}
