fragments of a copy whenever one time v exists with every source fragment earlier and every
dependent fragment later (times as floats; no byte may depend on another inside the merge)."""

for ci,(s,e) in enumerate(copies):

    for j in range(s,e):
        tv=tau[rowat[j]]
        if cur is None or tv!=ftau[cur]:
            cur=len(ftau); ftau.append(float(tv)); fpos.append([]); fcopy.append(ci)
        fid[j]=cur; fpos[cur].append(j)
src=[-1]*n; dep=[-1]*n
for j in range(n):
    x=rowat[j]
    if not isH[x]: src[j]=t[x-1]
    if x+1<n and not isH[x+1]: dep[j]=t[x+1]
def bounds(ps,own):
    lo=-INF; hi=INF
    for j in ps:
        s_=src[j]
        if s_>=0 and fid[s_]>=0:
            if fid[s_] in own: return None
            lo=max(lo,ftau[fid[s_]])
        d_=dep[j]
        if d_>=0:
            if fid[d_] in own: return None
            hi=min(hi,ftau[fid[d_]])
    return lo,hi
alive=set(range(len(ftau))); F0=len(alive)
# neighbours within a copy: next fragment id in text order
nxt={}
for s,e in copies:
    prev=None
    for j in range(s,e):
        if fid[j]!=prev:
            if prev is not None: nxt[prev]=fid[j]
            prev=fid[j]
rounds=0
while True:
    rounds+=1; merged=0
    for f in sorted(alive):
        if f not in alive or f not in nxt: continue
        g=nxt[f]
        b=bounds(fpos[f]+fpos[g],{f,g})
        if b is None: continue

            for j in fpos[g]: fid[j]=f
            alive.discard(g)
            if g in nxt: nxt[f]=nxt[g]; del nxt[g]
