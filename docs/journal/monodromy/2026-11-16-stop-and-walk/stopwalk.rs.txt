//! Stop the right-end induction at domain size m and walk what is left: the curve of
//! steps, walk points, letters, and word bytes (allocated, and reachable from the live
//! letters) along the induction. Not built by the crate.
//!   stopwalk FILE.bz2 [EVERY] [MIN_ROWS]
use mibz2::stream::{Event, Scanner, SliceSource};
use mibz2::{mtf, Kernels};

const NIL: u32 = u32::MAX;
struct W { len: u64, a: u32, b: u32 } // a == NIL: a byte leaf
fn cost(len: u64) -> u64 { if len > 16 { 16 } else if len > 1 { len } else { 0 } }

fn main() {
    let args: Vec<String> = std::env::args().collect();
    let data = std::fs::read(&args[1]).unwrap();
    let every: usize = args.get(2).map_or(1, |s| s.parse().unwrap());
    let min: usize = args.get(3).map_or(65536, |s| s.parse().unwrap());
    let k = Kernels::detect();
    let mut sc = Scanner::new(SliceSource(&data));
    let (mut ll, mut counts) = (Vec::new(), [0u32; 256]);
    let mut syms = Vec::new();
    let mut i = 0;
    while let Event::Block { symbols: b, .. } = sc.next(std::mem::take(&mut syms)).unwrap() {
        mtf::mtf(&b, &mut ll, &mut counts, k).unwrap();
        if ll.len() >= min {
            if i % every == 0 { run(i, &ll); }
            i += 1;
        }
        syms = b.syms;
    }
}

fn run(blk: usize, ll: &[u8]) {
    let n = ll.len();
    let (mut lam, mut byte) = (Vec::<u64>::new(), Vec::<u8>::new());
    let mut x = 0;
    while x < n {
        let mut y = x + 1;
        while y < n && ll[y] == ll[x] { y += 1; }
        lam.push((y - x) as u64); byte.push(ll[x]); x = y;
    }
    let r = lam.len();
    // bottom order: images sorted by (byte, position)
    let mut bord: Vec<usize> = (0..r).collect();
    bord.sort_by_key(|&j| (byte[j], j));
    // doubly linked orders
    let (mut tp, mut tn, mut bp, mut bn) = (vec![NIL; r], vec![NIL; r], vec![NIL; r], vec![NIL; r]);
    for j in 1..r { tp[j] = (j - 1) as u32; tn[j - 1] = j as u32; }
    for q in 1..r { bp[bord[q]] = bord[q - 1] as u32; bn[bord[q - 1]] = bord[q] as u32; }
    let (mut ttail, mut btail) = (r - 1, bord[r - 1]);
    let mut words: Vec<W> = (0..r).map(|_| W { len: 1, a: NIL, b: NIL }).collect();
    let mut word: Vec<u32> = (0..r as u32).collect();
    let mut touched = vec![false; r];
    let mut alive = vec![true; r];
    let mut m: u64 = n as u64;
    let (mut steps, mut alloc, mut live) = (0u64, 0u64, r as u64);
    let (mut live_touched, mut peak_touched) = (0u64, 0u64);
    let mut eq_steps = 0u64;
    // checkpoints: m <= n * 2^(-k/2)
    let mut kth = 1u32;
    let thr = |k: u32| (n as f64 * 2f64.powf(-(k as f64) / 2.0)) as u64;
    let mut rows: Vec<String> = Vec::new();
    let mut mark = vec![0u32; 0];
    let mut epoch = 0u32;
    let reach = |words: &Vec<W>, word: &Vec<u32>, alive: &Vec<bool>, mark: &mut Vec<u32>, epoch: u32| -> (u64, u64) {
        // reachable stored bytes: a short word is its bytes; a long one is 16 B and its factors
        if mark.len() < words.len() { mark.resize(words.len(), 0); }
        let (mut g, mut long_occ) = (0u64, 0u64);
        let mut st: Vec<u32> = Vec::new();
        for j in 0..word.len() { if alive[j] { st.push(word[j]); } }
        while let Some(w) = st.pop() {
            let wd = &words[w as usize];
            if wd.len > 16 { long_occ += 1; }
            if mark[w as usize] == epoch { continue; }
            mark[w as usize] = epoch;
            g += cost(wd.len);
            if wd.len > 16 { st.push(wd.a); st.push(wd.b); }
        }
        (g, long_occ)
    };
    let kstop: u32 = std::env::var("KSTOP").map_or(0, |v| v.parse().unwrap());
    let lanes: usize = std::env::var("LANES").map_or(8, |v| v.parse().unwrap());
    let kseg: usize = std::env::var("KSEG").map_or(256, |v| v.parse().unwrap());
    let mut hyb = String::new();
    loop {
        if live == 1 { break; }
        let (a, bb) = (ttail, btail);
        assert!(a != bb, "one-letter loop before the end");
        let (la, lb) = (lam[a], lam[bb]);
        let mk = |words: &mut Vec<W>, wb: u32, wa: u32| -> u32 {
            let len = words[wb as usize].len + words[wa as usize].len;
            words.push(W { len, a: wb, b: wa }); (words.len() - 1) as u32
        };
        for &z in &[a, bb] { if !touched[z] { touched[z] = true; live_touched += 1; } }
        if la > lb {
            // top wins: B's word = wB wA; B moves in the bottom order to just after A
            let nw = mk(&mut words, word[a], word[bb]); word[bb] = nw; alloc += cost(words[nw as usize].len);
            lam[a] -= lb; m -= lb;
            // unlink bb from bottom tail
            btail = bp[bb] as usize; bn[btail] = NIL;
            // insert after a in bottom
            let nx = bn[a]; bn[a] = bb as u32; bp[bb] = a as u32; bn[bb] = nx;
            if nx == NIL { btail = bb; } else { bp[nx as usize] = bb as u32; }
        } else if la < lb {
            let nw = mk(&mut words, word[a], word[bb]); word[a] = nw; alloc += cost(words[nw as usize].len);
            lam[bb] -= la; m -= la;
            ttail = tp[a] as usize; tn[ttail] = NIL;
            let nx = tn[bb]; tn[bb] = a as u32; tp[a] = bb as u32; tn[a] = nx;
            if nx == NIL { ttail = a; } else { tp[nx as usize] = a as u32; }
        } else {
            // equal: B's word = wB wA, B takes A's bottom place; A deleted
            let nw = mk(&mut words, word[a], word[bb]); word[bb] = nw; alloc += cost(words[nw as usize].len);
            m -= la; eq_steps += 1;
            ttail = tp[a] as usize; tn[ttail] = NIL;
            btail = bp[bb] as usize; bn[btail] = NIL;
            // bb takes a's bottom place
            let (p, q) = (bp[a], bn[a]);
            bp[bb] = p; bn[bb] = q;
            if p != NIL { bn[p as usize] = bb as u32; }
            if q != NIL { bp[q as usize] = bb as u32; } else { btail = bb; }
            if btail == a { btail = bb; }
            alive[a] = false; live -= 1; live_touched -= 1;
        }
        steps += 1;
        peak_touched = peak_touched.max(live_touched);
        while kth < 60 && m <= thr(kth) {
            epoch += 1;
            let (g, occ) = reach(&words, &word, &alive, &mut mark, epoch);
            // untouched live letters
            let untouched = (0..r).filter(|&j| alive[j] && !touched[j]).count() as u64;
            if kth == kstop { hyb = hybrid(ll, &lam, &byte, &words, &word, &alive, &tp, &tn, &bp, &bn, m, kseg, lanes); }
            rows.push(format!("[{},{},{},{},{},{},{},{}]", kth, m, steps, live, untouched, alloc, g, occ));
            kth += 1;
        }
    }
    let last = (0..r).find(|&j| alive[j]).unwrap();
    assert_eq!(words[word[last] as usize].len as usize, n, "final word not the block");
    println!("{{\"block\":{blk},\"n\":{n},\"r\":{r},\"S\":{steps},\"eq\":{eq_steps},\"alloc\":{alloc},\"peak_touched\":{peak_touched},\"hybrid\":[{hyb}],\"curve\":[{}]}}", rows.join(","));
}

/// The induction stopped at domain [0, m): its letters' tables in top order, K starts, pass 1
/// (segment lengths), the starts ranked, pass 2 (each point's word at its place), checked
/// against the block's own text (rotation(0) at 0). Returns [K, points, ns a point pass 1,
/// ns a point pass 2, long words emitted, bytes ok].
#[allow(clippy::too_many_arguments)]
fn hybrid(ll: &[u8], lam: &[u64], byte: &[u8], words: &[W], word: &[u32], alive: &[bool],
          tp: &[u32], tn: &[u32], bp: &[u32], bn: &[u32], m: u64, kseg: usize, lanes: usize) -> String {
    let n = ll.len();
    let r = lam.len();
    let _ = (tn, bn);
    // heads, then top starts and image starts
    let th = (0..r).find(|&j| alive[j] && tp[j] == NIL).unwrap();
    let bh = (0..r).find(|&j| alive[j] && bp[j] == NIL).unwrap();
    let mut bs = vec![0u64; r];
    let (mut y, mut j) = (0u64, bh as u32);
    while j != NIL { bs[j as usize] = y; y += lam[j as usize]; j = bn[j as usize]; }
    assert_eq!(y, m);
    // top order tables: start, delta = image start - start, word length, word id
    let (mut ts, mut dl, mut wl, mut wi) = (Vec::new(), Vec::new(), Vec::new(), Vec::new());
    let (mut x, mut j) = (0u64, th as u32);
    let mut bits = vec![0u64; (m as usize + 64) / 64 + 1];
    while j != NIL {
        let a = j as usize;
        bits[(x / 64) as usize] |= 1 << (x % 64);
        ts.push(x as u32); dl.push(bs[a] as i64 - x as i64); wl.push(words[word[a] as usize].len as u32); wi.push(word[a]);
        x += lam[a]; j = tn[a];
    }
    assert_eq!(x, m);
    // rank directory: ones before each 64-bit word
    let mut dir = vec![0u32; bits.len() + 1];
    for i in 0..bits.len() { dir[i + 1] = dir[i] + bits[i].count_ones(); }
    let letter = |x: u64| -> usize {
        let (w, b) = ((x / 64) as usize, x % 64);
        (dir[w] + (bits[w] & ((2u64 << b) - 1)).count_ones()) as usize - 1
    };
    let next = |x: u64, i: usize| -> u64 { (x as i64 + dl[i]) as u64 };
    // starts: 0 and K-1 evenly spaced points
    let k = kseg.min(m as usize).max(1);
    let mut st = vec![0u64; (m as usize + 64) / 64 + 1];
    let mut starts: Vec<u64> = (0..k).map(|i| (i as u64 * m) / k as u64).collect();
    starts.dedup();
    let k = starts.len();
    let mut sid = std::collections::HashMap::new();
    for (i, &s0) in starts.iter().enumerate() { st[(s0 / 64) as usize] |= 1 << (s0 % 64); sid.insert(s0, i); }
    let is_start = |x: u64| st[(x / 64) as usize] >> (x % 64) & 1 != 0;
    // optimized kernels on flat tables (counted by callgrind; see pass1/pass2)
    let tb = Tables::new(&bits, &dir, &dl, &wl, &st);
    let mut arena = Arena::new(words, byte, &wi);
    let mut res1 = (vec![0u64; starts.len()], vec![0u32; starts.len()]);
    let iters: usize = std::env::var("ITERS").map_or(1, |v| v.parse().unwrap());
    let to0 = std::time::Instant::now();
    for _ in 0..iters { pass1(&tb, &starts, &mut res1.0, &mut res1.1); }
    let tpass1 = to0.elapsed().as_nanos() as f64 / iters as f64;
    // pass 1: lanes interleaved, each segment walked from its start to the next start
    let t0 = std::time::Instant::now();
    let mut seg_len = vec![0u64; k];
    let mut seg_next = vec![0u64; k];
    let mut seg_pts = vec![0u64; k];
    let mut q = 0usize; // next segment to start
    let mut cur: Vec<(usize, u64, u64, u64)> = Vec::new(); // (segment, x, bytes, points)
    while q < k && cur.len() < lanes { cur.push((q, starts[q], 0, 0)); q += 1; }
    while !cur.is_empty() {
        let mut l = 0;
        while l < cur.len() {
            let (sg, x0, b0, p0) = cur[l];
            let i = letter(x0);
            let x1 = next(x0, i);
            let (b1, p1) = (b0 + wl[i] as u64, p0 + 1);
            if is_start(x1) {
                seg_len[sg] = b1; seg_next[sg] = x1; seg_pts[sg] = p1;
                if q < k { cur[l] = (q, starts[q], 0, 0); q += 1; l += 1; } else { cur.swap_remove(l); }
            } else { cur[l] = (sg, x1, b1, p1); l += 1; }
        }
    }
    let t1 = t0.elapsed().as_nanos() as f64;
    let pts: u64 = seg_pts.iter().sum();
    assert_eq!(pts, m, "segments do not cover the domain once");
    // rank the starts: from start 0 (place 0), places decrease along the walk
    let mut place = vec![u64::MAX; k];
    let (mut s0, mut p) = (0usize, 0u64);
    for _ in 0..k {
        place[s0] = p;
        p = (p + n as u64 * 4 - seg_len[s0]) % n as u64;
        s0 = sid[&seg_next[s0]];
    }
    assert_eq!(s0, 0); assert_eq!(seg_len.iter().sum::<u64>(), n as u64);
    // pass 2: every point's word at its place (a point x's word ends where x's place is)
    let mut out = vec![0u8; n];
    let mut longs = 0u64;
    let t0 = std::time::Instant::now();
    let mut stack: Vec<u32> = Vec::new();
    for sg in 0..k {
        let (mut x, mut pl) = (starts[sg], place[sg]);
        loop {
            let i = letter(x);
            let len = wl[i] as u64;
            let mut at = ((pl + n as u64 * 4 - len) % n as u64) as usize;
            if len > 16 { longs += 1; }
            stack.push(wi[i]);
            while let Some(w) = stack.pop() {
                let wd = &words[w as usize];
                if wd.a == NIL { out[at] = byte[w as usize]; at += 1; if at == n { at = 0; } }
                else { stack.push(wd.b); stack.push(wd.a); }
            }
            pl = (pl + n as u64 * 4 - len) % n as u64;
            x = next(x, i);
            if is_start(x) { break; }
        }
    }
    let t2 = t0.elapsed().as_nanos() as f64;
    assert_eq!(res1.0, seg_len, "pass1 lengths");
    for i in 0..k { assert_eq!(starts[res1.1[i] as usize], seg_next[i], "pass1 successors"); }
    let mut out2 = vec![0u8; n + 16];
    let to0 = std::time::Instant::now();
    // segments in chain order from start 0: each lies just below the one before
    let mut order = Vec::with_capacity(k);
    let mut s0 = 0usize;
    for _ in 0..k { order.push(s0); s0 = res1.1[s0] as usize; }
    for _ in 0..iters { pass2(&tb, &mut arena, &starts, &place, &order, n, &mut out2); }
    let tpass2 = to0.elapsed().as_nanos() as f64 / iters as f64;
    assert!(out2[16..] == out[..], "pass2 bytes");
    let mut out3 = vec![0u8; n + 16];
    for _ in 0..iters { pass2l(&tb, &arena, &starts, &place, &res1.0, n, &mut out3); }
    if out3[16..] != out[..] {
        let bad: Vec<usize> = (0..n).filter(|&j| out3[16 + j] != out[j]).collect();
        let mut sp: Vec<(u64, u64)> = (0..k).map(|i| (place[i], res1.0[i])).collect(); sp.sort();
        let near: Vec<&(u64, u64)> = sp.iter().filter(|(p0, _)| (*p0 as i64 - bad[0] as i64).abs() < 40).collect();
        panic!("pass2l bytes: {} bad, first {:?}, segments (top, len) near {:?}", bad.len(), &bad[..bad.len().min(20)], near);
    }
    let _ = (t1, t2);
    let (t1, t2) = (tpass1, tpass2);
    // the reference: rotation(0) at place 0, LF going back one place
    let mut c = [0usize; 257];
    for &b in ll { c[b as usize + 1] += 1; }
    for i in 0..256 { c[i + 1] += c[i]; }
    let mut lf = vec![0u32; n];
    for (i, &b) in ll.iter().enumerate() { lf[i] = c[b as usize] as u32; c[b as usize] += 1; }
    let mut text = vec![0u8; n];
    let mut xx = 0usize;
    for i in 1..=n { text[n - i] = ll[xx]; xx = lf[xx] as usize; }
    let ok = text == out;
    assert!(ok, "hybrid output differs from the block");
    format!("{k},{m},{:.2},{:.2},{longs},{}", t1 / m as f64, t2 / m as f64, ok as u8)
}

/// Flat tables of the stopped exchange, in top order
struct Tables { bits: Vec<u64>, dir: Vec<u32>, dl: Vec<i32>, wl: Vec<u32>, st: Vec<u64> }
impl Tables {
    fn new(bits: &[u64], dir: &[u32], dl: &[i64], wl: &[u32], st: &[u64]) -> Tables {
        Tables { bits: bits.to_vec(), dir: dir.to_vec(), dl: dl.iter().map(|&d| d as i32).collect(), wl: wl.to_vec(), st: st.to_vec() }
    }
    #[inline(always)]
    fn letter(&self, x: u32) -> usize {
        let (w, b) = ((x >> 6) as usize, x & 63);
        unsafe { (*self.dir.get_unchecked(w) + (*self.bits.get_unchecked(w) & ((2u64 << b) - 1)).count_ones()) as usize - 1 }
    }
    #[inline(always)]
    fn is_start(&self, x: u32) -> bool { unsafe { *self.st.get_unchecked((x >> 6) as usize) >> (x & 63) & 1 != 0 } }
}

/// Pass 1: eight lanes, each walking a segment from its start to the next start, summing
/// bytes. Writes each segment's bytes and the index of the start it reaches.
#[inline(never)]
fn pass1(t: &Tables, starts: &[u64], len: &mut [u64], nxt: &mut [u32]) {
    const L: usize = 8;
    let k = starts.len();
    let mut idx = std::collections::HashMap::with_capacity(k);
    for (i, &s) in starts.iter().enumerate() { idx.insert(s as u32, i as u32); }
    let (mut seg, mut x, mut b) = ([usize::MAX; L], [0u32; L], [0u64; L]);
    let mut q = 0usize;
    let mut live = 0usize;
    for l in 0..L { if q < k { seg[l] = q; x[l] = starts[q] as u32; q += 1; live += 1; } }
    while live > 0 {
        for l in 0..L {
            if seg[l] == usize::MAX { continue; }
            let i = t.letter(x[l]);
            let (d, w) = unsafe { (*t.dl.get_unchecked(i), *t.wl.get_unchecked(i)) };
            let y = (x[l] as i32 + d) as u32;
            b[l] += w as u64;
            x[l] = y;
            if t.is_start(y) {
                len[seg[l]] = b[l]; nxt[seg[l]] = idx[&y];
                b[l] = 0;
                if q < k { seg[l] = q; x[l] = starts[q] as u32; q += 1; } else { seg[l] = usize::MAX; live -= 1; }
            }
        }
    }
}

/// The words a stopped exchange's letters carry: a short word (at most 16 bytes) right-
/// aligned in a 16-byte slot; a long one a rule of two words.
struct Arena { slot: Vec<[u8; 16]>, kind: Vec<u32>, rule: Vec<(u32, u32, u32)>, top: Vec<u32> }
const SHORT: u32 = 1 << 31;
impl Arena {
    fn new(words: &[W], byte: &[u8], wi: &[u32]) -> Arena {
        let mut a = Arena { slot: Vec::new(), kind: vec![u32::MAX; words.len()], rule: Vec::new(), top: Vec::new() };
        for &w in wi { let z = a.put(words, byte, w); a.top.push(z); }
        a
    }
    fn bytes(words: &[W], byte: &[u8], w: u32, out: &mut Vec<u8>) {
        let mut st = vec![w];
        while let Some(v) = st.pop() { let d = &words[v as usize]; if d.a == NIL { out.push(byte[v as usize]); } else { st.push(d.b); st.push(d.a); } }
    }
    fn put(&mut self, words: &[W], byte: &[u8], w: u32) -> u32 {
        if self.kind[w as usize] != u32::MAX { return self.kind[w as usize]; }
        let d = &words[w as usize];
        let z = if d.len <= 16 {
            let mut v = Vec::new(); Arena::bytes(words, byte, w, &mut v);
            let mut s = [0u8; 16]; s[16 - v.len()..].copy_from_slice(&v);
            self.slot.push(s); SHORT | (self.slot.len() - 1) as u32
        } else {
            let (a, b) = (d.a, d.b);
            let za = self.put(words, byte, a); let zb = self.put(words, byte, b);
            self.rule.push((za, zb, words[b as usize].len as u32)); (self.rule.len() - 1) as u32
        };
        self.kind[w as usize] = z; z
    }
}

/// Pass 2: every segment from its start, each point's word written right to left ending at
/// its place (`out` has 16 bytes of slack before place 0).
#[inline(never)]
fn pass2(t: &Tables, a: &mut Arena, starts: &[u64], place: &[u64], order: &[usize], n: usize, out: &mut [u8]) {
    let mut stack: Vec<(u32, u32)> = Vec::with_capacity(64);
    let o = out.as_mut_ptr();
    for &sg in order {
        let (mut x, mut pl) = (starts[sg] as u32, place[sg] as usize);
        loop {
            let i = t.letter(x);
            let len = unsafe { *t.wl.get_unchecked(i) } as usize;
            let z = unsafe { *a.top.get_unchecked(i) };
            if len <= pl {
                // right to left: (word, end)
                stack.push((z, pl as u32));
                while let Some((w, e)) = stack.pop() {
                    if w & SHORT != 0 {
                        unsafe { std::ptr::copy_nonoverlapping(a.slot.as_ptr().add((w & !SHORT) as usize) as *const u8, o.add(e as usize), 16); }
                    } else {
                        let (fa, fb, lb) = unsafe { *a.rule.get_unchecked(w as usize) };
                        stack.push((fa, e - lb)); stack.push((fb, e));
                    }
                }
            } else {
                // the one word across place 0: slowly, byte by byte
                let mut tmp = vec![0u8; len + 16];
                stack.push((z, (len) as u32));
                while let Some((w, e)) = stack.pop() {
                    if w & SHORT != 0 { tmp[e as usize..e as usize + 16].copy_from_slice(&a.slot[(w & !SHORT) as usize]); }
                    else { let (fa, fb, lb) = a.rule[w as usize]; stack.push((fa, e - lb)); stack.push((fb, e)); }
                }
                // tmp[16 .. 16+len] is the word; its last pl bytes go to [0,pl), the rest to the end
                for j in 0..len { let p = (pl + n - len + j) % n; out[16 + p] = tmp[16 + j]; }
            }
            pl = if len <= pl { pl - len } else { pl + n - len };
            x = (x as i32 + unsafe { *t.dl.get_unchecked(i) }) as u32;
            if t.is_start(x) { break; }
        }
    }
}

/// The general path: a word (short or long) ending at place `e`; `exact` writes no byte
/// below the word (a segment's last word: the place below is another segment's).
#[inline(never)]
fn emit_slow(a: &Arena, z: u32, len: usize, e: usize, n: usize, exact: bool, out: &mut [u8]) {
    let mut tmp = [0u8; 0];
    let _ = &mut tmp;
    if len <= e && !exact {
        let o = out.as_mut_ptr();
        let mut stack: [(u32, u32); 64] = [(0, 0); 64];
        let mut sp = 1;
        stack[0] = (z, e as u32);
        while sp > 0 {
            sp -= 1;
            let (w, e) = stack[sp];
            if w & SHORT != 0 {
                unsafe { std::ptr::copy_nonoverlapping(a.slot.as_ptr().add((w & !SHORT) as usize) as *const u8, o.add(e as usize), 16); }
            } else {
                let (fa, fb, lb) = unsafe { *a.rule.get_unchecked(w as usize) };
                stack[sp] = (fa, e - lb); stack[sp + 1] = (fb, e); sp += 2;
            }
        }
    } else {
        let mut v = vec![0u8; len + 16];
        let mut stack = vec![(z, len as u32)];
        while let Some((w, e)) = stack.pop() {
            if w & SHORT != 0 { v[e as usize..e as usize + 16].copy_from_slice(&a.slot[(w & !SHORT) as usize]); }
            else { let (fa, fb, lb) = a.rule[w as usize]; stack.push((fa, e - lb)); stack.push((fb, e)); }
        }
        for j in 0..len { let p = (e + n - len + j) % n; out[16 + p] = v[16 + j]; }
    }
}

/// Pass 2, lean: eight lanes; a short word not last in its segment is one 16-byte store.
/// A store reaches up to 15 bytes below its word, so a segment's lowest words can reach into
/// the segment below it, written by another lane: each segment's top 16 bytes are kept when
/// written and put back at the end.
#[inline(never)]
fn pass2l(t: &Tables, a: &Arena, starts: &[u64], place: &[u64], seg_len: &[u64], n: usize, out: &mut [u8]) {
    const L: usize = 8;
    let k = starts.len();
    let o = out.as_mut_ptr();
    let mut keep = vec![[0u8; 16]; k];
    let (mut x, mut pl, mut sg, mut done, mut on) = ([0u32; L], [0usize; L], [0usize; L], [0usize; L], [false; L]);
    let mut q = 0usize;
    let mut live = 0;
    let snap = |out: &[u8], top: usize| -> [u8; 16] { let mut v = [0u8; 16]; for j in 0..16 { v[j] = out[16 + (top + n - 16 + j) % n]; } v };
    for l in 0..L { if q < k { x[l] = starts[q] as u32; pl[l] = place[q] as usize; sg[l] = q; done[l] = 0; on[l] = true; q += 1; live += 1; } }
    while live > 0 {
        for l in 0..L {
            if !on[l] { continue; }
            let i = t.letter(x[l]);
            let (d, len, z) = unsafe { (*t.dl.get_unchecked(i), *t.wl.get_unchecked(i) as usize, *a.top.get_unchecked(i)) };
            let y = (x[l] as i32 + d) as u32;
            let last = t.is_start(y);
            let e = pl[l];
            if z & SHORT != 0 && len <= e {
                unsafe { std::ptr::copy_nonoverlapping(a.slot.as_ptr().add((z & !SHORT) as usize) as *const u8, o.add(e), 16); }
            } else {
                emit_slow(a, z, len, e, n, false, out);
            }
            let before = done[l];
            done[l] += len;
            if before < 16 {
                // this word's bytes within its segment's top 16, kept now: another lane's
                // store may reach them before this lane writes again
                let top = place[sg[l]] as usize;
                let v = snap(out, top);
                for j in 16 - done[l].min(16)..16 - before { keep[sg[l]][j] = v[j]; }
            }
            pl[l] = if len <= e { e - len } else { e + n - len };
            x[l] = y;
            if last {
                if q < k { x[l] = starts[q] as u32; pl[l] = place[q] as usize; sg[l] = q; done[l] = 0; q += 1; } else { on[l] = false; live -= 1; }
            }
        }
    }
    for s0 in 0..k {
        let top = place[s0] as usize;
        let w = (seg_len[s0] as usize).min(16);
        for j in 16 - w..16 { out[16 + (top + n - 16 + j) % n] = keep[s0][j]; }
    }
}
