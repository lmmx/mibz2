//! Rows (N), decoded bytes (D) and the cycles of each block's permutation, per block
use mibz2::stream::{Event, Scanner, SliceSource};
use mibz2::{bwt, mtf, rle, Kernels};

fn main() {
    let path = std::env::args().nth(1).expect("team_rows FILE.bz2");
    let data = std::fs::read(path).unwrap();
    let k = Kernels::detect();
    let mut sc = Scanner::new(SliceSource(&data));
    let (mut syms, mut ll, mut counts) = (Vec::new(), Vec::new(), [0u32; 256]);
    let (mut tt, mut walked, mut out) = (bwt::TVector::new(), Vec::new(), Vec::new());
    let (mut big_n, mut big_d, mut blocks, mut multi, mut coop_n, mut coop_blocks) = (0u64, 0u64, 0, 0, 0u64, 0);
    loop {
        let Some(Event::Block { symbols: b, .. }) = Some(sc.next(std::mem::take(&mut syms)).unwrap()) else { break };
        let n = mtf::mtf(&b, &mut ll, &mut counts, k).unwrap();
        // cycles of LF: T[cum[c] + rank] = i, i.e. the build's link
        let mut cum = [0usize; 256];
        let mut s = 0;
        for c in 0..256 { cum[c] = s; s += counts[c] as usize; }
        let mut link = vec![0u32; n];
        for (i, &c) in ll.iter().enumerate() { link[cum[c as usize]] = i as u32; cum[c as usize] += 1; }
        let mut seen = vec![false; n];
        let mut cycles = 0;
        for r in 0..n { if !seen[r] { cycles += 1; let mut x = r; while !seen[x] { seen[x] = true; x = link[x] as usize; } } }
        if cycles > 1 { multi += 1; }
        bwt::build(&ll, &counts, &mut tt);
        bwt::walk(&tt, b.orig_ptr, &mut walked);
        out.clear();
        rle::unrle(&walked, &mut out, k).unwrap();
        println!("block {blocks} rows {n} decoded {} cycles {cycles}", out.len());
        big_n += n as u64; big_d += out.len() as u64; blocks += 1;
        if n >= mibz2::bwt::MIN_COOP { coop_n += n as u64; coop_blocks += 1; }
        syms = b.syms;
    }
    println!("TOTAL blocks {blocks} rows {big_n} decoded {big_d} multicycle_blocks {multi} coop_blocks {coop_blocks} coop_rows {coop_n}");
}
