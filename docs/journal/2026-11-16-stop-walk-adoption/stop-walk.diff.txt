+#!/usr/bin/env python3
+"""The stop-and-walk adoption run (`just bench-stop-walk`): its verdict, read from the run's
+records only after the records are checked whole. Standard library only.
+
+    stop_walk.py DIR [--rounds 10] [--threads 8,9,10,12,20] [--copies 4]
+
+DIR holds the run's four record files (`ab run`'s JSON lines):
+- `judged.jsonl`: `mibz2` and `mibz2-stop15` on the main corpus (`wikidata-xC`, C the copies
+  of the Wikidata sample, `--copies`) and the text
+  corpus (`text`), at the judged thread counts, every round;
+- `forced.jsonl` (diagnostic): `mibz2-workers-lazy`, `-stop14`, `-stop15`, `-stop16` on the main
+  corpus at the judged counts;
+- `overlap.jsonl` (diagnostic): `mibz2` and `mibz2-stop15` on the main corpus at 10 threads with
+  the held spinner sharing a decoder core;
+- `dose.jsonl` (diagnostic): the forced arms on one copy (`wikidata`) at 1 and 10 threads.
+
+**The registration** (#145; the Researcher's correction REVIEW 6921090225, accepted in LOG
+6921203037). It decides whether stop-and-walk becomes *the* lazy kernel, where production
+runs lazy workers now, and nothing else: no thread threshold of its own.
+- *The arms:* `mibz2` is production's decoder. `mibz2-stop15` is production's decoder, plan and
+  window with the lazy kernel replaced by stop-and-walk at β = 1.5
+  (`StreamDecoder::new_lazy_as`), so the two differ in the kernel alone. On the i9, 12 and 20
+  threads asked are ten lazy workers, as production runs them. Only β = 1.5 is eligible.
+- *The judged run* has no spinner sharing a decoder core (the spinner is held only where a
+  core is free: at 8 and 9 threads). The 10-thread case with the spinner sharing core 9 is a
+  diagnostic.
+- *A cell's step:* the rounds' paired wall ratios, `mibz2-stop15` over `mibz2`, their geometric
+  mean and its 95% interval (Student's t on the logs). FASTER if the interval lies below 1,
+  SLOWER if above, else INCONCLUSIVE (not equal).
+- *Gain:* Holm's one-sided tests (the mean log ratio below 0) over the five main-corpus cells,
+  family α = 0.05. The gain is established if Holm rejects at one cell or more.
+- *Loss screen:* no judged cell, main or text, is SLOWER. (A noisy loss can still pass a
+  screen that only finds the losses its intervals resolve.)
+- *Memory gate:* at every judged cell, main and text, the median over the rounds of each run's
+  peak resident memory, `mibz2-stop15` over `mibz2`, is at most 1.10.
+- **ADOPTS:** gain established, loss screen clean, memory gate passed.
+- **HELD ON MEMORY:** gain established and loss screen clean, memory gate failed: the
+  Coordinator weighs it under PERFORMANCE-POLICY rule 4.
+- **A FINDING, NOT ADOPTED:** gain established and a cell SLOWER. No range of its own: any
+  change would be to the lazy threshold itself, a separate decision.
+- **NOT ADOPTED:** otherwise.
+- **VOID** (exit 1, no verdict): any record file missing or unreadable; any record not a
+  successful run with finite positive wall, CPU and peak memory; any key (corpus, arm, threads,
+  round) duplicated, unexpected or missing from the full roster; a corpus whose outputs (size
+  and hash) differ; a judged, forced or dose run with the spinner sharing a decoder core.
+"""
+import argparse
+import json
+import math
+import os
+import statistics
+import sys
+
+JUDGED_ARMS = ["mibz2", "mibz2-stop15"]
+FORCED_ARMS = ["mibz2-workers-lazy", "mibz2-workers-stop14", "mibz2-workers-stop15", "mibz2-workers-stop16"]
+MAIN, GUARD, ONE = "wikidata-x4", "text", "wikidata"
+ALPHA, RSS_GATE = 0.05, 1.10
+# Student's t, two-sided 95%, by degrees of freedom (1..30); 1.96 beyond
+T95 = [12.706, 4.303, 3.182, 2.776, 2.571, 2.447, 2.365, 2.306, 2.262, 2.228, 2.201, 2.179,
+       2.160, 2.145, 2.131, 2.120, 2.110, 2.101, 2.093, 2.086, 2.080, 2.074, 2.069, 2.064, 2.060,
+       2.056, 2.052, 2.048, 2.045, 2.042]
+
+
+def betainc(a, b, x):
+    """The regularized incomplete beta function I_x(a, b) (a continued fraction, Lentz)"""
+    if x <= 0:
+        return 0.0
+    if x >= 1:
+        return 1.0
+    if x > (a + 1) / (a + b + 2):
+        return 1.0 - betainc(b, a, 1.0 - x)
+    lfront = math.lgamma(a + b) - math.lgamma(a) - math.lgamma(b) + a * math.log(x) + b * math.log1p(-x)
+    tiny = 1e-300
+    c, d = 1.0, 1.0 - (a + b) * x / (a + 1)
+    d = 1.0 / (d if abs(d) > tiny else tiny)
+    f = d
+    for m in range(1, 400):
+        for num in (m * (b - m) * x / ((a + 2 * m - 1) * (a + 2 * m)),
+                    -(a + m) * (a + b + m) * x / ((a + 2 * m) * (a + 2 * m + 1))):
+            d = 1.0 + num * d
+            d = 1.0 / (d if abs(d) > tiny else tiny)
+            c = 1.0 + num / c
+            c = c if abs(c) > tiny else tiny
+            f *= c * d
+        if abs(c * d - 1.0) < 1e-15:
+            break
+    return math.exp(lfront) * f / a
+
+
+def t_cdf(t, df):
+    """P(T <= t), T Student's t with df degrees of freedom"""
+    if math.isinf(t):
+        return 0.0 if t < 0 else 1.0
+    tail = 0.5 * betainc(df / 2, 0.5, df / (df + t * t))
+    return tail if t < 0 else 1.0 - tail
+
+
+def step(logs):
+    """(geometric mean, 95% low, 95% high, one-sided p that the mean log is below 0)"""
+    n = len(logs)
+    mu = statistics.mean(logs)
+    sd = statistics.stdev(logs)
+    t95 = T95[n - 2] if n - 1 <= len(T95) else 1.96
+    h = t95 * sd / math.sqrt(n)
+    if sd == 0:
+        p = 0.0 if mu < 0 else 1.0
+    else:
+        p = t_cdf(mu / (sd / math.sqrt(n)), n - 1)
+    return math.exp(mu), math.exp(mu - h), math.exp(mu + h), p
+
+
+def label(lo, hi):
+    return "FASTER" if hi < 1 else "SLOWER" if lo > 1 else "INCONCLUSIVE"
+
+
+def holm(ps, alpha):
+    """Which of `ps` (a dict) Holm's step-down procedure rejects at family `alpha`"""
+    rejected, m = set(), len(ps)
+    for i, (key, p) in enumerate(sorted(ps.items(), key=lambda kv: kv[1])):
+        if p > alpha / (m - i):
+            break
+        rejected.add(key)
+    return rejected
+
+
+def load(path, roster, errors, outputs, spinner_ok):
+    """The file's records by key, after checking each record and the roster; errors appended"""
+    name = os.path.basename(path)
+    rows = {}
+    if not os.path.exists(path):
+        errors.append(f"{name}: missing")
+        return rows
+    with open(path) as f:
+        for ln, line in enumerate(f, 1):
+            if not line.strip():
+                continue
+            where = f"{name}:{ln}"
+            try:
+                r = json.loads(line)
+            except ValueError as e:
+                errors.append(f"{where}: not JSON ({e})")
+                continue
+            if not isinstance(r, dict):
+                errors.append(f"{where}: not a record")
+                continue
+            try:
+                key = (str(r["corpus"]), str(r["arm"]), int(r["threads"]), int(r["round"]))
+            except (KeyError, TypeError, ValueError):
+                errors.append(f"{where}: no corpus, arm, threads or round")
+                continue
+            if key in rows:
+                errors.append(f"{where}: {key} duplicated")
+                continue
+            if key not in roster:
+                errors.append(f"{where}: {key} not in the roster")
+            if r.get("err") or r.get("termination") != "success":
+                errors.append(f"{where}: {key} did not succeed ({r.get('termination')!r}, {r.get('err')!r})")
+            for field in ("wall", "cpu", "rss_kb"):
+                v = r.get(field)
+                if isinstance(v, bool) or not isinstance(v, (int, float)) or not math.isfinite(v) or v <= 0:
+                    errors.append(f"{where}: {key} {field} {v!r} is not a finite positive number")
+            out, digest = r.get("out"), r.get("sha256") or r.get("hash")
+            if isinstance(out, bool) or not isinstance(out, int) or out <= 0 or not isinstance(digest, str) or not digest:
+                errors.append(f"{where}: {key} output size {out!r} or hash {digest!r} missing")
+            else:
+                outputs.setdefault(key[0], set()).add((out, digest))
+            hs = r.get("held_spinner")
+            if not spinner_ok and isinstance(hs, dict) and hs.get("overlap"):
+                errors.append(f"{where}: {key} ran beside a spinner sharing a decoder core")
+            rows[key] = r
+    missing = sorted(roster - set(rows))
+    if missing:
+        errors.append(f"{name}: {len(missing)} of {len(roster)} roster keys missing, e.g. {missing[:3]}")
+    return rows
+
+
+def roster(corpora, arms, threads, rounds):
+    return {(c, a, t, i) for c in corpora for a in arms for t in threads for i in range(rounds)}
+
+
+def cell(rows, corpus, arm, ref, t, rounds, key="wall"):
+    return [math.log(rows[(corpus, arm, t, i)][key] / rows[(corpus, ref, t, i)][key]) for i in range(rounds)]
+
+
+def peak(rows, corpus, arm, t, rounds):
+    return statistics.median(rows[(corpus, arm, t, i)]["rss_kb"] for i in range(rounds))
+
+
+def main(argv):
+    ap = argparse.ArgumentParser()
+    ap.add_argument("dir")
+    ap.add_argument("--rounds", type=int, default=10)
+    ap.add_argument("--threads", default="8,9,10,12,20")
+    ap.add_argument("--copies", type=int, default=4)
+    a = ap.parse_args(argv[1:])
+    global MAIN
+    MAIN = f"wikidata-x{a.copies}"
+    threads = [int(t) for t in a.threads.split(",")]
+    R = a.rounds
+    errors, outputs = [], {}
+    if R < 2:
+        errors.append(f"--rounds {R}: at least 2")
+    files = {
+        "judged": (roster([MAIN, GUARD], JUDGED_ARMS, threads, R), False),

+        "dose": (roster([ONE], FORCED_ARMS, [1, 10], R), False),

+    steps, ps, mem = {}, {}, {}
+    for corpus in (MAIN, GUARD):
+        print(f"\n{corpus}{' (gain family)' if corpus == MAIN else ' (loss and memory guard)'}")
+        for t in threads:

+                  + (f"; p {p:.2e}" if corpus == MAIN else "") + f"; memory {mem[(corpus, t)]:.3f}")
+    rejected = holm(ps, ALPHA)
+    print(f"\nHolm at family α {ALPHA} over the {len(ps)} main-corpus cells: "
+          + ", ".join(f"{t} p {ps[t]:.2e} {'rejects' if t in rejected else 'does not'}"
+                      for t in sorted(ps, key=ps.get)))
+    gain = bool(rejected)
+    slower = sorted(k for k, v in steps.items() if v == "SLOWER")
+    over = sorted(k for k, v in mem.items() if v > RSS_GATE)
+    print()

+        print(f"VERDICT: ADOPTS. A gain at {sorted(rejected)} threads (Holm), no judged cell SLOWER, and peak memory "
+              f"within {RSS_GATE:.2f} of production's at every judged cell: stop-and-walk at β = 1.5 becomes the lazy kernel")
+    elif gain and not slower:
+        print(f"VERDICT: HELD ON MEMORY. A gain at {sorted(rejected)} threads (Holm) and no judged cell SLOWER, but peak "
+              f"memory is above {RSS_GATE:.2f} of production's at {over}: PERFORMANCE-POLICY rule 4, the Coordinator's to weigh")
+    elif gain:
+        print(f"VERDICT: A FINDING, NOT ADOPTED. A gain at {sorted(rejected)} threads (Holm), but SLOWER at {slower}. "
+              "No range of its own: any change would be to the lazy threshold itself, a separate decision")
+    else:
+        print("VERDICT: NOT ADOPTED. No gain established (Holm rejects at no main-corpus cell)"
+              + (f"; SLOWER at {slower}" if slower else "") + (f"; memory above the gate at {over}" if over else ""))
+
+    print("\nDiagnostics (decide nothing):")
+    F = data["forced"]
+    print("  forced workers, each arm over mibz2-workers-lazy on the main corpus (12 and 20 are that many workers):")
+    for arm in FORCED_ARMS[1:]:
+        cells = []
+        for t in threads:
+            g, lo, hi, _ = step(cell(F, MAIN, arm, "mibz2-workers-lazy", t, R))
+            cells.append(f"{t} {g:.3f} {label(lo, hi)}")
+        print(f"    {arm}: " + ", ".join(cells))
+    O = data["overlap"]
+    g, lo, hi, _ = step(cell(O, MAIN, "mibz2-stop15", "mibz2", 10, R))
+    shared = sum(bool((r.get("held_spinner") or {}).get("overlap")) for r in O.values())
+    print(f"  10 threads, the spinner sharing a decoder core ({shared} of {len(O)} runs shared): "
+          f"{g:.4f} [{lo:.4f}, {hi:.4f}] {label(lo, hi)}")
+    D = data["dose"]
+    for arm in FORCED_ARMS[1:]:
+        g, lo, hi, _ = step(cell(D, ONE, arm, "mibz2-workers-lazy", 1, R))
+        print(f"  1 thread, {arm} over mibz2-workers-lazy: wall {g:.4f} [{lo:.4f}, {hi:.4f}]")
+    dose = {arm: {i: D[(ONE, arm, 10, i)]["cpu"] / D[(ONE, arm, 1, i)]["cpu"] for i in range(R)} for arm in FORCED_ARMS}
+    for arm in FORCED_ARMS:
+        g, lo, hi, _ = step([math.log(v) for v in dose[arm].values()])

+def run(files, rounds=R):
+    d = Path(tempfile.mkdtemp())

+// Stop and walk: the lazy kernel's induction stopped once it has made ⌈β r⌉ cells, and the
+// rest of the block written by walking the stopped exchange with one lane (MONODROMY, #145
+// HANDOFF 6916839849; `docs/journal/monodromy/2026-11-16-stop-walk-reopened.md` on #485). Bench
+// arms only: the default kernel is unchanged. Included in `cells.rs`.
+//
+// **The stop.** `induce_lazy` counts its cells down from a budget on every cell already. The
+// arms start that counter at ⌈β r⌉ (r the block's letters) where that is below the budget, and
+// keep the rest aside. Once the counter is out, each further cell takes one from the reserve,
+// in the counter's cold path, until both fronts' letters are whole (the next fresh pair); then
+// the induction stops. No word past that moment is made: about a third of the word bytes at
+// β = 1.5 on Wikidata (the entry's §0). The test on every cell is the shipped one.
+//
+// **The stopped exchange.** At that moment no front holds part of a letter, and a front never
+// leaves a live letter, so every live letter is one no front has read: a cell, in its two
+// slots, or a letter of the block's own, at its stream place. Both orders are read from the
+// fronts' links and streams, without changing them, into tables in top order: each letter's
+// first point (as a bit, with a rank directory), the offset from its top interval to its
+// bottom one, its word's length and its word. T sends x in a letter's top interval to
+// x + offset. The domain [0, m) is the letters' rows, summed.
+//
+// **The walk (the entries' Lemmas 1 and 2).** For a point x of a letter whose word is w,
+// place(T x) = place(x) − |w| (mod n): w lies just below x's place. One lane walks T from x_μ,
+// the marked letter's one point, whose place the marked word's rules and `SHIFT` give, writing
+// each point's word ending at its place, downward. After m points it is back at x_μ, n bytes
+// lower: once round the block.
+// - *A point's stores reach below its word* (by up to `COPY` bytes), into places the walk
+//   has not reached yet, because places only fall. Only near place 0, where the walk wraps to
+//   n, and in its last `COPY` bytes, which lie just above where it began, is a word written
+//   exactly.
+// - *Each point copies its word from one place,* `COPY` bytes whatever its length and 16 at a
+//   time past that, top down, so the walk does not branch on the word: a word kept as bytes
+//   from the arena; a rule from where the walk first met the letter and expanded it. The two
+//   occurrences are disjoint and the first lies above, and the copy reads below the first
+//   place only where it writes below this one, which the walk writes over later.
+// - *The walk is one chain,* point to point: a rank, a popcount and the letter's offset,
+//   which is kept apart from the rest of its record so that the chain's loads stay in the
+//   first-level cache. It is latency-bound: about half its time at β = 1.5.
+//
+// **When it does not apply.** If the walk comes back to x_μ before m points or not n bytes
+// lower (the stopped exchange is several cycles: the block is a shorter cycle's copies), or
+// anything read does not add up, the block is decoded again by the shipped induction and
+// writer. If the induction ends before its stop, the writer is the shipped one.

+/// The walk: one lane from x_μ (`xmu`), whose word ends at place `p`, each point's word
+/// written ending at its place, downward, m points (`walk_stopped`). Err: it was not one
+/// cycle of n bytes. With the CPU's popcount where it has one (a point's letter is a popcount).
+#[allow(clippy::too_many_arguments)]
+fn walk(buf: &[u8], rules: usize, w: &mut Walk, xmu: usize, p: usize, n: usize, m: usize, out: &mut [u8]) -> Result<(), ()> {
+    #[cfg(all(target_arch = "x86_64", not(miri)))]
+    if std::arch::is_x86_feature_detected!("popcnt") {
+        // SAFETY: the CPU has popcnt
+        return unsafe { walk_popcnt(buf, rules, w, xmu, p, n, m, out) };
+    }
+    walk_body(buf, rules, w, xmu, p, n, m, out)

+/// (the walk's body, `walk`)
+///
+/// Places are counted on a line that goes once round: v from p + n down to p, the place
+/// being v − n while v > n (the first lap, down to place 0) and v after (the second, from
+/// place n down to p). A word is written with 16-byte stores where it lies at least 16 bytes
+/// above its lap's floor (0 in the first lap, p in the second), else exactly.
+#[inline(always)]
+#[allow(clippy::too_many_arguments)]
+fn walk_body(buf: &[u8], rules: usize, w: &mut Walk, xmu: usize, p: usize, n: usize, m: usize, out: &mut [u8]) -> Result<(), ()> {
+    if p >= n || xmu >= m || out.len() < n + 32 || w.recs.is_empty() {
+        return Err(());
+    }
+    let base = buf.as_ptr();
+    let o = out.as_mut_ptr();
+    let recs = w.recs.as_mut_ptr();
+    let dls = w.dls.as_ptr();
+    let mut stack = std::mem::take(&mut w.stack);
+    let mut tmp = std::mem::take(&mut w.tmp);
+    let mut st: Frame = [(0, 0); DEEP];
+    let mut v = p + n;
+    // `COPY` above the lap's floor on the line, and the line's offset from the place
+    let (mut lim, mut lap) = if v > n { (n + COPY, n) } else { (p + COPY, 0) };
+    let mut x = xmu as u32;
+    let mut points = 0usize;
+    // (T sends [0, m) into itself: `tables` checked each letter's two intervals lie in it)
+    let r = loop {
+        let i = w.letter(x);
+        // SAFETY: i is a letter of the tables (`letter`)
+        let (rec, dl) = unsafe { (recs.add(i), *dls.add(i)) };
+        let (src, len) = unsafe { *rec };
+        let len = len as usize;
+        if v >= lim + len {
+            let at = v - lap;
+            if src != 0 {
+                // SAFETY: the word's bytes end at `src`, with `COPY` readable bytes before it
+                // and its whole length: a word kept as bytes in the arena (the buffer's
+                // padding lies below it), else where the walk first wrote it, at least
+                // `COPY` + len above its lap's floor and above this point's word. The stores
+                // lie in [at − max(len, COPY) − 15, at), inside [0, n] above the floor. The
+                // first `COPY` bytes are read before any is stored
+                unsafe {
+                    let s = src as *const u8;
+                    let (a0, a1) = (ld16(s.sub(16)), ld16(s.sub(32)));
+                    st16(o.add(at - 16), a0);
+                    st16(o.add(at - 32), a1);
+                    let mut k = COPY;
+                    while k < len {
+                        k += 16;
+                        copy16(o.add(at - k), s.sub(k));
+                    }
+                }
+            } else {
+                // SAFETY: as above, and a rule's factors are the kernel's words
+                unsafe {
+                    (*rec).0 = o.add(at) as usize;
+                    emit(base, rules, *w.words.get_unchecked(i), at, o, &mut st, &mut stack);
+                }
+            }
+        } else {
+            if v < p + len {
+                break Err(());
+            }
+            let at = v - lap;
+            // SAFETY: i is a letter of the tables
+            let word = unsafe { *w.words.get_unchecked(i) };
+            emit_exact(buf, rules, word, len, at, n, out, &mut st, &mut stack, &mut tmp);
+            if v - len <= n {
+                (lim, lap) = (p + COPY, 0);
+            }
+        }
+        v -= len;
+        x = (x as i32 + dl) as u32;
+        points += 1;
+        if x as usize == xmu {
+            break if points == m && v == p { Ok(()) } else { Err(()) };
+        }
+    };
+    w.stack = stack;
+    w.tmp = tmp;
+    r

+/// The bytes each point copies whatever its word's length (a word longer copies the rest 16
+/// at a time): one fixed copy, so the walk does not branch on the length
+const COPY: usize = 32;
+
+/// 16 bytes, loaded and stored
+#[inline(always)]
+unsafe fn ld16(p: *const u8) -> [u8; 16] {
+    // SAFETY: the caller's
+    unsafe { (p as *const [u8; 16]).read_unaligned() }
+}
+#[inline(always)]
+unsafe fn st16(p: *mut u8, v: [u8; 16]) {
+    // SAFETY: the caller's
+    unsafe { (p as *mut [u8; 16]).write_unaligned(v) }

+/// Pending factors a word's expansion keeps before it spills to the heap
+const DEEP: usize = 32;
+type Frame = [(u64, usize); DEEP];
+
+/// A word (`h`: offset << 32 | length with the mark) written ending at `at` in `o`: a word
+/// kept as bytes by one 16-byte store ending there, a rule by its factors, right to left. The
+/// stores may reach 15 bytes below the word. Out of line: the walk's own loop is kept to
+/// its registers.
+///
+/// SAFETY: [at − |word| − 16, at) lies in `o`; every word's bytes or rule lie in the buffer
+/// at `base` (words kept as bytes below `rules`, with 16 bytes before each one's end).
+#[inline(never)]
+unsafe fn emit(base: *const u8, rules: usize, h: u64, at: usize, o: *mut u8, st: &mut Frame, stack: &mut Vec<(u64, usize)>) {
+    let off = (h >> 32) as usize;
+    if off < rules {
+        let len = (h as u32 & !MARK) as usize;
+        // SAFETY: the caller's
+        unsafe { copy16(o.add(at - 16), base.add(off + len - 16)) };
+        return;
+    }
+    // a rule: its factors right to left, on a stack in `st` (`DEEP` pending factors, the
+    // walk's, so not zeroed for each word), and past that on `stack`
+    let mut sp = 0usize;
+    let (mut h, mut at) = (h, at);
+    loop {
+        let off = (h >> 32) as usize;
+        if off < rules {
+            let len = (h as u32 & !MARK) as usize;
+            // SAFETY: the caller's
+            unsafe { copy16(o.add(at - 16), base.add(off + len - 16)) };
+            // the next pending factor: from the frame, else the heap
+            if sp > 0 {
+                sp -= 1;
+                // SAFETY: sp < DEEP
+                (h, at) = unsafe { *st.get_unchecked(sp) };
+            } else if let Some(v) = stack.pop() {
+                (h, at) = v;
+            } else {
+                return;
+            }
+        } else {
+            // SAFETY: a rule is 16 bytes at its offset
+            let (a, b) = unsafe { (rd64(base, off), rd64(base, off + 8)) };
+            let lb = (b as u32 & !MARK) as usize;
+            // B now (the right factor), A pending
+            if sp < DEEP {
+                // SAFETY: sp < DEEP
+                unsafe { *st.get_unchecked_mut(sp) = (a, at - lb) };
+                sp += 1;
+            } else {
+                // (deeper than the frame: the frame's pending factors, all newer than the
+                // heap's, go on top of it in order, so the order is kept; this is rare)
+                stack.extend_from_slice(&st[..]);
+                sp = 0;
+                st[sp] = (a, at - lb);
+                sp += 1;
+            }
+            h = b;
+        }
+    }
+}
+
+/// A word written exactly, ending at `at` (mod n) in `out`: expanded into `tmp` first
+#[allow(clippy::too_many_arguments)]
+fn emit_exact(
+    buf: &[u8],
+    rules: usize,
+    h: u64,
+    len: usize,
+    at: usize,
+    n: usize,
+    out: &mut [u8],
+    st: &mut Frame,
+    stack: &mut Vec<(u64, usize)>,
+    tmp: &mut Vec<u8>,
+) {
+    if tmp.len() < len + 16 {
+        tmp.resize(len + 16, 0);
+    }
+    // SAFETY: `tmp` is the word's length and 16 bytes before it; the words are the kernel's
+    unsafe { emit(buf.as_ptr(), rules, h, len + 16, tmp.as_mut_ptr(), st, stack) };
+    let start = (at + 2 * n - len) % n;
+    for (j, &v) in tmp[16..16 + len].iter().enumerate() {
+        out[(start + j) % n] = v;
+    }
+}

+fn compare(
+    z: &[u8],
+    arms: &mut (
+        ForcedStopLazyCells14,
+        ForcedStopLazyCells15,
+        ForcedStopLazyCells16,
+    ),
+) -> usize {

+    let mut syms = Vec::new();

+    let mut runs = Runs::default();
+    let mut blocks = 0;
+    while let Event::Block { symbols, index, .. } = sc.next(std::mem::take(&mut syms)).unwrap() {

+        let p = symbols.orig_ptr;
+        let mut want = Vec::new();
+        lazy.invert_runs(&runs, n, p, &mut want, k);
+        want.truncate(n);
+        let check = |beta: usize, got: Vec<u8>| {

+                got == want,

+        check(15, invert(&mut arms.1, &runs, n, p, &want));
+        check(16, invert(&mut arms.2, &runs, n, p, &want));
+        blocks += 1;
